59 // The set at p[i] ('[' already seen); its end in `end`, whether c is in it in `hit`; false if unclosed
60 constexpr bool match_set(std::string_view p, std::size_t i, char c, std::size_t& end, bool& hit) noexcept
61 {
62 std::size_t j = i;
63 bool negate = false;
64 if (j < p.size() && (p[j] == '!' || p[j] == '^')) {
65 negate = true;
66 ++j;
67 }
68 bool in = false;
69 bool first = true;
70 while (j < p.size() && (p[j] != ']' || first)) {
71 first = false;
72 if (j + 2 < p.size() && p[j + 1] == '-' && p[j + 2] != ']') {
73 in = in || (c >= p[j] && c <= p[j + 2]);
74 j += 3;
75 } else {
76 in = in || c == p[j];
77 ++j;
78 }
79 }
80 if (j >= p.size())
81 return false;
82 end = j + 1;
83 hit = in != negate;
84 return true;
85 }
86} // namespace detail

90[[nodiscard]] constexpr bool matches(std::string_view pattern, std::string_view path) noexcept
91{
92 if (!pattern.empty() && pattern.back() == '/') // a directory: everything below it
93 return path.starts_with(pattern);
94 std::size_t p = 0, s = 0, star = std::string_view::npos, resume = 0;
95 while (s < path.size()) {
96 if (p < pattern.size() && pattern[p] == '*') {
97 star = p++;
98 resume = s;
99 continue;
100 }
101 if (p < pattern.size()) {
102 if (pattern[p] == '?') {
103 ++p;
104 ++s;
105 continue;
106 }
107 if (pattern[p] == '[') {
108 std::size_t end = 0;
109 bool hit = false;
110 if (detail::match_set(pattern, p + 1, path[s], end, hit)) {
111 if (hit) {
112 p = end;
113 ++s;
114 continue;
115 }
116 } else if (path[s] == '[') { // unclosed: a literal
117 ++p;
118 ++s;
119 continue;
120 }
121 } else if (pattern[p] == path[s]) {
122 ++p;
123 ++s;
124 continue;
125 }
126 }
127 if (star == std::string_view::npos)
128 return false;
129 p = star + 1;
130 s = ++resume;
131 }
132 while (p < pattern.size() && pattern[p] == '*')
133 ++p;
134 return p == pattern.size();
135}

139[[nodiscard]] constexpr bool select(std::string_view path, std::span<const std::string> include, std::span<const std::string> exclude = {}) noexcept
140{
141 bool in = include.empty();
142 for (const auto& p : include)
143 in = in || matches(p, path);
144 for (const auto& p : exclude)
145 in = in && !matches(p, path);
146 return in;
147}
