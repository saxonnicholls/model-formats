114 [[nodiscard]] static hub_options from_environment(const config& env = environment_config())
115 {
116 hub_options o;
117 if (const std::string* e = env.raw("HF_ENDPOINT"); e && !e->empty())
118 o.endpoint = *e;
119 while (!o.endpoint.empty() && o.endpoint.back() == '/')
120 o.endpoint.pop_back();

122 if (!t || t->empty())

166 // Percent-encoding, as huggingface_hub's quote: unreserved characters and `safe` kept
167 inline std::string quote(std::string_view s, std::string_view safe)
168 {
169 static constexpr char hex[] = "0123456789ABCDEF";
170 std::string out;
171 for (const char c : s) {
172 const auto u = static_cast<unsigned char>(c);
173 if ((u >= 'a' && u <= 'z') || (u >= 'A' && u <= 'Z') || (u >= '0' && u <= '9') || c == '-' || c == '.' || c == '_' || c == '~'
174 || safe.find(c) != std::string_view::npos) {
175 out.push_back(c);
176 } else {
177 out.push_back('%');
178 out.push_back(hex[u >> 4]);
179 out.push_back(hex[u & 15]);
180 }
181 }
182 return out;
183 }
184 inline std::string origin_of(const std::string& url)
185 {
186 net::url u;
187 if (!net::parse_url(url, u))
188 return {};
189 return u.scheme + "://" + u.authority();
190 }
191 inline error status_error(int status, const std::string& what)
192 {
193 std::string why = "HTTP " + std::to_string(status);
194 if (status == 401 || status == 403)
195 why += ": the repository is private or gated (set HF_TOKEN), or the token lacks access";
196 else if (status == 404)
197 why += ": not found";
198 else if (status == 429)
199 why += ": rate limited; try again later";
200 return error{status == 404 ? errc::not_found : errc::http, what + ": " + why};
201 }

228 [[nodiscard]] result<repo_info> resolve(std::string_view repo, std::string_view revision = "main")
229 {
230 if (!valid_repo_id(repo) || !valid_revision(revision))
231 return fail(errc::bad_argument, "hub: " + quote_name(repo) + "@" + quote_name(revision) + " is not a repository and revision");
232 SNICHOLLS_TRY_ASSIGN(const cache c, cache::open(opt_.cache_dir, repo));
233 if (opt_.offline)
234 return resolve_offline(c, repo, revision);
235 const std::string url = opt_.endpoint + "/api/models/" + std::string(repo) + "/revision/" + detail::quote(revision, "") + "?blobs=true";
236 SNICHOLLS_TRY_ASSIGN(const http::response r, get(url, {}));
237 if (r.status != 200)
238 return err(detail::status_error(r.status, "hub: " + std::string(repo) + "@" + std::string(revision)));
239 SNICHOLLS_TRY_ASSIGN(const formats::detail::json j, formats::detail::parse_json(r.body, "hub API answer"));
240 repo_info info;
241 info.repo = repo;
242 info.revision = revision;
243 const std::string* sha = formats::detail::string_at(j, "sha");
244 const formats::detail::json* siblings = formats::detail::member(j, "siblings");
245 if (!sha || !is_commit(*sha) || !siblings || !siblings->is_array())
246 return fail(errc::bad_json, "hub: the API answer has no commit sha and siblings");
247 info.commit = *sha;
248 for (const auto& s : *siblings) {
249 repo_file f;
250 const std::string* name = formats::detail::string_at(s, "rfilename");
251 const std::string* blob = formats::detail::string_at(s, "blobId");
252 if (!name || !safetensors::safe_relative_name(*name))
253 return fail(errc::bad_argument, "hub: the API lists an unsafe file name");
254 f.path = *name;
255 f.size = formats::detail::u64_at(s, "size").value_or(0);
256 f.blob_id = blob ? *blob : std::string();
257 if (const formats::detail::json* lfs = formats::detail::member(s, "lfs"); lfs && lfs->is_object()) {
258 if (const std::string* h = formats::detail::string_at(*lfs, "sha256"))
259 f.sha256 = *h;
260 f.size = formats::detail::u64_at(*lfs, "size").value_or(f.size);
261 }
262 if (!safetensors::safe_relative_name(f.etag()) || f.etag().size() < 40)
263 return fail(errc::bad_json, "hub: " + quote_name(f.path) + " has no usable blob id");
264 info.files.push_back(std::move(f));
265 }
266 if (revision != info.commit)
267 SNICHOLLS_TRY_VOID(c.write_ref(revision, info.commit));
268 return ok(std::move(info));
269 }

274 [[nodiscard]] result<download_report> download(const repo_info& info, std::span<const std::string> include = {},
275 std::span<const std::string> exclude = {})
276 {
277 SNICHOLLS_TRY_ASSIGN(const cache c, cache::open(opt_.cache_dir, info.repo));
278 download_report report;
279 report.snapshot = c.snapshot(info.commit);
280 std::vector<const repo_file*> todo;
281 std::uint64_t needed = 0;
282 for (const auto& f : info.files) {
283 if (!select(f.path, include, exclude))
284 continue;
285 if (c.has(info.commit, f.path)) {
286 report.cached.push_back(f.path);
287 continue;
288 }
289 if (std::filesystem::exists(c.blob(f.etag()))) { // the same blob under another commit
290 SNICHOLLS_TRY_VOID(c.link(info.commit, f.path, f.etag()));
291 report.cached.push_back(f.path);
292 continue;
293 }
294 if (opt_.offline)
295 return fail(errc::offline, "hub: " + f.path + " is not in the cache");
296 todo.push_back(&f);
297 needed += f.size;
298 }
299 if (opt_.max_download_bytes && needed > opt_.max_download_bytes)
300 return fail(errc::too_large, "hub: " + std::to_string(needed) + " bytes to download, over the cap of " + std::to_string(opt_.max_download_bytes));
301 std::filesystem::create_directories(c.snapshot(info.commit));
302 if (todo.empty())
303 return ok(std::move(report));
304 SNICHOLLS_TRY_VOID(fetch_all(c, info, todo, report));
305 return ok(std::move(report));
306 }

321 result<repo_info> resolve_offline(const cache& c, std::string_view repo, std::string_view revision) const
322 {
323 SNICHOLLS_TRY_ASSIGN(const std::string commit, c.resolve_offline(revision));
324 repo_info info{std::string(repo), std::string(revision), commit, {}};
325 std::error_code ec;
326 const auto root = c.snapshot(commit);
327 for (auto it = std::filesystem::recursive_directory_iterator(root, ec); !ec && it != std::filesystem::recursive_directory_iterator();
328 it.increment(ec)) {
329 if (!it->is_symlink() && !it->is_regular_file())
330 continue;
331 repo_file f;
332 f.path = it->path().lexically_relative(root).generic_string();
333 const auto target = std::filesystem::read_symlink(it->path(), ec);
334 f.blob_id = ec ? std::string() : target.filename().string();
335 ec.clear();
336 f.size = std::filesystem::file_size(it->path(), ec);
337 ec.clear();
338 info.files.push_back(std::move(f));
339 }
340 return ok(std::move(info));
341 }

363 // One request on its own loop, following redirects
364 result<http::response> get(std::string url, std::vector<std::pair<std::string, std::string>> extra)
365 {
366 for (int hop = 0; hop <= 10; ++hop) {
367 auto headers = headers_for(url);
368 headers.insert(headers.end(), extra.begin(), extra.end());
369 auto r = http::fetch({"GET", url, std::move(headers), {}, opt_.timeout}, http_config());
370 if (r.is_err())
371 return fail(errc::http, "hub: " + detail::origin_of(url) + ": " + r.unwrap_err_unchecked().what());
372 const int s = r->status;
373 if (s == 301 || s == 302 || s == 303 || s == 307 || s == 308) {
374 const std::string* loc = r->header("location");
375 if (!loc)
376 return fail(errc::http, "hub: a redirect without a Location");
377 url = loc->starts_with("http://") || loc->starts_with("https://") ? *loc : detail::origin_of(url) + *loc;
378 continue;
379 }
380 return ok(std::move(*r));
381 }
382 return fail(errc::http, "hub: more than 10 redirects");
383 }

385 struct job
386 {
387 const repo_file* file = nullptr;
388 std::string url; // the resolve URL, then the redirect's target
389 std::uint64_t offset = 0;
390 int hops = 0;
391 mapped_region out;
392 };

394 result<void> start(const cache& c, const repo_info& info, job& j, download_report& report)
395 {
396 std::error_code ec;
397 std::filesystem::create_directories(c.blob("").parent_path(), ec);
398 const auto part = c.incomplete(j.file->etag());
399 const auto existing = std::filesystem::exists(part, ec) ? std::filesystem::file_size(part, ec) : 0;
400 j.offset = std::min<std::uint64_t>(existing / opt_.chunk_bytes * opt_.chunk_bytes, j.file->size); // whole ranges only
401 report.resumed += j.offset;
402 auto r = mapped_region::open(part.string(), {.reserve = std::max<std::size_t>(static_cast<std::size_t>(j.file->size), 1), .create = true,
403 .truncate = false, .preallocate = false, .mode = 0644});
404 if (r.is_err())
405 return err(io::io_error(part, "cannot open", r.unwrap_err_unchecked()));
406 j.out = std::move(*r);
407 j.url = opt_.endpoint + "/" + info.repo + "/resolve/" + info.commit + "/" + detail::quote(j.file->path, "/");
408 return ok();
409 }

411 result<void> finish(const cache& c, const repo_info& info, job& j)
412 {
413 const int fd = j.out.fd();
414 if (auto s = j.out.flush(); s.is_err())
415 return err(io::io_error(c.incomplete(j.file->etag()), "cannot sync", s.unwrap_err_unchecked()));
416 if (::ftruncate(fd, static_cast<off_t>(j.file->size)) != 0 || snicholls::full_sync(fd).is_err())
417 return err(io::io_error(c.incomplete(j.file->etag()), "cannot finish", std::error_code(errno, std::generic_category())));
418 j.out = mapped_region{};
419 const auto part = c.incomplete(j.file->etag());
420 SNICHOLLS_TRY_ASSIGN(const std::string got, file_digest_hex(part, !j.file->sha256));
421 if (got != j.file->etag()) {
422 std::error_code ec;
423 std::filesystem::remove(part, ec);
424 return fail(errc::checksum_mismatch, "hub: " + j.file->path + " has digest " + got + ", the Hub says " + j.file->etag());
425 }
426 std::error_code ec;
427 std::filesystem::rename(part, c.blob(j.file->etag()), ec);
428 if (ec)
429 return err(io::io_error(c.blob(j.file->etag()), "cannot rename onto", ec));
430 return c.link(info.commit, j.file->path, j.file->etag());
431 }

433 result<void> fetch_all(const cache& c, const repo_info& info, const std::vector<const repo_file*>& files, download_report& report)
434 {
435 event_loop loop;
436 std::optional<error> failure;
437 std::deque<job> jobs;
438 std::size_t next = 0, active = 0;
439 {
440 http::client net(loop, http_config());
441 std::function<void()> start_more;
442 std::function<void(job&)> request;
443 const auto done = [&](job& j, std::optional<error> e) {
444 if (e && !failure)
445 failure = std::move(e);
446 if (!e)
447 report.downloaded.push_back(j.file->path);
448 j.out = mapped_region{};
449 --active;
450 start_more();
451 if (active == 0)
452 loop.stop();
453 };
454 request = [&](job& j) {
455 if (j.offset >= j.file->size) {
456 auto r = finish(c, info, j);
457 done(j, r.is_err() ? std::optional<error>(r.unwrap_err_unchecked()) : std::nullopt);
458 return;
459 }
460 const std::uint64_t last = std::min<std::uint64_t>(j.offset + opt_.chunk_bytes, j.file->size) - 1;
461 auto headers = headers_for(j.url);
462 headers.emplace_back("Range", "bytes=" + std::to_string(j.offset) + "-" + std::to_string(last));
463 net.request({"GET", j.url, std::move(headers), {}, opt_.timeout}, [&, jp = &j](http::client_result r) {
464 job& j = *jp;
465 if (r.is_err())
466 return done(j, error{errc::http, "hub: " + j.file->path + ": " + r.unwrap_err_unchecked().what()});
467 const int s = r->status;
468 if (s == 301 || s == 302 || s == 303 || s == 307 || s == 308) {
469 const std::string* loc = r->header("location");
470 if (!loc || ++j.hops > 10)
471 return done(j, error{errc::http, "hub: " + j.file->path + ": a bad or endless redirect"});
472 j.url = loc->starts_with("http://") || loc->starts_with("https://") ? *loc : detail::origin_of(j.url) + *loc;
473 return request(j);
474 }
475 if (s == 416 && j.offset == j.file->size)
476 return request(j);
477 if (s != 200 && s != 206)
478 return done(j, detail::status_error(s, "hub: " + j.file->path));
479 if (s == 200 && j.offset > 0) // the server ignored the range: start again from the beginning
480 j.offset = 0;
481 const auto& body = r->body;
482 if (body.empty() || j.offset + body.size() > j.file->size)
483 return done(j, error{errc::http, "hub: " + j.file->path + ": a range of the wrong length"});
484 if (auto g = j.out.grow_or_move(static_cast<std::size_t>(j.offset + body.size())); g.is_err())
485 return done(j, io::io_error(c.incomplete(j.file->etag()), "cannot grow", g.unwrap_err_unchecked()));
486 std::memcpy(j.out.data() + j.offset, body.data(), body.size());
487 (void)j.out.flush_async(static_cast<std::size_t>(j.offset), body.size());
488 j.offset += body.size();
489 report.bytes += body.size();
490 request(j);
491 });
492 };
493 start_more = [&] {
494 while (!failure && active < opt_.parallel && next < files.size()) {
495 jobs.emplace_back();
496 job& j = jobs.back();
497 j.file = files[next++];
498 ++active;
499 if (auto s = start(c, info, j, report); s.is_err()) {
500 done(j, s.unwrap_err_unchecked());
501 continue;
502 }
503 loop.post([&, jp = &j] { request(*jp); });
504 }
505 };
506 start_more();
507 if (active > 0)
508 loop.run();
509 }
510 if (failure)
511 return err(std::move(*failure));
512 return ok();
513 }
