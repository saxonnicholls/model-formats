7// GPT-2's byte-level alphabet (`bytes_to_unicode` in its encoder.py, and Hugging Face's ByteLevel pre-tokenizer):
8// each of the 256 bytes is one printable code point, so a vocabulary of byte strings can be written as text. The
9// bytes '!'..'~', 0xA1..0xAC and 0xAE..0xFF are their own code points; the other 68 are 256, 257, ... in byte order.
10// Byte-level BPE is core's (ml/bpe.hpp), so this alphabet belongs beside it; it moves there when core takes it
11// (proposal in family/proposals/model-formats.md).
12//

25constexpr bool byte_is_printable(unsigned b) noexcept
26{
27 return (b >= 0x21 && b <= 0x7e) || (b >= 0xa1 && b <= 0xac) || (b >= 0xae && b <= 0xff);
28}

30// The code point of each byte
31inline constexpr std::array<std::uint16_t, 256> byte_to_code_point = [] {
32 std::array<std::uint16_t, 256> t{};
33 unsigned next = 256;
34 for (unsigned b = 0; b < 256; ++b)
35 t[b] = static_cast<std::uint16_t>(byte_is_printable(b) ? b : next++);
36 return t;
37}();
38
39// The bytes of code points 256 to 323, in order
40inline constexpr std::array<std::uint8_t, 68> high_code_point_byte = [] {
41 std::array<std::uint8_t, 68> t{};
42 for (unsigned b = 0; b < 256; ++b)
43 if (byte_to_code_point[b] >= 256)
44 t[byte_to_code_point[b] - 256] = static_cast<std::uint8_t>(b);
45 return t;
46}();

48// The byte of a code point of the alphabet, or none
49constexpr std::optional<std::uint8_t> code_point_to_byte(std::uint32_t cp) noexcept
50{
51 if (cp < 256)
52 return byte_is_printable(cp) ? std::optional<std::uint8_t>(static_cast<std::uint8_t>(cp)) : std::nullopt;
53 if (cp < 256 + high_code_point_byte.size())
54 return high_code_point_byte[cp - 256];
55 return std::nullopt;
56}

58inline void append_utf8(std::string& out, std::uint32_t cp)
59{
60 if (cp < 0x80) {
61 out.push_back(static_cast<char>(cp));
62 } else if (cp < 0x800) {
63 out.push_back(static_cast<char>(0xc0 | (cp >> 6)));
64 out.push_back(static_cast<char>(0x80 | (cp & 0x3f)));
65 } else if (cp < 0x10000) {
66 out.push_back(static_cast<char>(0xe0 | (cp >> 12)));
67 out.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3f)));
68 out.push_back(static_cast<char>(0x80 | (cp & 0x3f)));
69 } else {
70 out.push_back(static_cast<char>(0xf0 | (cp >> 18)));
71 out.push_back(static_cast<char>(0x80 | ((cp >> 12) & 0x3f)));
72 out.push_back(static_cast<char>(0x80 | ((cp >> 6) & 0x3f)));
73 out.push_back(static_cast<char>(0x80 | (cp & 0x3f)));
74 }
75}

77// The next code point of UTF-8 text at `at`, advancing it; none if malformed
78constexpr std::optional<std::uint32_t> next_code_point(std::string_view s, std::size_t& at) noexcept
79{
80 const auto byte = [&](std::size_t i) { return static_cast<unsigned char>(s[i]); };
81 const unsigned c = byte(at);
82 std::size_t n = c < 0x80 ? 1 : (c >> 5) == 6 ? 2 : (c >> 4) == 14 ? 3 : (c >> 3) == 30 ? 4 : 0;
83 if (n == 0 || at + n > s.size())
84 return std::nullopt;
85 std::uint32_t cp = n == 1 ? c : c & (0x7f >> n);
86 for (std::size_t i = 1; i < n; ++i) {
87 if ((byte(at + i) & 0xc0) != 0x80)
88 return std::nullopt;
89 cp = (cp << 6) | (byte(at + i) & 0x3f);
90 }
91 at += n;
92 return cp;
93}

95// Byte-level text to bytes ("Ġhello" to " hello"); none if a code point is outside the alphabet
96inline std::optional<std::string> byte_level_decode(std::string_view text)
97{
98 std::string out;
99 out.reserve(text.size());
100 for (std::size_t at = 0; at < text.size();) {
101 const auto cp = next_code_point(text, at);
102 if (!cp)
103 return std::nullopt;
104 const auto b = code_point_to_byte(*cp);
105 if (!b)
106 return std::nullopt;
107 out.push_back(static_cast<char>(*b));
108 }
109 return out;
110}

112// Bytes to byte-level text
113inline std::string byte_level_encode(std::string_view bytes)
114{
115 std::string out;
116 out.reserve(bytes.size() * 2);
117 for (const char c : bytes)
118 append_utf8(out, byte_to_code_point[static_cast<unsigned char>(c)]);
119 return out;
120}
