74struct tensor_shape
75{
76 std::array<std::uint64_t, max_rank> dims{};
77 std::size_t rank = 0;
78
79 constexpr tensor_shape() noexcept = default;
81 constexpr tensor_shape(std::initializer_list<std::uint64_t> d) noexcept
82 {
83 for (const auto e : d)
84 if (!push_back(e))
85 break;
86 }
87
89 constexpr bool push_back(std::uint64_t e) noexcept
90 {
91 if (rank == max_rank)
92 return false;
93 dims[rank++] = e;
94 return true;
95 }
96
97 [[nodiscard]] constexpr std::span<const std::uint64_t> view() const noexcept { return {dims.data(), rank}; }
99 [[nodiscard]] constexpr std::uint64_t operator[](std::size_t i) const noexcept { return dims[i]; }
101 [[nodiscard]] constexpr std::optional<std::uint64_t> elements() const noexcept
102 {
103 std::uint64_t n = 1;
104 for (std::size_t i = 0; i < rank; ++i) {
105 const auto m = checked_mul(n, dims[i]);
106 if (!m)
107 return std::nullopt;
108 n = *m;
109 }
110 return n;
111 }
112
113 [[nodiscard]] constexpr tensor_shape reversed() const noexcept
114 {
115 tensor_shape r;
116 for (std::size_t i = rank; i > 0; --i)
117 r.push_back(dims[i - 1]);
118 return r;
119 }
120
121 friend constexpr bool operator==(const tensor_shape& a, const tensor_shape& b) noexcept
122 {
123 if (a.rank != b.rank)
124 return false;
125 for (std::size_t i = 0; i < a.rank; ++i)
126 if (a.dims[i] != b.dims[i])
127 return false;
128 return true;
129 }
130};
131
134[[nodiscard]] constexpr std::optional<std::uint64_t> checked_byte_size(const tensor_shape& shape, dtype t) noexcept
135{
136 const auto n = shape.elements();
137 if (!n)
138 return std::nullopt;
139 return checked_mul(*n, std::uint64_t{size_of(t)});
140}
141
143[[nodiscard]] inline std::string to_string(const tensor_shape& s)
144{
145 std::string out = "[";
146 for (std::size_t i = 0; i < s.rank; ++i) {
147 if (i)
148 out += ", ";
149 out += std::to_string(s.dims[i]);
150 }
151 return out + "]";
152}

164struct tensor
165{
166 std::string name;
167 dtype type = dtype::f32;
168 tensor_shape shape;
169 std::vector<std::byte> data;
170
172 [[nodiscard]] tensor_view view() const noexcept { return {name, type, shape, data}; }
173
175 [[nodiscard]] static tensor from(const tensor_view& v)
176 {
177 return {std::string(v.name), v.type, v.shape, std::vector<std::byte>(v.data.begin(), v.data.end())};
178 }
179};
180
182[[nodiscard]] inline std::vector<tensor_view> views(std::span<const tensor> ts)
183{
184 std::vector<tensor_view> v;
185 v.reserve(ts.size());
186 for (const auto& t : ts)
187 v.push_back(t.view());
188 return v;
