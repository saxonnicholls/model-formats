134[[nodiscard]] constexpr bool allowed_global(std::string_view module, std::string_view name) noexcept
135{
136 if (module == "collections")
137 return name == "OrderedDict";
138 if (module == "torch._utils")
139 return name == "_rebuild_tensor_v2" || name == "_rebuild_parameter";
140 if (module == "torch") {
141 constexpr std::string_view storages[] = {"DoubleStorage", "FloatStorage", "HalfStorage", "BFloat16Storage", "LongStorage",
142 "IntStorage", "ShortStorage", "CharStorage", "ByteStorage", "BoolStorage",
143 "Float8_e4m3fnStorage", "Float8_e5m2Storage"};
144 for (const auto s : storages)
145 if (s == name)
146 return true;
147 }
148 return false;
149}

153[[nodiscard]] inline result<pickled> unpickle(std::span<const std::byte> bytes, const unpickle_limits& limits = {})
154{
155 pickled p;
156 std::vector<std::size_t> stack, memo;
157 std::size_t at = 0;
158 const auto need = [&](std::size_t n) { return n <= bytes.size() - at; };
159 const auto make = [&](object o) -> result<std::size_t> {
160 if (p.objects.size() >= limits.max_objects)
161 return fail(errc::too_large, "pickle: more than " + std::to_string(limits.max_objects) + " objects");
162 p.objects.push_back(std::move(o));
163 return ok(p.objects.size() - 1);
164 };
165 const auto push = [&](std::size_t i) -> result<void> {
166 if (stack.size() >= limits.max_stack)
167 return fail(errc::too_large, "pickle: the stack is deeper than " + std::to_string(limits.max_stack));
168 stack.push_back(i);
169 return ok();
170 };
171 const auto pop = [&]() -> result<std::size_t> {
172 if (stack.empty() || p.objects[stack.back()].kind == kind::mark)
173 return fail(errc::bad_header, "pickle: the stack is empty");
174 const std::size_t i = stack.back();
175 stack.pop_back();
176 return ok(i);
177 };
178 // The items above the topmost MARK, and the MARK removed
179 const auto pop_mark = [&]() -> result<std::vector<std::size_t>> {
180 std::size_t m = stack.size();
181 while (m > 0 && p.objects[stack[m - 1]].kind != kind::mark)
182 --m;
183 if (m == 0)
184 return fail(errc::bad_header, "pickle: no MARK on the stack");
185 std::vector<std::size_t> items(stack.begin() + static_cast<std::ptrdiff_t>(m), stack.end());
186 stack.resize(m - 1);
187 return ok(std::move(items));
188 };
189 const auto memo_put = [&](std::uint64_t idx) -> result<void> {
190 if (stack.empty())
191 return fail(errc::bad_header, "pickle: PUT on an empty stack");
192 if (idx >= limits.max_memo)
193 return fail(errc::too_large, "pickle: memo index " + std::to_string(idx));
194 if (memo.size() <= idx)
195 memo.resize(static_cast<std::size_t>(idx) + 1, static_cast<std::size_t>(-1));
196 memo[static_cast<std::size_t>(idx)] = stack.back();
197 return ok();
198 };
199 const auto memo_get = [&](std::uint64_t idx) -> result<void> {
200 if (idx >= memo.size() || memo[static_cast<std::size_t>(idx)] == static_cast<std::size_t>(-1))
201 return fail(errc::bad_header, "pickle: GET of memo " + std::to_string(idx) + ", never PUT");
202 return push(memo[static_cast<std::size_t>(idx)]);
203 };
204 const auto text = [&](std::size_t n, kind k) -> result<void> {
205 if (!need(n))
206 return fail(errc::truncated, "pickle: a string runs past the end");
207 object o;
208 o.kind = k;
209 o.text.assign(reinterpret_cast<const char*>(bytes.data() + at), n);
210 at += n;
211 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(o)));
212 return push(i);
213 };
214 const auto integer = [&](std::int64_t v) -> result<void> {
215 object o;
216 o.kind = kind::integer;
217 o.integer = v;
218 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(o)));
219 return push(i);
220 };
221 const auto container = [&](kind k, std::vector<std::size_t> items) -> result<void> {
222 object o;
223 o.kind = k;
224 o.items = std::move(items);
225 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(o)));
226 return push(i);
227 };
228 const auto global = [&](std::string_view module, std::string_view name) -> result<void> {
229 if (!allowed_global(module, name))
230 return fail(errc::unsafe_pickle, "pickle: global " + quote_name(std::string(module) + "." + std::string(name)) + " is not allowed");
231 object o;
232 o.kind = kind::global;
233 o.text = std::string(module) + "." + std::string(name);
234 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(o)));
235 return push(i);
236 };
237 const auto line = [&]() -> std::optional<std::string_view> {
238 const std::string_view rest(reinterpret_cast<const char*>(bytes.data() + at), bytes.size() - at);
239 const auto nl = rest.find('\n');
240 if (nl == std::string_view::npos || nl > 256)
241 return std::nullopt;
242 at += nl + 1;
243 return rest.substr(0, nl);
244 };
245 while (true) {
246 if (!need(1))
247 return fail(errc::truncated, "pickle: no STOP");
248 const auto op = std::to_integer<std::uint8_t>(bytes[at++]);
249 switch (op) {
250 case 0x80: // PROTO
251 if (!need(1))
252 return fail(errc::truncated, "pickle: PROTO without a version");
253 if (const auto v = std::to_integer<int>(bytes[at++]); v < 2 || v > 5)
254 return fail(errc::unsafe_pickle, "pickle: protocol " + std::to_string(v) + " is not read");
255 break;
256 case 0x95: // FRAME: the frame's length; the opcodes follow inline
257 if (!need(8))
258 return fail(errc::truncated, "pickle: FRAME without a length");
259 at += 8;
260 break;
261 case '.': { // STOP
262 SNICHOLLS_TRY_ASSIGN(p.root, pop());
263 return ok(std::move(p));
264 }
265 case '(': { // MARK
266 object m;
267 m.kind = kind::mark;
268 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(m)));
269 SNICHOLLS_TRY_VOID(push(i));
270 break;
271 }
272 case 'p': { // PUT (text index)
273 const auto l = line();
274 std::uint64_t idx = 0;
275 if (!l || std::from_chars(l->data(), l->data() + l->size(), idx).ec != std::errc{})
276 return fail(errc::bad_header, "pickle: a malformed PUT");
277 SNICHOLLS_TRY_VOID(memo_put(idx));
278 break;
279 }
280 case 'q': // BINPUT
281 if (!need(1))
282 return fail(errc::truncated, "pickle: BINPUT");
283 SNICHOLLS_TRY_VOID(memo_put(std::to_integer<std::uint8_t>(bytes[at++])));
284 break;
285 case 'r': // LONG_BINPUT
286 if (!need(4))
287 return fail(errc::truncated, "pickle: LONG_BINPUT");
288 SNICHOLLS_TRY_VOID(memo_put(load_le<std::uint32_t>(bytes.data() + at)));
289 at += 4;
290 break;
291 case 0x94: // MEMOIZE
292 SNICHOLLS_TRY_VOID(memo_put(memo.size()));
293 break;
294 case 'g': { // GET (text index)
295 const auto l = line();
296 std::uint64_t idx = 0;
297 if (!l || std::from_chars(l->data(), l->data() + l->size(), idx).ec != std::errc{})
298 return fail(errc::bad_header, "pickle: a malformed GET");
299 SNICHOLLS_TRY_VOID(memo_get(idx));
300 break;
301 }
302 case 'h': // BINGET
303 if (!need(1))
304 return fail(errc::truncated, "pickle: BINGET");
305 SNICHOLLS_TRY_VOID(memo_get(std::to_integer<std::uint8_t>(bytes[at++])));
306 break;
307 case 'j': // LONG_BINGET
308 if (!need(4))
309 return fail(errc::truncated, "pickle: LONG_BINGET");
310 SNICHOLLS_TRY_VOID(memo_get(load_le<std::uint32_t>(bytes.data() + at)));
311 at += 4;
312 break;
313 case 'N': {
314 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(object{}));
315 SNICHOLLS_TRY_VOID(push(i));
316 break;
317 }
318 case 0x88: // NEWTRUE
319 case 0x89: { // NEWFALSE
320 object o;
321 o.kind = kind::boolean;
322 o.integer = op == 0x88 ? 1 : 0;
323 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(o)));
324 SNICHOLLS_TRY_VOID(push(i));
325 break;
326 }
327 case 'J': // BININT
328 if (!need(4))
329 return fail(errc::truncated, "pickle: BININT");
330 SNICHOLLS_TRY_VOID(integer(static_cast<std::int32_t>(load_le<std::uint32_t>(bytes.data() + at))));
331 at += 4;
332 break;
333 case 'K': // BININT1
334 if (!need(1))
335 return fail(errc::truncated, "pickle: BININT1");
336 SNICHOLLS_TRY_VOID(integer(std::to_integer<std::uint8_t>(bytes[at++])));
337 break;
338 case 'M': // BININT2
339 if (!need(2))
340 return fail(errc::truncated, "pickle: BININT2");
341 SNICHOLLS_TRY_VOID(integer(load_le<std::uint16_t>(bytes.data() + at)));
342 at += 2;
343 break;
344 case 0x8a: { // LONG1: little-endian two's complement, at most 8 bytes here
345 if (!need(1))
346 return fail(errc::truncated, "pickle: LONG1");
347 const std::size_t n = std::to_integer<std::uint8_t>(bytes[at++]);
348 if (n > 8 || !need(n))
349 return fail(n > 8 ? errc::unsupported : errc::truncated, "pickle: a LONG1 of " + std::to_string(n) + " bytes");
350 std::uint64_t v = 0;
351 for (std::size_t k = 0; k < n; ++k)
352 v |= std::uint64_t{std::to_integer<std::uint8_t>(bytes[at + k])} << (8 * k);
353 if (n > 0 && n < 8 && (std::to_integer<std::uint8_t>(bytes[at + n - 1]) & 0x80))
354 v |= ~std::uint64_t{0} << (8 * n); // sign-extend
355 at += n;
356 SNICHOLLS_TRY_VOID(integer(static_cast<std::int64_t>(v)));
357 break;
358 }
359 case 'G': { // BINFLOAT: big-endian
360 if (!need(8))
361 return fail(errc::truncated, "pickle: BINFLOAT");
362 object o;
363 o.kind = kind::floating;
364 o.floating = std::bit_cast<double>(load_be<std::uint64_t>(bytes.data() + at));
365 at += 8;
366 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(o)));
367 SNICHOLLS_TRY_VOID(push(i));
368 break;
369 }
370 case 'X': // BINUNICODE
371 case 'B': // BINBYTES
372 if (!need(4))
373 return fail(errc::truncated, "pickle: a string length");
374 at += 4;
375 SNICHOLLS_TRY_VOID(text(load_le<std::uint32_t>(bytes.data() + at - 4), op == 'X' ? kind::string : kind::bytes));
376 break;
377 case 0x8c: // SHORT_BINUNICODE
378 case 'C': // SHORT_BINBYTES
379 if (!need(1))
380 return fail(errc::truncated, "pickle: a string length");
381 ++at;
382 SNICHOLLS_TRY_VOID(text(std::to_integer<std::uint8_t>(bytes[at - 1]), op == 'C' ? kind::bytes : kind::string));
383 break;
384 case 0x8d: { // BINUNICODE8
385 if (!need(8))
386 return fail(errc::truncated, "pickle: a string length");
387 const std::uint64_t n = load_le<std::uint64_t>(bytes.data() + at);
388 at += 8;
389 if (n > bytes.size())
390 return fail(errc::truncated, "pickle: a string runs past the end");
391 SNICHOLLS_TRY_VOID(text(static_cast<std::size_t>(n), kind::string));
392 break;
393 }
394 case ')':
395 SNICHOLLS_TRY_VOID(container(kind::tuple, {}));
396 break;
397 case ']':
398 SNICHOLLS_TRY_VOID(container(kind::list, {}));
399 break;
400 case '}':
401 SNICHOLLS_TRY_VOID(container(kind::dict, {}));
402 break;
403 case 't': {
404 SNICHOLLS_TRY_ASSIGN(std::vector<std::size_t> items, pop_mark());
405 SNICHOLLS_TRY_VOID(container(kind::tuple, std::move(items)));
406 break;
407 }
408 case 0x85:
409 case 0x86:
410 case 0x87: {
411 const std::size_t n = op - 0x84u;
412 std::vector<std::size_t> items(n);
413 for (std::size_t k = n; k > 0; --k) {
414 SNICHOLLS_TRY_ASSIGN(items[k - 1], pop());
415 }
416 SNICHOLLS_TRY_VOID(container(kind::tuple, std::move(items)));
417 break;
418 }
419 case 'a': { // APPEND
420 SNICHOLLS_TRY_ASSIGN(const std::size_t v, pop());
421 if (stack.empty() || p.objects[stack.back()].kind != kind::list)
422 return fail(errc::bad_header, "pickle: APPEND to something not a list");
423 p.objects[stack.back()].items.push_back(v);
424 break;
425 }
426 case 'e': { // APPENDS
427 SNICHOLLS_TRY_ASSIGN(std::vector<std::size_t> items, pop_mark());
428 if (stack.empty() || p.objects[stack.back()].kind != kind::list)
429 return fail(errc::bad_header, "pickle: APPENDS to something not a list");
430 auto& l = p.objects[stack.back()].items;
431 l.insert(l.end(), items.begin(), items.end());
432 break;
433 }
434 case 's': { // SETITEM
435 SNICHOLLS_TRY_ASSIGN(const std::size_t v, pop());
436 SNICHOLLS_TRY_ASSIGN(const std::size_t k, pop());
437 if (stack.empty() || p.objects[stack.back()].kind != kind::dict)
438 return fail(errc::bad_header, "pickle: SETITEM on something not a dict");
439 p.objects[stack.back()].items.push_back(k);
440 p.objects[stack.back()].items.push_back(v);
441 break;
442 }
443 case 'u': { // SETITEMS
444 SNICHOLLS_TRY_ASSIGN(std::vector<std::size_t> items, pop_mark());
445 if (items.size() % 2 != 0 || stack.empty() || p.objects[stack.back()].kind != kind::dict)
446 return fail(errc::bad_header, "pickle: SETITEMS on something not a dict, or an odd count");
447 auto& d = p.objects[stack.back()].items;
448 d.insert(d.end(), items.begin(), items.end());
449 break;
450 }
451 case 'c': { // GLOBAL: module\nname\n
452 const auto module = line();
453 const auto name = module ? line() : std::nullopt;
454 if (!module || !name)
455 return fail(errc::truncated, "pickle: a malformed GLOBAL");
456 SNICHOLLS_TRY_VOID(global(*module, *name));
457 break;
458 }
459 case 0x93: { // STACK_GLOBAL
460 SNICHOLLS_TRY_ASSIGN(const std::size_t n, pop());
461 SNICHOLLS_TRY_ASSIGN(const std::size_t m, pop());
462 if (p.objects[m].kind != kind::string || p.objects[n].kind != kind::string)
463 return fail(errc::bad_header, "pickle: STACK_GLOBAL of something not strings");
464 const std::string module = p.objects[m].text, name = p.objects[n].text;
465 SNICHOLLS_TRY_VOID(global(module, name));
466 break;
467 }
468 case 'Q': { // BINPERSID: ('storage', type, key, location, numel)
469 SNICHOLLS_TRY_ASSIGN(const std::size_t pid, pop());
470 const object& t = p.objects[pid];
471 const auto is = [&](std::size_t k, kind want) { return t.items.size() > k && p.objects[t.items[k]].kind == want; };
472 if (t.kind != kind::tuple || t.items.size() != 5 || !is(0, kind::string) || p.objects[t.items[0]].text != "storage"
473 || !is(1, kind::global) || (!is(2, kind::string) && !is(2, kind::integer)) || !is(4, kind::integer))
474 return fail(errc::bad_header, "pickle: a persistent id that is not ('storage', type, key, location, numel)");
475 object s;
476 s.kind = kind::storage;
477 const std::string& type = p.objects[t.items[1]].text;
478 s.text = type.substr(type.rfind('.') + 1);
479 s.key = p.objects[t.items[2]].kind == kind::string ? p.objects[t.items[2]].text : std::to_string(p.objects[t.items[2]].integer);
480 s.integer = p.objects[t.items[4]].integer;
481 if (s.text.find("Storage") == std::string::npos || s.integer < 0)
482 return fail(errc::bad_header, "pickle: a persistent id whose type is not a storage");
483 SNICHOLLS_TRY_ASSIGN(const std::size_t i, make(std::move(s)));
484 SNICHOLLS_TRY_VOID(push(i));
485 break;
486 }
487 case 'R': { // REDUCE: callable(*args), for the allowed callables only
488 SNICHOLLS_TRY_ASSIGN(const std::size_t args, pop());
489 SNICHOLLS_TRY_ASSIGN(const std::size_t fn, pop());
490 const object& f = p.objects[fn];
491 const object& a = p.objects[args];
492 if (f.kind != kind::global || a.kind != kind::tuple)
493 return fail(errc::unsafe_pickle, "pickle: REDUCE of something not an allowed global");
494 if (f.text == "collections.OrderedDict") {
495 if (!a.items.empty())
496 return fail(errc::unsafe_pickle, "pickle: OrderedDict with arguments");
497 SNICHOLLS_TRY_VOID(container(kind::dict, {}));
498 } else if (f.text == "torch._utils._rebuild_tensor_v2") {
499 const auto arg = [&](std::size_t k) -> const object& { return p.objects[a.items[k]]; };
500 if (a.items.size() < 4 || arg(0).kind != kind::storage || arg(1).kind != kind::integer || arg(2).kind != kind::tuple
501 || arg(3).kind != kind::tuple)
502 return fail(errc::bad_header, "pickle: _rebuild_tensor_v2 without (storage, offset, size, stride)");
503 SNICHOLLS_TRY_VOID(container(kind::tensor, {a.items[0], a.items[1], a.items[2], a.items[3]}));
504 } else if (f.text == "torch._utils._rebuild_parameter") {
505 if (a.items.empty() || p.objects[a.items[0]].kind != kind::tensor)
506 return fail(errc::bad_header, "pickle: _rebuild_parameter without a tensor");
507 SNICHOLLS_TRY_VOID(push(a.items[0]));
508 } else {
509 return fail(errc::unsafe_pickle, "pickle: REDUCE of " + quote_name(f.text) + " is not allowed");
510 }
511 break;
512 }
513 case 'b': { // BUILD: state on a dict or tensor is discarded
514 SNICHOLLS_TRY_ASSIGN(const std::size_t state, pop());
515 (void)state;
516 if (stack.empty() || (p.objects[stack.back()].kind != kind::dict && p.objects[stack.back()].kind != kind::tensor))
517 return fail(errc::unsafe_pickle, "pickle: BUILD on something not a dict or tensor");
518 break;
519 }
520 default: {
521 char hex[8];
522 std::snprintf(hex, sizeof hex, "0x%02x", op);
523 return fail(errc::unsafe_pickle, std::string("pickle: opcode ") + hex + " at byte " + std::to_string(at - 1) + " is not allowed");
524 }
525 }
526 }
