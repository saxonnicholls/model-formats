121 [[nodiscard]] result<std::vector<std::string>> token_bytes() const
122 {
123 if (kind != tokenizer_kind::byte_level_bpe)
124 return fail(errc::unsupported, "vocabulary: token bytes need a byte-level BPE vocabulary; "
125 + std::string(gguf_tokenizer_model(kind)) + " waits for core's tokenizer track");
126 std::vector<std::string> out;
127 out.reserve(tokens.size());
128 for (std::uint32_t id = 0; id < tokens.size(); ++id) {
129 if (auto b = detail::byte_level_decode(tokens[id])) {
130 out.push_back(std::move(*b));
131 continue;
132 }
133 const auto t = type_of(id);
134 if (t != gguf::token_type::control && t != gguf::token_type::user_defined && t != gguf::token_type::unused)
135 return fail(errc::bad_header, "vocabulary: token " + std::to_string(id) + " " + quote_name(tokens[id]) + " is not byte-level");
136 out.push_back(tokens[id]); // an added token is its UTF-8 text
137 }
138 return ok(std::move(out));
139 }

143class bpe_ranks
144{
145public:
148 [[nodiscard]] static result<bpe_ranks> make(const vocabulary& v)
149 {
150 SNICHOLLS_TRY_ASSIGN(std::vector<std::string> bytes, v.token_bytes());
151 bpe_ranks r;
152 r.ids_.reserve(bytes.size());
153 for (std::uint32_t id = 0; id < bytes.size(); ++id)
154 if (v.type_of(id) == gguf::token_type::normal || v.type_of(id) == gguf::token_type::byte)
155 r.ids_.emplace(bytes[id], id); // the first id of a duplicated text wins
156 // Merged tokens' ids must increase with the merge's rank for id-as-rank to be Hugging Face's merge order
157 std::int64_t last = -1;
158 for (const auto& m : v.merges) {
159 const auto space = m.find(' ');
160 if (space == std::string::npos)
161 return fail(errc::bad_header, "vocabulary: merge " + quote_name(m) + " is not \"left right\"");
162 const auto left = detail::byte_level_decode(std::string_view(m).substr(0, space));
163 const auto right = detail::byte_level_decode(std::string_view(m).substr(space + 1));
164 if (!left || !right)
165 return fail(errc::bad_header, "vocabulary: merge " + quote_name(m) + " is not byte-level");
166 const auto it = r.ids_.find(*left + *right);
167 if (it == r.ids_.end() || r.ids_.find(*left) == r.ids_.end() || r.ids_.find(*right) == r.ids_.end())
168 return fail(errc::bad_header, "vocabulary: merge " + quote_name(m) + " names a token not in the vocabulary");
169 if (static_cast<std::int64_t>(it->second) <= last)
170 r.ordered_ = false;
171 last = it->second;
172 }
173 return ok(std::move(r));
174 }
175
177 [[nodiscard]] std::uint32_t operator()(std::string_view bytes) const
178 {
179 const auto it = ids_.find(bytes);
180 return it == ids_.end() ? ml::bpe_no_rank : it->second;
181 }
182
184 [[nodiscard]] bool ids_follow_merges() const noexcept { return ordered_; }
186 [[nodiscard]] std::size_t size() const noexcept { return ids_.size(); }
187
188private:
189 struct hash
190 {
191 using is_transparent = void;
192 std::size_t operator()(std::string_view s) const noexcept { return std::hash<std::string_view>{}(s); }
193 };
194 std::unordered_map<std::string, std::uint32_t, hash, std::equal_to<>> ids_;
195 bool ordered_ = true;
196};
