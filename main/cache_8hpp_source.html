73[[nodiscard]] constexpr bool valid_repo_id(std::string_view repo) noexcept
74{
75 if (repo.empty() || repo.size() > 200)
76 return false;
77 int slashes = 0;
78 for (const char c : repo) {
79 if (c == '/')
80 ++slashes;
81 else if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '.' || c == '_' || c == '-'))
82 return false;
83 }
84 return slashes <= 1 && safetensors::safe_relative_name(repo) && !repo.starts_with("-");
85}

89[[nodiscard]] constexpr bool valid_revision(std::string_view rev) noexcept
90{
91 if (rev.empty() || rev.size() > 256)
92 return false;
93 for (const char c : rev)
94 if (static_cast<unsigned char>(c) < 0x21 || c == '\\' || c == '%' || c == '?' || c == '#')
95 return false;
96 return safetensors::safe_relative_name(rev);
97}

100[[nodiscard]] constexpr bool is_commit(std::string_view rev) noexcept
101{
102 if (rev.size() != 40)
103 return false;
104 for (const char c : rev)
105 if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f')))
106 return false;
107 return true;
108}

139 [[nodiscard]] static result<cache> open(const std::filesystem::path& root, std::string_view repo)
140 {
141 if (!valid_repo_id(repo))
142 return fail(errc::bad_argument, "hub: " + quote_name(repo) + " is not a repository id");
143 cache c;
144 std::string folder = "models--";
145 for (const char ch : repo)
146 folder += ch == '/' ? std::string("--") : std::string(1, ch);
147 c.dir_ = root / folder;
148 return ok(std::move(c));
149 }

152 [[nodiscard]] const std::filesystem::path& dir() const noexcept { return dir_; }
154 [[nodiscard]] std::filesystem::path blob(std::string_view etag) const { return dir_ / "blobs" / std::string(etag); }
156 [[nodiscard]] std::filesystem::path incomplete(std::string_view etag) const { return dir_ / "blobs" / (std::string(etag) + ".incomplete"); }
158 [[nodiscard]] std::filesystem::path snapshot(std::string_view commit) const { return dir_ / "snapshots" / std::string(commit); }

162 [[nodiscard]] result<void> write_ref(std::string_view revision, std::string_view commit) const
163 {
164 if (!valid_revision(revision) || !is_commit(commit))
165 return fail(errc::bad_argument, "hub: an unsafe revision or commit");
166 if (revision == commit)
167 return ok();
168 return io::write_file(dir_ / "refs" / std::string(revision), std::as_bytes(std::span(commit.data(), commit.size())), false);
169 }
170
172 [[nodiscard]] result<std::string> resolve_offline(std::string_view revision) const
173 {
174 if (!valid_revision(revision))
175 return fail(errc::bad_argument, "hub: " + quote_name(revision) + " is not a revision");
176 if (is_commit(revision)) {
177 if (std::filesystem::is_directory(snapshot(revision)))
178 return ok(std::string(revision));
179 return fail(errc::offline, "hub: commit " + std::string(revision) + " is not in the cache");
180 }
181 auto r = read_file(dir_ / "refs" / std::string(revision));
182 if (r.is_err() || !is_commit(*r))
183 return fail(errc::offline, "hub: revision " + quote_name(revision) + " was never resolved into this cache");
184 return ok(std::move(*r));
185 }

189 [[nodiscard]] result<void> link(std::string_view commit, std::string_view path, std::string_view etag) const
190 {
191 if (!is_commit(commit) || !safetensors::safe_relative_name(path) || !safetensors::safe_relative_name(etag))
192 return fail(errc::bad_argument, "hub: an unsafe commit, path or blob name");
193 const auto target = snapshot(commit) / std::string(path);
194 std::error_code ec;
195 std::filesystem::create_directories(target.parent_path(), ec);
196 std::string up = "../../";
197 for (const char c : path)
198 if (c == '/')
199 up += "../";
200 std::filesystem::remove(target, ec);
201 std::filesystem::create_symlink(std::filesystem::path(up + "blobs/" + std::string(etag)), target, ec);
202 if (ec)
203 return err(io::io_error(target, "cannot link", ec));
204 return ok();
205 }

208 [[nodiscard]] bool has(std::string_view commit, std::string_view path) const
209 {
210 std::error_code ec;
211 return std::filesystem::exists(snapshot(commit) / std::string(path), ec); // follows the link to the blob
212 }

214private:

216};
