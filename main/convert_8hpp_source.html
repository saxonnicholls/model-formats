108[[nodiscard]] constexpr std::optional<file_format> format_from_extension(std::string_view path) noexcept
109{
110 if (path.ends_with(".safetensors"))
111 return file_format::safetensors;
112 if (path.ends_with(".gguf"))
113 return file_format::gguf;
114 if (path.ends_with(".npy"))
115 return file_format::npy;
116 if (path.ends_with(".npz"))
117 return file_format::npz;
118 if (path.ends_with(".pt") || path.ends_with(".pth"))
119 return file_format::pytorch;
120 return std::nullopt;

125[[nodiscard]] inline std::optional<file_format> detect(std::span<const std::byte> bytes) noexcept
126{
127 if (bytes.size() >= 4 && load_le<std::uint32_t>(bytes.data()) == gguf::magic)
128 return file_format::gguf;
129 if (bytes.size() >= 6 && std::memcmp(bytes.data(), "\x93NUMPY", 6) == 0)
130 return file_format::npy;
131 if (bytes.size() >= 1024) {
132 const auto m = static_cast<std::int32_t>(load_le<std::uint32_t>(bytes.data()));
133 if (m == llmc::checkpoint_magic)
134 return file_format::llmc_checkpoint;
135 if (m == llmc::tokens_magic)
136 return file_format::llmc_tokens;
137 }
138 if (bytes.size() >= 4 && load_le<std::uint32_t>(bytes.data()) == 0x04034b50) {
139 const auto a = zip::parse(bytes);
140 if (a.is_ok())
141 for (const auto& e : a->entries)
142 if (e.name.ends_with("/data.pkl"))
143 return file_format::pytorch;
144 return file_format::npz;
145 }
146 if (bytes.size() >= 9 && std::to_integer<char>(bytes[8]) == '{' && load_le<std::uint64_t>(bytes.data()) <= bytes.size() - 8)
147 return file_format::safetensors;
148 return std::nullopt;
149}

182[[nodiscard]] inline result<tensor> dequantize(const gguf::tensor_info& t)
183{
184 if (const auto v = t.dense())
185 return cast(*v, dtype::f32);
186 tensor out{std::string(t.name), dtype::f32, t.shape, {}};
187 const auto n = t.shape.elements();
188 if (!n)
189 return fail(errc::bad_shape, "dequantize: " + quote_name(t.name) + " overflows");
190 out.data.resize(static_cast<std::size_t>(*n) * 4);
191 const auto f = std::span(reinterpret_cast<float*>(out.data.data()), static_cast<std::size_t>(*n));
192 const auto blocks = [&]<typename B>(std::type_identity<B>) {
193 const std::span<const B> b(reinterpret_cast<const B*>(t.data.data()), t.data.size() / sizeof(B));
194 ml::dequantize(b, f);
195 };
196 switch (t.type) {
197 case gguf::ggml_type::q8_0: blocks(std::type_identity<ml::block_q8_0>{}); break;
198 case gguf::ggml_type::q4_0: blocks(std::type_identity<ml::block_q4_0>{}); break;
199 case gguf::ggml_type::mxfp4: blocks(std::type_identity<ml::block_mxfp4>{}); break;
200 default:
201 return fail(errc::unsupported, "dequantize: " + std::string(gguf::traits(t.type).name) + " (" + quote_name(t.name)
202 + ") waits for core's quantisation track");
203 }
204 return ok(std::move(out));
205}

209 inline std::string value_text(const gguf::value& v)
210 {
211 switch (v.type) {
212 case gguf::value_type::string: return std::string(*v.as_string());
213 case gguf::value_type::boolean: return *v.as_bool() ? "true" : "false";
214 case gguf::value_type::f32: case gguf::value_type::f64: return std::to_string(*v.as_f64());
215 case gguf::value_type::array: return "[" + std::to_string(v.count) + " x " + std::string(gguf::to_string(v.element_type)) + "]";
216 case gguf::value_type::i8: case gguf::value_type::i16: case gguf::value_type::i32: case gguf::value_type::i64:
217 return std::to_string(*v.as_i64());
218 default: return std::to_string(*v.as_u64());
219 }
220 }

231[[nodiscard]] inline result<loaded> read_any(std::span<const std::byte> bytes, const read_options& opt = {})
232{
233 const auto f = detect(bytes);
234 if (!f)
235 return fail(errc::unsupported, "convert: not a format this library reads");
236 loaded l;
237 l.format = *f;
238 switch (*f) {
239 case file_format::safetensors: {
240 SNICHOLLS_TRY_ASSIGN(safetensors::file s, safetensors::parse(bytes));
241 for (auto& e : s.tensors) { // names are owned by the parsed file: keep a copy beside the view
242 l.owned.push_back(tensor{std::move(e.name), e.type, e.shape, {}});
243 l.tensors.push_back(tensor_view{l.owned.back().name, e.type, e.shape, e.data});
244 }
245 l.metadata = std::move(s.metadata);
246 break;
247 }
248 case file_format::gguf: {
249 SNICHOLLS_TRY_ASSIGN(const gguf::file g, gguf::parse(bytes));
250 for (const auto& e : g.metadata)
251 l.metadata.emplace_back(std::string(e.key), detail::value_text(e.val));
252 for (const auto& t : g.tensors) {
253 if (const auto v = t.dense()) {
254 l.tensors.push_back(*v);
255 continue;
256 }
257 if (!opt.dequantize)
258 return fail(errc::unsupported, "convert: " + quote_name(t.name) + " is quantised (" + std::string(gguf::traits(t.type).name) + ")");
259 SNICHOLLS_TRY_ASSIGN(tensor d, dequantize(t));
260 l.notes.push_back(std::string(t.name) + ": dequantised from " + std::string(gguf::traits(t.type).name) + " to f32");
261 l.owned.push_back(std::move(d));
262 l.tensors.push_back(l.owned.back().view());
263 }
264 break;
265 }
266 case file_format::npy: {
267 SNICHOLLS_TRY_ASSIGN(const tensor_view v, numpy::parse_npy(bytes, "array"));
268 l.tensors.push_back(v);
269 break;
270 }
271 case file_format::npz: {
272 SNICHOLLS_TRY_ASSIGN(l.tensors, numpy::parse_npz(bytes));
273 break;
274 }
275 case file_format::pytorch: {
276 SNICHOLLS_TRY_ASSIGN(pytorch::checkpoint c, pytorch::parse(bytes));
277 for (auto& t : c.tensors) {
278 if (t.contiguous) {
279 l.owned.push_back(tensor{t.name, t.type, t.shape, {}});
280 l.tensors.push_back(tensor_view{l.owned.back().name, t.type, t.shape, t.data});
281 } else {
282 l.owned.push_back(t.materialize());
283 l.tensors.push_back(l.owned.back().view());
284 l.notes.push_back(t.name + ": not contiguous; copied to row-major order");
285 }
286 }
287 for (const auto& s : c.skipped)
288 l.notes.push_back(s + ": not a tensor; left out");
289 break;
290 }
291 case file_format::llmc_checkpoint: {
292 SNICHOLLS_TRY_ASSIGN(const llmc::checkpoint c, llmc::parse_checkpoint(bytes));
293 l.tensors.assign(c.tensors.begin(), c.tensors.end());
294 const auto& s = c.shape;
295 for (const auto& [k, v] : {std::pair<const char*, std::size_t>{"max_seq", s.max_seq}, {"vocab", s.vocab}, {"layers", s.layers},
296 {"heads", s.heads}, {"channels", s.channels}, {"padded_vocab", s.padded_vocab}})
297 l.metadata.emplace_back(std::string("llmc.") + k, std::to_string(v));
298 break;
299 }
300 case file_format::llmc_tokens: {
301 SNICHOLLS_TRY_ASSIGN(const std::span<const std::byte> t, llmc::parse_tokens(bytes));
302 l.tensors.push_back(tensor_view{"tokens", dtype::u16, tensor_shape{t.size() / 2}, t});
303 break;
304 }
305 }
306 return ok(std::move(l));
307}

312[[nodiscard]] inline result<std::vector<std::byte>> write_any(file_format to, std::span<const tensor_view> tensors,
313 std::span<const std::pair<std::string, std::string>> metadata = {})
314{
315 switch (to) {
316 case file_format::safetensors: return safetensors::write(tensors, metadata);
317 case file_format::gguf: {
318 gguf::writer w;
319 for (const auto& [k, v] : metadata)
320 w.set(k, gguf::value_string(v));
321 for (const auto& t : tensors)
322 SNICHOLLS_TRY_VOID(w.add_tensor(t));
323 return w.write();
324 }
325 case file_format::npz: return numpy::write_npz(tensors);
326 case file_format::npy:
327 if (tensors.size() != 1)
328 return fail(errc::bad_argument, "convert: an .npy holds exactly one array, not " + std::to_string(tensors.size()));
329 return numpy::write_npy(tensors[0]);
330 default: return fail(errc::unsupported, "convert: " + std::string(to_string(to)) + " is not written by write_any");
331 }
332}
