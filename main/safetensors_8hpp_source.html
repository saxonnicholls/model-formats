144[[nodiscard]] inline result<file> parse(std::span<const std::byte> bytes, const parse_options& opt = {})
145{
146 if (bytes.size() < 8)
147 return fail(errc::truncated, "safetensors: shorter than the 8-byte header length");
148 const std::uint64_t n = load_le<std::uint64_t>(bytes.data());
149 if (n > opt.max_header)
150 return fail(errc::too_large, "safetensors: header of " + std::to_string(n) + " bytes is over the limit");
151 if (n > bytes.size() - 8)
152 return fail(errc::truncated, "safetensors: header runs past the end of the file");
153 const std::string_view text(reinterpret_cast<const char*>(bytes.data() + 8), static_cast<std::size_t>(n));
154 if (text.empty() || text.front() != '{')
155 return fail(errc::bad_header, "safetensors: header does not start with '{'");
156 SNICHOLLS_TRY_ASSIGN(const detail::json j, detail::parse_json(text, "safetensors header"));
157 if (!j.is_object())
158 return fail(errc::bad_header, "safetensors: header is not an object");
159
160 file f;
161 f.header_size = n;
162 f.data = bytes.subspan(static_cast<std::size_t>(8 + n));
163 for (const auto& [key, value] : j.items()) {
164 if (key == "__metadata__") {
165 if (!value.is_object())
166 return fail(errc::bad_header, "safetensors: __metadata__ is not an object");
167 for (const auto& [mk, mv] : value.items()) {
168 const std::string* s = detail::as_string(mv);
169 if (!s)
170 return fail(errc::bad_header, "safetensors: metadata value of " + quote_name(mk) + " is not a string");
171 f.metadata.emplace_back(mk, *s);
172 }
173 continue;
174 }
175 if (!value.is_object())
176 return fail(errc::bad_header, "safetensors: entry " + quote_name(key) + " is not an object");
177 entry e;
178 e.name = key;
179 const std::string* dt = detail::string_at(value, "dtype");
180 if (!dt)
181 return fail(errc::bad_header, "safetensors: " + quote_name(key) + " has no dtype");
182 const auto t = dtype_from_safetensors(*dt);
183 if (!t)
184 return fail(errc::bad_dtype, "safetensors: " + quote_name(key) + " has dtype " + quote_name(*dt));
185 e.type = *t;
186 const detail::json* shape = detail::member(value, "shape");
187 if (!shape || !shape->is_array())
188 return fail(errc::bad_header, "safetensors: " + quote_name(key) + " has no shape array");
189 for (const auto& d : *shape) {
190 const auto v = detail::as_u64(d);
191 if (!v)
192 return fail(errc::bad_shape, "safetensors: " + quote_name(key) + " has an extent that is not a non-negative integer");
193 if (!e.shape.push_back(*v))
194 return fail(errc::bad_shape, "safetensors: " + quote_name(key) + " has more than " + std::to_string(max_rank) + " dimensions");
195 }
196 const detail::json* offsets = detail::member(value, "data_offsets");
197 if (!offsets || !offsets->is_array() || offsets->size() != 2)
198 return fail(errc::bad_header, "safetensors: " + quote_name(key) + " has no [begin, end] data_offsets");
199 const auto b = detail::as_u64((*offsets)[0]), en = detail::as_u64((*offsets)[1]);
200 if (!b || !en || *b > *en)
201 return fail(errc::bad_header, "safetensors: " + quote_name(key) + " has invalid data_offsets");
202 e.begin = *b;
203 e.end = *en;
204 const auto size = checked_byte_size(e.shape, e.type);
205 if (!size || *size != e.end - e.begin)
206 return fail(errc::bad_shape, "safetensors: " + quote_name(key) + " has " + std::to_string(e.end - e.begin)
207 + " bytes for shape " + to_string(e.shape));
208 if (e.end > f.data.size())
209 return fail(errc::out_of_bounds, "safetensors: " + quote_name(key) + " ends past the data");
210 e.data = f.data.subspan(static_cast<std::size_t>(e.begin), static_cast<std::size_t>(e.end - e.begin));
211 f.tensors.push_back(std::move(e));
212 }
213 std::sort(f.tensors.begin(), f.tensors.end(), [](const entry& a, const entry& b) {
214 return a.begin != b.begin ? a.begin < b.begin : a.end < b.end;
215 });
216 std::uint64_t at = 0;
217 for (const auto& e : f.tensors) {
218 if (e.begin < at)
219 return fail(errc::overlap, "safetensors: " + quote_name(e.name) + " overlaps the tensor before it");
220 if (e.begin > at && !opt.allow_holes)
221 return fail(errc::overlap, "safetensors: a hole in the data before " + quote_name(e.name));
222 at = e.end;
223 }
224 if (at != f.data.size() && !opt.allow_holes)
225 return fail(errc::overlap, "safetensors: " + std::to_string(f.data.size() - at) + " bytes after the last tensor");
226 std::sort(f.metadata.begin(), f.metadata.end());
227 return ok(std::move(f));
228}

241[[nodiscard]] inline result<layout> plan(std::span<const tensor_view> tensors, std::span<const metadata_entry> metadata = {})
242{
243 layout l;
244 l.order.assign(tensors.begin(), tensors.end());
245 std::sort(l.order.begin(), l.order.end(), [](const tensor_view& a, const tensor_view& b) {
246 const std::size_t sa = size_of(a.type), sb = size_of(b.type);
247 return sa != sb ? sa > sb : a.name < b.name;
248 });
249 detail::json j = detail::json::object();
250 for (std::size_t i = 0; i < l.order.size(); ++i) {
251 const auto& t = l.order[i];
252 if (t.name == "__metadata__")
253 return fail(errc::bad_argument, "safetensors: a tensor may not be called __metadata__");
254 const auto size = checked_byte_size(t.shape, t.type);
255 if (!size || *size != t.data.size())
256 return fail(errc::bad_shape, "safetensors: " + quote_name(t.name) + " has " + std::to_string(t.data.size())
257 + " bytes for shape " + to_string(t.shape));
258 detail::json shape = detail::json::array();
259 for (const auto d : t.shape.view())
260 shape.push_back(d);
261 const std::string name(t.name);
262 if (j.contains(name))
263 return fail(errc::duplicate_name, "safetensors: two tensors named " + quote_name(t.name));
264 j[name] = {{"dtype", std::string(safetensors_name(t.type))}, {"shape", std::move(shape)},
265 {"data_offsets", {l.data_size, l.data_size + *size}}};
266 l.data_size += *size;
267 }
268 if (!metadata.empty()) {
269 detail::json m = detail::json::object();
270 for (const auto& [k, v] : metadata)
271 m[k] = v;
272 j["__metadata__"] = std::move(m);
273 }
274 SNICHOLLS_TRY_ASSIGN(std::string text, detail::dump_json(j, "safetensors header"));
275 while ((8 + text.size()) % 8 != 0)
276 text.push_back(' ');
277 l.header.resize(8 + text.size());
278 store_le(l.header.data(), static_cast<std::uint64_t>(text.size()));
279 std::memcpy(l.header.data() + 8, text.data(), text.size());
280 return ok(std::move(l));
281}

287template <typename Sink>
288[[nodiscard]] result<void> write_to(std::span<const tensor_view> tensors, std::span<const metadata_entry> metadata, Sink&& sink)
289{
290 SNICHOLLS_TRY_ASSIGN(const layout l, plan(tensors, metadata));
291 sink(std::span<const std::byte>(l.header));
292 for (const auto& t : l.order)
293 sink(t.data);
294 return ok();
295}

300{

302 SNICHOLLS_TRY_VOID(write_to(tensors, metadata, [&out](std::span<const std::byte> s) { out.insert(out.end(), s.begin(), s.end()); }));
303 return ok(std::move(out));
