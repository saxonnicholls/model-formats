102 // Whether a component (pre_tokenizer, decoder) is, or contains, one of type `type`
103 inline bool has_type(const json* j, std::string_view type)
104 {
105 if (!j || !j->is_object())
106 return false;
107 if (const std::string* t = formats::detail::string_at(*j, "type"); t && *t == type)
108 return true;
109 for (const char* list : {"pretokenizers", "decoders", "processors"})
110 if (const json* l = formats::detail::member(*j, list); l && l->is_array())
111 for (const auto& e : *l)
112 if (has_type(&e, type))
113 return true;
114 return false;
115 }
116 inline result<void> set_token(vocabulary& v, std::uint64_t id, std::string text, gguf::token_type type)
117 {
118 if (id > 0x7fffffffu)
119 return fail(errc::bad_header, "tokenizer.json: token id " + std::to_string(id) + " is too large");
120 if (v.tokens.size() <= id) {
121 const std::size_t from = v.tokens.size();
122 v.tokens.resize(static_cast<std::size_t>(id) + 1);
123 v.types.resize(static_cast<std::size_t>(id) + 1, gguf::token_type::unused);
124 for (std::size_t i = from; i < v.tokens.size(); ++i)
125 v.tokens[i] = "[PAD" + std::to_string(i) + "]";
126 } else if (v.types[id] != gguf::token_type::unused) {
127 if (v.tokens[id] == text)
128 return ok(); // an added token that repeats the vocabulary's entry
129 return fail(errc::bad_header, "tokenizer.json: id " + std::to_string(id) + " is given twice");
130 }
131 v.tokens[id] = std::move(text);
132 v.types[id] = type;
133 return ok();
134 }
135 // A special token named in tokenizer_config.json: a string, or {"content": ...}
136 inline const std::string* token_text(const json& cfg, std::string_view key)
137 {
138 const json* m = formats::detail::member(cfg, key);
139 if (!m)
140 return nullptr;
141 if (const std::string* s = formats::detail::as_string(*m))
142 return s;
143 return formats::detail::string_at(*m, "content");
144 }

148[[nodiscard]] inline result<tokenizer> parse_tokenizer(std::string_view text)
149{
150 using formats::detail::member;
151 using formats::detail::string_at;
152 SNICHOLLS_TRY_ASSIGN(const detail::json j, formats::detail::parse_json(text, "tokenizer.json"));
153 const detail::json* model = member(j, "model");
154 if (!model || !model->is_object())
155 return fail(errc::bad_header, "tokenizer.json: no model");
156 const std::string* type = string_at(*model, "type");
157 const std::string t = type ? *type : "BPE"; // tokenizers omits the type of a BPE model in old files
158 tokenizer tok;
159 auto& v = tok.vocab;
160 if (t == "BPE") {
161 const bool fallback = formats::detail::bool_at(*model, "byte_fallback").value_or(false);
162 const bool byte_level = detail::has_type(member(j, "pre_tokenizer"), "ByteLevel") || detail::has_type(member(j, "decoder"), "ByteLevel");
163 if (!byte_level && !fallback)
164 return fail(errc::unsupported, "tokenizer.json: a BPE model that is neither byte-level nor SentencePiece-style");
165 v.kind = byte_level ? tokenizer_kind::byte_level_bpe : tokenizer_kind::sentencepiece;
166 } else if (t == "WordPiece") {
167 v.kind = tokenizer_kind::wordpiece;
168 } else if (t == "Unigram") {
169 v.kind = tokenizer_kind::unigram;
170 } else {
171 return fail(errc::unsupported, "tokenizer.json: model type \"" + t + "\" is not read");
172 }
173 const detail::json* vocab = member(*model, "vocab");
174 if (v.kind == tokenizer_kind::unigram) {
175 if (!vocab || !vocab->is_array())
176 return fail(errc::bad_header, "tokenizer.json: a Unigram vocab is a list of [token, score]");
177 std::uint64_t id = 0;
178 for (const auto& e : *vocab) {
179 const std::string* s = e.is_array() && e.size() == 2 ? formats::detail::as_string(e[0]) : nullptr;
180 const auto score = s ? formats::detail::as_double(e[1]) : std::nullopt;
181 if (!s || !score)
182 return fail(errc::bad_header, "tokenizer.json: a Unigram entry is not [token, score]");
183 SNICHOLLS_TRY_VOID(detail::set_token(v, id++, *s, gguf::token_type::normal));
184 v.scores.push_back(static_cast<float>(*score));
185 }
186 if (const auto unk = formats::detail::u64_at(*model, "unk_id"); unk && *unk <= 0xffffffffu)
187 v.unk = static_cast<std::uint32_t>(*unk);
188 } else {
189 if (!vocab || !vocab->is_object())
190 return fail(errc::bad_header, "tokenizer.json: no vocab object");
191 for (const auto& [k, idv] : vocab->items()) {
192 const auto id = formats::detail::as_u64(idv);
193 if (!id)
194 return fail(errc::bad_header, "tokenizer.json: the id of " + quote_name(k) + " is not a non-negative integer");
195 gguf::token_type type = gguf::token_type::normal;
196 if (v.kind == tokenizer_kind::sentencepiece && k.size() == 6 && k.starts_with("<0x") && k.back() == '>')
197 type = gguf::token_type::byte;
198 SNICHOLLS_TRY_VOID(detail::set_token(v, *id, k, type));
199 }
200 }
201 if (const detail::json* merges = member(*model, "merges"); merges && merges->is_array()) {
202 v.merges.reserve(merges->size());
203 for (const auto& m : *merges) {
204 if (const std::string* s = formats::detail::as_string(m)) {
205 v.merges.push_back(*s);
206 } else if (m.is_array() && m.size() == 2 && m[0].is_string() && m[1].is_string()) {
207 v.merges.push_back(*formats::detail::as_string(m[0]) + " " + *formats::detail::as_string(m[1]));
208 } else {
209 return fail(errc::bad_header, "tokenizer.json: a merge is neither \"a b\" nor [\"a\", \"b\"]");
210 }
211 }
212 }
213 if (const std::string* unk = string_at(*model, "unk_token"))
214 v.unk = tok.id_of(*unk);
215 if (const detail::json* added = member(j, "added_tokens"); added && added->is_array()) {
216 for (const auto& a : *added) {
217 const auto id = formats::detail::u64_at(a, "id");
218 const std::string* content = string_at(a, "content");
219 if (!id || !content)
220 return fail(errc::bad_header, "tokenizer.json: an added token without id or content");
221 const bool special = formats::detail::bool_at(a, "special").value_or(false);
222 const auto type = special ? gguf::token_type::control : gguf::token_type::user_defined;
223 if (*id < v.tokens.size() && v.tokens[*id] == *content) {
224 v.types[*id] = type;
225 continue;
226 }
227 SNICHOLLS_TRY_VOID(detail::set_token(v, *id, *content, type));
228 }
229 }
230 return ok(std::move(tok));
231}

236[[nodiscard]] inline result<void> parse_tokenizer_config(tokenizer& tok, std::string_view text)
237{
238 SNICHOLLS_TRY_ASSIGN(const detail::json j, formats::detail::parse_json(text, "tokenizer_config.json"));
239 if (!j.is_object())
240 return fail(errc::bad_header, "tokenizer_config.json: not an object");
241 tok.add_bos = formats::detail::bool_at(j, "add_bos_token");
242 tok.add_eos = formats::detail::bool_at(j, "add_eos_token");
243 const auto special = [&](std::string_view key, std::optional<std::uint32_t>& out) {
244 if (const std::string* s = detail::token_text(j, key))
245 if (const auto id = tok.id_of(*s))
246 out = id;
247 };
248 special("bos_token", tok.vocab.bos);
249 special("eos_token", tok.vocab.eos);
250 special("unk_token", tok.vocab.unk);
251 special("pad_token", tok.vocab.pad);
252 if (const detail::json* ct = formats::detail::member(j, "chat_template")) {
253 if (const std::string* s = formats::detail::as_string(*ct)) {
254 tok.chat_templates.push_back({"default", *s});
255 } else if (ct->is_array()) {
256 for (const auto& e : *ct) {
257 const std::string* name = formats::detail::string_at(e, "name");
258 const std::string* tpl = formats::detail::string_at(e, "template");
259 if (!name || !tpl)
260 return fail(errc::bad_header, "tokenizer_config.json: a chat template without name or template");
261 tok.chat_templates.push_back({*name, *tpl});
262 }
263 } else if (!ct->is_null()) {
264 return fail(errc::bad_header, "tokenizer_config.json: chat_template is neither a string nor a list");
265 }
266 }
267 return ok();
268}
