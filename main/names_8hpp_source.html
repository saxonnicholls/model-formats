195 // "<digits>." at the front of s: the number, and s advanced past it
196 inline std::optional<std::size_t> take_index(std::string_view& s)
197 {
198 std::size_t n = 0;
199 const auto [p, ec] = std::from_chars(s.data(), s.data() + s.size(), n);
200 if (ec != std::errc{} || p == s.data() || p == s.data() + s.size() || *p != '.')
201 return std::nullopt;
202 s.remove_prefix(static_cast<std::size_t>(p - s.data()) + 1);
203 return n;
204 }
205 // "<base>.weight" or "<base>.bias": the base and which
206 inline std::optional<std::pair<std::string_view, bool>> split_suffix(std::string_view s)
207 {
208 if (s.ends_with(".weight"))
209 return std::pair{s.substr(0, s.size() - 7), false};
210 if (s.ends_with(".bias"))
211 return std::pair{s.substr(0, s.size() - 5), true};
212 return std::nullopt;
213 }
214 template <std::size_t N>
215 std::optional<role> find_role(const std::array<pattern, N>& table, std::string_view base)
216 {
217 for (const auto& p : table)
218 if (!p.name.empty() && p.name == base)
219 return p.what;
220 return std::nullopt;
221 }
222 template <std::size_t N>
223 std::string_view role_name(const std::array<pattern, N>& table, role r)
224 {
225 for (const auto& p : table)
226 if (p.what == r && !p.name.empty())
227 return p.name;
228 return {};
229 }

234[[nodiscard]] inline std::optional<tensor_name> parse_hf_name(architecture a, std::string_view name)
235{
236 tensor_name t;
237 if (a == architecture::gpt2) {
238 if (name.starts_with("transformer."))
239 name.remove_prefix(12);
240 if (name == "attn.bias" || name.ends_with(".attn.bias") || name.ends_with(".attn.masked_bias")) {
241 std::string_view rest = name;
242 if (rest.starts_with("h.")) {
243 rest.remove_prefix(2);
244 t.layer = detail::take_index(rest);
245 }
246 t.what = role::mask_buffer;
247 t.bias = !name.ends_with("masked_bias");
248 return t;
249 }
250 const auto s = detail::split_suffix(name);
251 if (!s)
252 return std::nullopt;
253 t.bias = s->second;
254 std::string_view base = s->first;
255 if (base == "wte" || base == "lm_head" || base == "wpe" || base == "ln_f") {
256 t.what = base == "wte" ? role::token_embedding : base == "lm_head" ? role::lm_head : base == "wpe" ? role::position_embedding : role::final_norm;
257 return t;
258 }
259 if (!base.starts_with("h."))
260 return std::nullopt;
261 base.remove_prefix(2);
262 t.layer = detail::take_index(base);
263 const auto r = t.layer ? detail::find_role(detail::gpt2_block, base) : std::nullopt;
264 if (!r)
265 return std::nullopt;
266 t.what = *r;
267 return t;
268 }
269 const auto s = detail::split_suffix(name);
270 if (!s) {
271 if (name.ends_with(".mlp.gate.e_score_correction_bias") && name.starts_with("model.layers.")) {
272 std::string_view rest = name.substr(13);
273 t.layer = detail::take_index(rest);
274 t.what = role::router_bias;
275 t.bias = true;
276 return t.layer ? std::optional<tensor_name>(t) : std::nullopt;
277 }
278 return std::nullopt;
279 }
280 t.bias = s->second;
281 std::string_view base = s->first;
282 if (base == "model.embed_tokens" || base == "lm_head" || base == "model.norm") {
283 t.what = base == "model.embed_tokens" ? role::token_embedding : base == "lm_head" ? role::lm_head : role::final_norm;
284 return t;
285 }
286 if (!base.starts_with("model.layers."))
287 return std::nullopt;
288 base.remove_prefix(13);
289 t.layer = detail::take_index(base);
290 if (!t.layer)
291 return std::nullopt;
292 if (base.starts_with("mlp.experts.")) {
293 base.remove_prefix(12);
294 t.expert = detail::take_index(base);
295 if (!t.expert)
296 return std::nullopt;
297 if (base == "gate_proj" || base == "up_proj" || base == "down_proj") {
298 t.what = base == "gate_proj" ? role::expert_gate : base == "up_proj" ? role::expert_up : role::expert_down;
299 return t;
300 }
301 return std::nullopt;
302 }
303 const auto r = detail::find_role(detail::llama_block, base);
304 if (!r || *r == role::expert_gate || *r == role::expert_up || *r == role::expert_down)
305 return std::nullopt;
306 t.what = *r;
307 return t;
308}

312[[nodiscard]] inline std::string hf_name(architecture a, const tensor_name& t)
313{
314 const std::string suffix = t.bias ? ".bias" : ".weight";
315 if (a == architecture::gpt2) {
316 switch (t.what) {
317 case role::token_embedding: return "wte" + suffix;
318 case role::position_embedding: return "wpe" + suffix;
319 case role::final_norm: return "ln_f" + suffix;
320 case role::lm_head: return "lm_head" + suffix;
321 case role::mask_buffer: return t.layer ? "h." + std::to_string(*t.layer) + (t.bias ? ".attn.bias" : ".attn.masked_bias") : std::string();
322 default: break;
323 }
324 const std::string_view n = detail::role_name(detail::gpt2_block, t.what);
325 if (n.empty() || !t.layer)
326 return {};
327 return "h." + std::to_string(*t.layer) + "." + std::string(n) + suffix;
328 }
329 switch (t.what) {
330 case role::token_embedding: return "model.embed_tokens" + suffix;
331 case role::final_norm: return "model.norm" + suffix;
332 case role::lm_head: return "lm_head" + suffix;
333 case role::position_embedding: case role::qkv: case role::mask_buffer: return {};
334 default: break;
335 }
336 if (!t.layer)
337 return {};
338 const std::string layer = "model.layers." + std::to_string(*t.layer) + ".";
339 if (t.what == role::router_bias)
340 return layer + "mlp.gate.e_score_correction_bias";
341 if (t.what == role::expert_gate || t.what == role::expert_up || t.what == role::expert_down) {
342 if (!t.expert)
343 return {};
344 return layer + "mlp.experts." + std::to_string(*t.expert) + "." + std::string(detail::role_name(detail::llama_block, t.what)) + suffix;
345 }
346 const std::string_view n = detail::role_name(detail::llama_block, t.what);
347 return n.empty() ? std::string() : layer + std::string(n) + suffix;
348}

351[[nodiscard]] inline std::optional<tensor_name> parse_gguf_name(std::string_view name)
352{
353 const auto s = detail::split_suffix(name);
354 if (!s)
355 return std::nullopt;
356 tensor_name t;
357 t.bias = s->second;
358 std::string_view base = s->first;
359 if (base == "token_embd" || base == "position_embd" || base == "output_norm" || base == "output") {
360 t.what = base == "token_embd" ? role::token_embedding : base == "position_embd" ? role::position_embedding
361 : base == "output_norm" ? role::final_norm
362 : role::lm_head;
363 return t;
364 }
365 if (!base.starts_with("blk."))
366 return std::nullopt;
367 base.remove_prefix(4);
368 t.layer = detail::take_index(base);
369 const auto r = t.layer ? detail::find_role(detail::gguf_block, base) : std::nullopt;
370 if (!r)
371 return std::nullopt;
372 t.what = *r;
373 return t;
374}

378[[nodiscard]] inline std::string gguf_name(const tensor_name& t)
379{
380 const std::string suffix = t.bias ? ".bias" : ".weight";
381 switch (t.what) {
382 case role::token_embedding: return "token_embd" + suffix;
383 case role::position_embedding: return "position_embd" + suffix;
384 case role::final_norm: return "output_norm" + suffix;
385 case role::lm_head: return "output" + suffix;
386 case role::mask_buffer: return {};
387 default: break;
388 }

390 if (n.empty() || !t.layer)
391 return {};
392 return "blk." + std::to_string(*t.layer) + "." + std::string(n) + suffix;
393}

413inline void permute_rope(std::span<const std::byte> in, std::span<std::byte> out, std::size_t heads, std::size_t row_bytes) noexcept
414{
415 assert(in.size() == out.size() && heads > 0 && row_bytes > 0 && in.size() % (heads * row_bytes * 2) == 0);
416 const std::size_t rows = in.size() / row_bytes, head_rows = rows / heads, half = head_rows / 2;
417 for (std::size_t h = 0; h < heads; ++h)
418 for (std::size_t i = 0; i < half; ++i)
419 for (std::size_t p = 0; p < 2; ++p)
420 std::memcpy(out.data() + (h * head_rows + 2 * i + p) * row_bytes, in.data() + (h * head_rows + p * half + i) * row_bytes, row_bytes);
421}

425inline void unpermute_rope(std::span<const std::byte> in, std::span<std::byte> out, std::size_t heads, std::size_t row_bytes) noexcept
426{
427 assert(in.size() == out.size() && heads > 0 && row_bytes > 0 && in.size() % (heads * row_bytes * 2) == 0);
428 const std::size_t rows = in.size() / row_bytes, head_rows = rows / heads, half = head_rows / 2;
429 for (std::size_t h = 0; h < heads; ++h)
430 for (std::size_t i = 0; i < half; ++i)
431 for (std::size_t p = 0; p < 2; ++p)
432 std::memcpy(out.data() + (h * head_rows + p * half + i) * row_bytes, in.data() + (h * head_rows + 2 * i + p) * row_bytes, row_bytes);
433}
