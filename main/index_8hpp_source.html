92[[nodiscard]] constexpr bool safe_relative_name(std::string_view name) noexcept
93{
94 if (name.empty() || name.front() == '/' || name.find('\\') != std::string_view::npos || name.find('\0') != std::string_view::npos)
95 return false;
96 std::size_t at = 0;
97 while (at <= name.size()) {
98 const std::size_t slash = std::min(name.find('/', at), name.size());
99 const std::string_view part = name.substr(at, slash - at);
100 if (part.empty() || part == "." || part == "..")
101 return false;
102 at = slash + 1;
103 }
104 return true;
105}

108[[nodiscard]] inline result<index> parse_index(std::string_view text)
109{
110 SNICHOLLS_TRY_ASSIGN(const detail::json j, detail::parse_json(text, "safetensors index"));
111 const detail::json* map = detail::member(j, "weight_map");
112 if (!map || !map->is_object())
113 return fail(errc::bad_header, "safetensors index: no weight_map object");
114 index ix;
115 for (const auto& [k, v] : map->items()) {
116 const std::string* s = detail::as_string(v);
117 if (!s)
118 return fail(errc::bad_header, "safetensors index: shard of " + quote_name(k) + " is not a string");
119 if (!safe_relative_name(*s))
120 return fail(errc::bad_argument, "safetensors index: unsafe shard name " + quote_name(*s));
121 ix.weight_map.emplace_back(k, *s);
122 }
123 std::sort(ix.weight_map.begin(), ix.weight_map.end());
124 if (const detail::json* meta = detail::member(j, "metadata")) {
125 if (!meta->is_object())
126 return fail(errc::bad_header, "safetensors index: metadata is not an object");
127 for (const auto& [k, v] : meta->items()) {
128 if (k == "total_size") {
129 ix.total_size = detail::as_u64(v);
130 if (!ix.total_size)
131 return fail(errc::bad_header, "safetensors index: total_size is not a non-negative integer");
132 continue;
133 }
134 SNICHOLLS_TRY_ASSIGN(std::string t, detail::dump_json(v, "safetensors index"));
135 ix.metadata.emplace_back(k, std::move(t));
136 }
137 }
138 return ok(std::move(ix));
139}

142[[nodiscard]] inline result<std::string> write_index(const index& ix)
143{
144 detail::json j = detail::json::object();
145 detail::json meta = detail::json::object();
146 if (ix.total_size)
147 meta["total_size"] = *ix.total_size;
148 for (const auto& [k, v] : ix.metadata) {
149 SNICHOLLS_TRY_ASSIGN(detail::json parsed, detail::parse_json(v, "safetensors index metadata"));
150 meta[k] = std::move(parsed);
151 }
152 j["metadata"] = std::move(meta);
153 detail::json map = detail::json::object();
154 for (const auto& [t, s] : ix.weight_map) {
155 if (!safe_relative_name(s))
156 return fail(errc::bad_argument, "safetensors index: unsafe shard name " + quote_name(s));
157 if (map.contains(t))
158 return fail(errc::duplicate_name, "safetensors index: two entries for " + quote_name(t));
159 map[t] = s;
160 }
161 j["weight_map"] = std::move(map);
162 SNICHOLLS_TRY_ASSIGN(std::string out, detail::dump_json(j, "safetensors index", 2));
163 return ok(out + "\n");
164}

168[[nodiscard]] inline std::vector<std::string> shard_names(std::string_view stem, std::size_t count)
169{
170 std::vector<std::string> names;
171 if (count <= 1) {
172 names.push_back(std::string(stem) + ".safetensors");
173 return names;
174 }
175 for (std::size_t i = 1; i <= count; ++i) {
176 char buf[64];
177 std::snprintf(buf, sizeof buf, "-%05zu-of-%05zu.safetensors", i, count);
178 names.push_back(std::string(stem) + buf);
179 }
180 return names;
181}
