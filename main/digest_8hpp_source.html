93 [[nodiscard]] std::string hex()
94 {
95 if (!done_) {
96 unsigned int n = 0;
97 EVP_DigestFinal_ex(ctx_.get(), out_, &n);
98 size_ = n;
99 done_ = true;
100 }
101 static constexpr char digits[] = "0123456789abcdef";
102 std::string h;
103 for (unsigned i = 0; i < size_; ++i) {
104 h.push_back(digits[out_[i] >> 4]);
105 h.push_back(digits[out_[i] & 15]);
106 }
107 return h;
108 }
109
110private:
111 struct free_ctx
112 {
113 void operator()(EVP_MD_CTX* c) const noexcept { EVP_MD_CTX_free(c); }
114 };
115 std::unique_ptr<EVP_MD_CTX, free_ctx> ctx_;
116 unsigned char out_[EVP_MAX_MD_SIZE]{};
117 unsigned size_ = 0;
118 bool done_ = false;
119};
