81[[nodiscard]] constexpr std::optional<dtype> storage_dtype(std::string_view name) noexcept
82{
83 struct row
84 {
85 std::string_view name;
86 dtype type;
87 };
88 constexpr row rows[] = {{"DoubleStorage", dtype::f64}, {"FloatStorage", dtype::f32}, {"HalfStorage", dtype::f16},
89 {"BFloat16Storage", dtype::bf16}, {"LongStorage", dtype::i64}, {"IntStorage", dtype::i32},
90 {"ShortStorage", dtype::i16}, {"CharStorage", dtype::i8}, {"ByteStorage", dtype::u8},
91 {"BoolStorage", dtype::boolean}, {"Float8_e4m3fnStorage", dtype::f8_e4m3},
92 {"Float8_e5m2Storage", dtype::f8_e5m2}};
93 for (const auto& r : rows)
94 if (r.name == name)
95 return r.type;
96 return std::nullopt;

100struct tensor_entry
101{
102 std::string name;
103 dtype type = dtype::f32;
104 tensor_shape shape;
105 tensor_shape stride;
106 std::uint64_t offset = 0;
107 std::string storage_key;
108 std::span<const std::byte> storage;
109 bool contiguous = true;
110 std::span<const std::byte> data;
111
113 [[nodiscard]] tensor_view view() const noexcept { return {name, type, shape, data}; }
115 [[nodiscard]] tensor materialize() const
116 {
117 tensor t{name, type, shape, {}};
118 const std::size_t es = size_of(type);
119 const auto n = static_cast<std::size_t>(*shape.elements());
120 t.data.resize(n * es);
121 std::array<std::uint64_t, max_rank> index{};
122 for (std::size_t i = 0; i < n; ++i) {
123 std::uint64_t at = offset;
124 for (std::size_t d = 0; d < shape.rank; ++d)
125 at += index[d] * stride[d];
126 std::memcpy(t.data.data() + i * es, storage.data() + at * es, es);
127 for (std::size_t d = shape.rank; d > 0; --d) { // odometer, innermost first
128 if (++index[d - 1] < shape[d - 1])
129 break;
130 index[d - 1] = 0;
131 }
132 }
133 return t;
134 }
135};

156 inline result<tensor_shape> shape_of(const pickled& p, std::size_t tuple, std::string_view what)
157 {
158 tensor_shape s;
159 for (const std::size_t i : p.at(tuple).items) {
160 const object& e = p.at(i);
161 if (e.kind != kind::integer || e.integer < 0 || !s.push_back(static_cast<std::uint64_t>(e.integer)))
162 return fail(errc::bad_shape, "pytorch: a tensor's " + std::string(what) + " is not a tuple of at most 8 non-negative ints");
163 }
164 return ok(s);
165 }
166 inline result<void> flatten(const pickled& p, std::size_t dict, const std::string& prefix, const zip::archive& a,
167 const std::string& root, checkpoint& out, int depth)
168 {
169 if (depth > 16)
170 return fail(errc::bad_header, "pytorch: dicts nested more than 16 deep");
171 const auto& items = p.at(dict).items;
172 for (std::size_t k = 0; k + 1 < items.size(); k += 2) {
173 const object& key = p.at(items[k]);
174 if (key.kind != kind::string && key.kind != kind::integer)
175 return fail(errc::bad_header, "pytorch: a state dict key is neither a string nor an int");
176 const std::string name = prefix + (key.kind == kind::string ? key.text : std::to_string(key.integer));
177 const object& v = p.at(items[k + 1]);
178 if (v.kind == kind::dict) {
179 SNICHOLLS_TRY_VOID(flatten(p, items[k + 1], name + ".", a, root, out, depth + 1));
180 continue;
181 }
182 if (v.kind != kind::tensor) {
183 out.skipped.push_back(name);
184 continue;
185 }
186 const object& st = p.at(v.items[0]);
187 tensor_entry t;
188 t.name = name;
189 const auto type = storage_dtype(st.text);
190 if (!type)
191 return fail(errc::unsupported, "pytorch: " + quote_name(name) + " has storage type " + quote_name(st.text));
192 t.type = *type;
193 t.storage_key = st.key;
194 if (p.at(v.items[1]).integer < 0)
195 return fail(errc::bad_header, "pytorch: " + quote_name(name) + " has a negative storage offset");
196 t.offset = static_cast<std::uint64_t>(p.at(v.items[1]).integer);
197 SNICHOLLS_TRY_ASSIGN(t.shape, shape_of(p, v.items[2], "size"));
198 SNICHOLLS_TRY_ASSIGN(t.stride, shape_of(p, v.items[3], "stride"));
199 if (t.stride.rank != t.shape.rank)
200 return fail(errc::bad_shape, "pytorch: " + quote_name(name) + " has a stride of another rank");
201 const zip::entry* member = a.find(root + "/data/" + t.storage_key);
202 if (!member)
203 return fail(errc::not_found, "pytorch: storage " + quote_name(t.storage_key) + " of " + quote_name(name) + " is missing");
204 const auto storage_bytes = checked_mul(static_cast<std::uint64_t>(st.integer), std::uint64_t{size_of(t.type)});
205 if (!storage_bytes || *storage_bytes > member->data.size())
206 return fail(errc::out_of_bounds, "pytorch: storage " + quote_name(t.storage_key) + " is shorter than its element count");
207 t.storage = member->data.first(static_cast<std::size_t>(*storage_bytes));
208 // the extent: offset + sum((size - 1) * stride) + 1 elements, unless empty
209 const auto n = t.shape.elements();
210 if (!n)
211 return fail(errc::bad_shape, "pytorch: " + quote_name(name) + "'s size overflows");
212 std::uint64_t expected = 1;
213 for (std::size_t d = t.shape.rank; d > 0; --d) {
214 if (t.shape[d - 1] != 1 && t.stride[d - 1] != expected)
215 t.contiguous = false;
216 expected *= t.shape[d - 1];
217 }
218 if (*n > 0) {
219 std::optional<std::uint64_t> last = t.offset;
220 for (std::size_t d = 0; d < t.shape.rank && last; ++d) {
221 const auto step = checked_mul(t.shape[d] - 1, t.stride[d]);
222 last = step ? checked_add(*last, *step) : std::nullopt;
223 }
224 if (!last || *last >= static_cast<std::uint64_t>(st.integer))
225 return fail(errc::out_of_bounds, "pytorch: " + quote_name(name) + " reaches past its storage");
226 if (t.contiguous)
227 t.data = t.storage.subspan(static_cast<std::size_t>(t.offset * size_of(t.type)), static_cast<std::size_t>(*n * size_of(t.type)));
228 }
229 out.tensors.push_back(std::move(t));
230 }
231 return ok();
232 }

237[[nodiscard]] inline result<checkpoint> parse(std::span<const std::byte> bytes, const unpickle_limits& limits = {})
238{
239 if (bytes.size() >= 2 && bytes[0] == std::byte{0x80} && std::to_integer<int>(bytes[1]) <= 5)
240 return fail(errc::unsupported, "pytorch: the legacy (pre-1.6, non-zip) format is not read; re-save with torch.save's default or convert to safetensors");
241 if (bytes.size() < 4 || load_le<std::uint32_t>(bytes.data()) != 0x04034b50)
242 return fail(errc::bad_magic, "pytorch: not a zip archive");
243 SNICHOLLS_TRY_ASSIGN(const zip::archive a, zip::parse(bytes));
244 checkpoint c;
245 const zip::entry* pkl = nullptr;
246 for (const auto& e : a.entries) {
247 const auto slash = e.name.find('/');
248 if (slash != std::string_view::npos && e.name.substr(slash) == "/data.pkl") {
249 if (pkl)
250 return fail(errc::bad_header, "pytorch: more than one data.pkl");
251 pkl = &e;
252 c.prefix = std::string(e.name.substr(0, slash));
253 }
254 }
255 if (!pkl)
256 return fail(errc::not_found, "pytorch: no <archive>/data.pkl");
257 if (const zip::entry* order = a.find(c.prefix + "/byteorder")) {
258 const std::string_view o(reinterpret_cast<const char*>(order->data.data()), order->data.size());
259 if (o != "little")
260 return fail(errc::unsupported, "pytorch: byte order " + quote_name(o));
261 }
262 SNICHOLLS_TRY_ASSIGN(const pickled p, unpickle(pkl->data, limits));
263 if (p.at(p.root).kind == kind::tensor) { // torch.save(tensor): one tensor, named ""
264 object wrapper;
265 wrapper.kind = kind::dict;
266 pickled q = p;
267 object key;
268 key.kind = kind::string;
269 q.objects.push_back(key);
270 wrapper.items = {q.objects.size() - 1, p.root};
271 q.objects.push_back(wrapper);
272 SNICHOLLS_TRY_VOID(detail::flatten(q, q.objects.size() - 1, "", a, c.prefix, c, 0));
273 return ok(std::move(c));
274 }
275 if (p.at(p.root).kind != kind::dict)
276 return fail(errc::bad_header, "pytorch: the saved object is not a dict of tensors");
277 SNICHOLLS_TRY_VOID(detail::flatten(p, p.root, "", a, c.prefix, c, 0));
278 return ok(std::move(c));
279}
