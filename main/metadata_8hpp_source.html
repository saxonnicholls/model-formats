88 inline std::optional<std::size_t> size_key(const file& f, std::string_view arch, std::string_view suffix)
89 {
90 const value* v = f.find(arch_key(arch, suffix));
91 const auto u = v ? v->as_u64() : std::nullopt;
92 return u ? std::optional<std::size_t>(static_cast<std::size_t>(*u)) : std::nullopt;
93 }
94 // A float32 key as the decimal it was written from: 1e-5 stored as f32 reads back as 1e-5, not 9.99999974e-6
95 // (the shortest text that round-trips the float, read as a double)
96 inline std::optional<double> double_key(const file& f, std::string_view arch, std::string_view suffix)
97 {
98 const value* v = f.find(arch_key(arch, suffix));
99 if (!v)
100 return std::nullopt;
101 if (v->type != value_type::f32)
102 return v->as_f64();
103 char buf[32];
104 const auto r = std::to_chars(buf, buf + sizeof buf - 1, static_cast<float>(*v->as_f64()));
105 *r.ptr = '\0';
106 return std::strtod(buf, nullptr);
107 }
108 inline std::optional<std::uint32_t> token_key(const file& f, std::string_view key)
109 {
110 const value* v = f.find(key);
111 const auto u = v ? v->as_u64() : std::nullopt;
112 return u && *u <= 0xffffffffu ? std::optional<std::uint32_t>(static_cast<std::uint32_t>(*u)) : std::nullopt;
113 }

118[[nodiscard]] inline result<model_config> read_model_config(const file& f)
119{
120 const std::string_view a = f.architecture();
121 model_config c;
122 if (a == "gpt2")
123 c.arch = architecture::gpt2;
124 else if (a == "llama")
125 c.arch = architecture::llama;
126 else if (a == "qwen2")
127 c.arch = architecture::qwen2;
128 else if (a == "qwen3")
129 c.arch = architecture::qwen3;
130 else if (a == "olmoe")
131 c.arch = architecture::olmoe;
132 else if (a == "deepseek2")
133 c.arch = architecture::deepseek_v2;
134 else
135 return fail(errc::unsupported, "gguf: architecture \"" + std::string(a) + "\" is not mapped onto core");
136 const auto need = [&](std::string_view suffix) -> result<std::size_t> {
137 if (const auto v = detail::size_key(f, a, suffix))
138 return ok(*v);
139 return fail(errc::bad_header, "gguf: no " + arch_key(a, suffix));
140 };
141 auto& b = c.block;
142 SNICHOLLS_TRY_ASSIGN(b.d_model, need(llm::embedding_length));
143 SNICHOLLS_TRY_ASSIGN(c.layers, need(llm::block_count));
144 SNICHOLLS_TRY_ASSIGN(b.n_heads, need(attention::head_count));
145 SNICHOLLS_TRY_ASSIGN(b.d_ff, need(llm::feed_forward_length));
146 c.context = detail::size_key(f, a, llm::context_length).value_or(0);
147 b.n_kv_heads = detail::size_key(f, a, attention::head_count_kv).value_or(b.n_heads);
148 b.head_dim = detail::size_key(f, a, attention::key_length).value_or(b.n_heads ? b.d_model / b.n_heads : 0);
149 c.vocab = detail::size_key(f, a, llm::vocab_size).value_or(0);
150 if (c.vocab == 0)
151 if (const value* t = f.find(tokenizer::tokens); t && t->type == value_type::array)
152 c.vocab = static_cast<std::size_t>(t->count);
153 c.bos = detail::token_key(f, tokenizer::bos_token_id);
154 c.eos = detail::token_key(f, tokenizer::eos_token_id);
155 c.tied_embeddings = f.find_tensor("output.weight") == nullptr;
156 b.bias = f.find_tensor("blk.0.attn_q.bias") != nullptr || f.find_tensor("blk.0.attn_qkv.bias") != nullptr;
157 b.rope_base = detail::double_key(f, a, rope::freq_base).value_or(10000.0);
158 if (const auto e = detail::double_key(f, a, attention::layer_norm_rms_epsilon)) {
159 b.norm = ml::norm_kind::rms;
160 b.norm_eps = *e;
161 } else if (const auto l = detail::double_key(f, a, attention::layer_norm_epsilon)) {
162 b.norm = ml::norm_kind::layer;
163 b.norm_eps = *l;
164 }
165 if (const value* s = f.find(arch_key(a, rope::scaling_type))) {
166 const std::string_view kind = s->as_string().value_or("");
167 const double factor = detail::double_key(f, a, rope::scaling_factor).value_or(1.0);
168 if (kind == "linear") {
169 b.rope_scaling = ml::rope_scaling_kind::linear;
170 b.rope_factor = factor;
171 } else if (kind == "yarn") {
172 b.rope_scaling = ml::rope_scaling_kind::yarn_index_ramp;
173 b.rope_factor = factor;
174 b.rope_original_context = detail::size_key(f, a, rope::scaling_original_context_length).value_or(c.context);
175 b.yarn_beta_fast = detail::double_key(f, a, rope::scaling_yarn_beta_fast).value_or(32.0);
176 b.yarn_beta_slow = detail::double_key(f, a, rope::scaling_yarn_beta_slow).value_or(1.0);
177 // llama.cpp stores DeepSeek's 0.1 * mscale_all_dim as the log multiplier
178 if (const auto m = detail::double_key(f, a, rope::scaling_yarn_log_multiplier))
179 b.yarn_mscale = *m / 0.1;
180 } else if (kind != "none" && !kind.empty()) {
181 c.notes.push_back("rope.scaling.type \"" + std::string(kind) + "\" has no counterpart in core");
182 }
183 }
184 switch (c.arch) {
185 case architecture::gpt2:
186 b.mlp = ml::mlp_kind::gelu_tanh;
187 b.rope = ml::rope_kind::none;
188 b.bias = true;
189 break;
190 case architecture::llama:
191 case architecture::mistral:
192 b.mlp = ml::mlp_kind::swiglu;
193 b.rope = ml::rope_kind::interleaved;
194 b.window = detail::size_key(f, a, attention::sliding_window).value_or(0);
195 break;
196 case architecture::qwen2:
197 case architecture::qwen3:
198 b.mlp = ml::mlp_kind::swiglu;
199 b.rope = ml::rope_kind::half;
200 if (c.arch == architecture::qwen3)
201 b.qk_norm = ml::qk_norm_kind::before_rope;
202 break;
203 case architecture::olmoe:
204 case architecture::deepseek_v2:
205 case architecture::deepseek_v3:
206 b.mlp = ml::mlp_kind::swiglu;
207 b.rope = c.arch == architecture::olmoe ? ml::rope_kind::half : ml::rope_kind::interleaved;
208 break;
209 }
210 if (const auto experts = detail::size_key(f, a, llm::expert_count); experts && *experts > 0) {
211 b.experts = *experts;
212 b.top_k = detail::size_key(f, a, llm::expert_used_count).value_or(0);
213 b.shared_experts = detail::size_key(f, a, llm::expert_shared_count).value_or(0);
214 b.expert_d_ff = detail::size_key(f, a, llm::expert_feed_forward_length).value_or(b.d_ff);
215 b.route_scale = detail::double_key(f, a, llm::expert_weights_scale).value_or(1.0);
216 b.expert_groups = detail::size_key(f, a, llm::expert_group_count).value_or(0);
217 b.group_top_k = detail::size_key(f, a, llm::expert_group_used_count).value_or(0);
218 c.leading_dense_layers = detail::size_key(f, a, llm::leading_dense_block_count).value_or(0);
219 const auto gating = detail::size_key(f, a, llm::expert_gating_func).value_or(1);
220 const value* norm = f.find(arch_key(a, llm::expert_weights_norm));
221 const bool renormalise = norm && norm->as_bool().value_or(false);
222 b.scoring = gating == 2 ? ml::moe_scoring::sigmoid : renormalise ? ml::moe_scoring::softmax_top_k : ml::moe_scoring::softmax;
223 if (gating == 2)
224 c.arch = architecture::deepseek_v3;
225 c.notes.push_back("gguf: the router's training weights (balance and z losses, the bias update rate) are not stored; left 0");
226 }
227 if (c.arch == architecture::deepseek_v2 || c.arch == architecture::deepseek_v3) {
228 b.q_lora_rank = detail::size_key(f, a, attention::q_lora_rank).value_or(0);
229 b.kv_lora_rank = detail::size_key(f, a, attention::kv_lora_rank).value_or(0);
230 b.qk_rope_dim = detail::size_key(f, a, rope::dimension_count).value_or(0);
231 b.head_dim = detail::size_key(f, a, attention::key_length_mla).value_or(b.head_dim);
232 b.v_head_dim = detail::size_key(f, a, attention::value_length_mla)
233 .value_or(detail::size_key(f, a, attention::value_length).value_or(0));
234 b.n_kv_heads = b.n_heads; // latent attention; llama.cpp writes 1 for its absorbed form
235 b.mtp_depth = detail::size_key(f, a, llm::nextn_predict_layers).value_or(0);
236 if (c.layers > b.mtp_depth)
237 c.layers -= b.mtp_depth; // llama.cpp counts the MTP layers as blocks
238 }
239 if (c.arch == architecture::olmoe)
240 c.notes.push_back("olmoe: q_norm and k_norm normalise the whole projection; core's qk_norm is per head, so it is left none");
241 if (!ml::valid(b))
242 return fail(errc::bad_header, "gguf: the " + std::string(a) + " configuration is not one core accepts (ml::valid)");
243 return ok(std::move(c));
244}

249[[nodiscard]] inline result<void> write_model_config(writer& w, const model_config& c, std::string_view name)
250{
251 const auto& b = c.block;
252 if (b.mask != ml::attention_mask::causal || (b.qk_norm != ml::qk_norm_kind::none && c.arch != architecture::qwen3))
253 return fail(errc::unsupported, "gguf: llama.cpp's architectures have no such attention");
254 const std::string_view a = gguf_architecture(c.arch);
255 const auto u32 = [&](std::string_view suffix, std::size_t v) { w.set(arch_key(a, suffix), value_u32(static_cast<std::uint32_t>(v))); };
256 const auto f32 = [&](std::string_view suffix, double v) { w.set(arch_key(a, suffix), value_f32(static_cast<float>(v))); };
257 w.set(general::architecture, value_string(a));
258 w.set(general::name, value_string(name));
259 w.set(general::quantization_version, value_u32(2));
260 u32(llm::context_length, c.context);
261 u32(llm::embedding_length, b.d_model);
262 u32(llm::block_count, c.layers + b.mtp_depth);
263 u32(llm::feed_forward_length, b.d_ff);
264 u32(attention::head_count, b.n_heads);
265 if (c.vocab)
266 u32(llm::vocab_size, c.vocab);
267 if (b.norm == ml::norm_kind::layer)
268 f32(attention::layer_norm_epsilon, b.norm_eps);
269 else
270 f32(attention::layer_norm_rms_epsilon, b.norm_eps);
271 if (b.rope != ml::rope_kind::none) {
272 u32(attention::head_count_kv, b.latent() ? 1 : b.n_kv_heads);
273 u32(attention::key_length, b.latent() ? b.kv_lora_rank + b.qk_rope_dim : b.head_dim);
274 u32(attention::value_length, b.latent() ? b.kv_lora_rank : b.head_dim);
275 f32(rope::freq_base, b.rope_base);
276 u32(rope::dimension_count, b.latent() ? b.qk_rope_dim : b.head_dim);
277 }
278 if (b.window)
279 u32(attention::sliding_window, b.window);
280 if (b.rope_scaling == ml::rope_scaling_kind::linear) {
281 w.set(arch_key(a, rope::scaling_type), value_string("linear"));
282 f32(rope::scaling_factor, b.rope_factor);
283 } else if (b.rope_scaling == ml::rope_scaling_kind::yarn || b.rope_scaling == ml::rope_scaling_kind::yarn_index_ramp) {
284 w.set(arch_key(a, rope::scaling_type), value_string("yarn"));
285 f32(rope::scaling_factor, b.rope_factor);
286 u32(rope::scaling_original_context_length, b.rope_original_context);
287 f32(rope::scaling_yarn_beta_fast, b.yarn_beta_fast);
288 f32(rope::scaling_yarn_beta_slow, b.yarn_beta_slow);
289 f32(rope::scaling_yarn_log_multiplier, 0.1 * b.yarn_mscale);
290 }
291 if (b.moe()) {
292 u32(llm::expert_count, b.experts);
293 u32(llm::expert_used_count, b.top_k);
294 u32(llm::expert_shared_count, b.shared_experts);
295 u32(llm::expert_feed_forward_length, b.expert_d_ff);
296 f32(llm::expert_weights_scale, b.route_scale);
297 w.set(arch_key(a, llm::expert_weights_norm), value_bool(b.scoring != ml::moe_scoring::softmax));
298 u32(llm::expert_gating_func, b.scoring == ml::moe_scoring::sigmoid ? 2 : 1);
299 if (b.expert_groups) {
300 u32(llm::expert_group_count, b.expert_groups);
301 u32(llm::expert_group_used_count, b.group_top_k);
302 }
303 u32(llm::leading_dense_block_count, c.leading_dense_layers);
304 }
305 if (b.latent()) {
306 if (b.q_lora_rank)
307 u32(attention::q_lora_rank, b.q_lora_rank);
308 u32(attention::kv_lora_rank, b.kv_lora_rank);
309 u32(attention::key_length_mla, b.head_dim);
310 u32(attention::value_length_mla, b.v_head_dim);
311 if (b.mtp_depth)
312 u32(llm::nextn_predict_layers, b.mtp_depth);
313 }
314 if (c.bos)
315 w.set(tokenizer::bos_token_id, value_u32(*c.bos));
316 if (c.eos)
317 w.set(tokenizer::eos_token_id, value_u32(*c.eos));
318 return ok();
319}

323[[nodiscard]] inline result<vocabulary> read_vocabulary(const file& f)
324{
325 const value* model = f.find(tokenizer::model);
326 const value* toks = f.find(tokenizer::tokens);
327 if (!model || !toks)
328 return fail(errc::not_found, "gguf: no tokenizer.ggml.model and tokens");
329 vocabulary v;
330 const std::string_view m = model->as_string().value_or("");
331 if (m == "gpt2")
332 v.kind = tokenizer_kind::byte_level_bpe;
333 else if (m == "llama")
334 v.kind = tokenizer_kind::sentencepiece;
335 else if (m == "bert")
336 v.kind = tokenizer_kind::wordpiece;
337 else if (m == "t5")
338 v.kind = tokenizer_kind::unigram;
339 else
340 return fail(errc::unsupported, "gguf: tokenizer model \"" + std::string(m) + "\" is not read");
341 if (const value* pre = f.find(tokenizer::pre))
342 v.pre = std::string(pre->as_string().value_or(""));
343 const auto strings = toks->strings();
344 if (!strings)
345 return fail(errc::bad_header, "gguf: tokenizer.ggml.tokens is not an array of strings");
346 v.tokens.assign(strings->begin(), strings->end());
347 if (const value* types = f.find(tokenizer::token_type)) {
348 const auto t = types->integers();
349 if (!t || t->size() != v.tokens.size())
350 return fail(errc::bad_header, "gguf: tokenizer.ggml.token_type does not match the tokens");
351 for (const auto x : *t)
352 v.types.push_back(static_cast<token_type>(x));
353 }
354 if (const value* scores = f.find(tokenizer::scores)) {
355 const auto s = scores->numbers();
356 if (!s || s->size() != v.tokens.size())
357 return fail(errc::bad_header, "gguf: tokenizer.ggml.scores does not match the tokens");
358 for (const auto x : *s)
359 v.scores.push_back(static_cast<float>(x));
360 }
361 if (const value* merges = f.find(tokenizer::merges)) {
362 const auto s = merges->strings();
363 if (!s)
364 return fail(errc::bad_header, "gguf: tokenizer.ggml.merges is not an array of strings");
365 v.merges.assign(s->begin(), s->end());
366 }
367 v.bos = detail::token_key(f, tokenizer::bos_token_id);
368 v.eos = detail::token_key(f, tokenizer::eos_token_id);
369 v.unk = detail::token_key(f, tokenizer::unknown_token_id);
370 v.pad = detail::token_key(f, tokenizer::padding_token_id);
371 return ok(std::move(v));
372}

376inline void write_vocabulary(writer& w, const vocabulary& v, std::string_view chat_template = {})
377{
378 w.set(tokenizer::model, value_string(gguf_tokenizer_model(v.kind)));
379 w.set(tokenizer::pre, value_string(v.pre.empty() ? std::string_view("default") : std::string_view(v.pre)));
380 w.set(tokenizer::tokens, value_strings(v.tokens));
381 std::vector<metadata_value> types;
382 types.reserve(v.tokens.size());
383 for (std::uint32_t i = 0; i < v.tokens.size(); ++i)
384 types.push_back(value_i32(static_cast<std::int32_t>(v.type_of(i))));
385 w.set(tokenizer::token_type, value_array(value_type::i32, std::move(types)));
386 if (!v.scores.empty()) {
387 std::vector<metadata_value> scores;
388 for (const float s : v.scores)
389 scores.push_back(value_f32(s));
390 w.set(tokenizer::scores, value_array(value_type::f32, std::move(scores)));
391 }
392 if (!v.merges.empty() || v.kind == tokenizer_kind::byte_level_bpe) // llama.cpp requires the key for gpt2, even empty
393 w.set(tokenizer::merges, value_strings(v.merges));
394 if (v.bos)
395 w.set(tokenizer::bos_token_id, value_u32(*v.bos));
396 if (v.eos)
397 w.set(tokenizer::eos_token_id, value_u32(*v.eos));
398 if (v.unk)
399 w.set(tokenizer::unknown_token_id, value_u32(*v.unk));
400 if (v.pad)
401 w.set(tokenizer::padding_token_id, value_u32(*v.pad));
402 if (!chat_template.empty())
403 w.set(tokenizer::chat_template, value_string(chat_template));
404}
