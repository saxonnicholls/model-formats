113[[nodiscard]] inline result<archive> parse(std::span<const std::byte> bytes)
114{
115 using detail::at;
116 using detail::fits;
117 if (bytes.size() < 22)
118 return fail(errc::truncated, "zip: shorter than an end-of-central-directory record");
119 std::uint64_t end = bytes.size() - 22;
120 const std::uint64_t lowest = bytes.size() > 22 + 65535 ? bytes.size() - 22 - 65535 : 0;
121 for (;; --end) {
122 if (at<std::uint32_t>(bytes, end) == detail::end_sig && end + 22 + at<std::uint16_t>(bytes, end + 20) == bytes.size())
123 break;
124 if (end == lowest)
125 return fail(errc::bad_magic, "zip: no end-of-central-directory record");
126 }
127 archive a;
128 if (at<std::uint16_t>(bytes, end + 4) != 0 || at<std::uint16_t>(bytes, end + 6) != 0)
129 return fail(errc::unsupported, "zip: multi-disk archives are not read");
130 std::uint64_t count = at<std::uint16_t>(bytes, end + 10);
131 std::uint64_t cd_size = at<std::uint32_t>(bytes, end + 12);
132 std::uint64_t cd_offset = at<std::uint32_t>(bytes, end + 16);
133 if (count == 0xffff || cd_size == 0xffffffff || cd_offset == 0xffffffff) {
134 if (end < 20 || at<std::uint32_t>(bytes, end - 20) != detail::locator64_sig)
135 return fail(errc::bad_header, "zip: ZIP64 values without a ZIP64 locator");
136 const std::uint64_t rec = at<std::uint64_t>(bytes, end - 20 + 8);
137 if (!fits(bytes, rec, 56) || at<std::uint32_t>(bytes, rec) != detail::end64_sig)
138 return fail(errc::bad_header, "zip: the ZIP64 end record is not where the locator says");
139 count = at<std::uint64_t>(bytes, rec + 32);
140 cd_size = at<std::uint64_t>(bytes, rec + 40);
141 cd_offset = at<std::uint64_t>(bytes, rec + 48);
142 a.zip64 = true;
143 }
144 if (!fits(bytes, cd_offset, cd_size))
145 return fail(errc::out_of_bounds, "zip: the central directory runs past the end");
146 if (count > cd_size / 46)
147 return fail(errc::bad_header, "zip: more members than the central directory can hold");
148 a.entries.reserve(static_cast<std::size_t>(count));
149 std::uint64_t p = cd_offset;
150 const std::uint64_t cd_end = cd_offset + cd_size;
151 for (std::uint64_t i = 0; i < count; ++i) {
152 if (p + 46 > cd_end || at<std::uint32_t>(bytes, p) != detail::central_sig)
153 return fail(errc::bad_header, "zip: central directory entry " + std::to_string(i) + " is malformed");
154 const std::uint16_t flags = at<std::uint16_t>(bytes, p + 8), method = at<std::uint16_t>(bytes, p + 10);
155 entry e;
156 e.crc = at<std::uint32_t>(bytes, p + 16);
157 std::uint64_t csize = at<std::uint32_t>(bytes, p + 20), usize = at<std::uint32_t>(bytes, p + 24);
158 const std::uint16_t n = at<std::uint16_t>(bytes, p + 28), x = at<std::uint16_t>(bytes, p + 30), c = at<std::uint16_t>(bytes, p + 32);
159 std::uint64_t local = at<std::uint32_t>(bytes, p + 42);
160 if (p + 46 + n + x + c > cd_end)
161 return fail(errc::bad_header, "zip: central directory entry " + std::to_string(i) + " runs past the directory");
162 e.name = std::string_view(reinterpret_cast<const char*>(bytes.data() + p + 46), n);
163 // ZIP64 extended information: the fields that were 0xFFFFFFFF, in order
164 for (std::uint64_t q = p + 46 + n; q + 4 <= p + 46 + n + x;) {
165 const std::uint16_t id = at<std::uint16_t>(bytes, q), len = at<std::uint16_t>(bytes, q + 2);
166 if (q + 4 + len > p + 46 + n + x)
167 return fail(errc::bad_header, "zip: an extra field of " + quote_name(e.name) + " runs past its entry");
168 if (id == 0x0001) {
169 std::uint64_t r = q + 4;
170 const auto take = [&](std::uint64_t& v) {
171 if (v != 0xffffffff)
172 return true;
173 if (r + 8 > q + 4 + len)
174 return false;
175 v = at<std::uint64_t>(bytes, r);
176 r += 8;
177 return true;
178 };
179 if (!take(usize) || !take(csize) || !take(local))
180 return fail(errc::bad_header, "zip: the ZIP64 field of " + quote_name(e.name) + " is too short");
181 }
182 q += 4 + len;
183 }
184 if (flags & 1)
185 return fail(errc::unsupported, "zip: " + quote_name(e.name) + " is encrypted");
186 if (method != 0)
187 return fail(errc::unsupported, "zip: " + quote_name(e.name) + " is compressed (method " + std::to_string(method)
188 + "); only stored members are read");
189 if (csize != usize)
190 return fail(errc::bad_header, "zip: stored member " + quote_name(e.name) + " has different sizes");
191 if (!fits(bytes, local, 30) || at<std::uint32_t>(bytes, local) != detail::local_sig)
192 return fail(errc::bad_header, "zip: the local header of " + quote_name(e.name) + " is missing");
193 if (at<std::uint16_t>(bytes, local + 26) != n)
194 return fail(errc::bad_header, "zip: the local header of " + quote_name(e.name) + " disagrees with the directory");
195 e.offset = local + 30 + n + at<std::uint16_t>(bytes, local + 28);
196 if (!fits(bytes, e.offset, usize))
197 return fail(errc::out_of_bounds, "zip: " + quote_name(e.name) + " runs past the end");
198 e.data = bytes.subspan(static_cast<std::size_t>(e.offset), static_cast<std::size_t>(usize));
199 a.entries.push_back(e);
200 p += 46 + n + x + c;
201 }
202 std::vector<const entry*> order;
203 for (const auto& e : a.entries)
204 order.push_back(&e);
205 std::sort(order.begin(), order.end(), [](const entry* l, const entry* r) { return l->offset < r->offset; });
206 for (std::size_t i = 1; i < order.size(); ++i)
207 if (order[i]->offset < order[i - 1]->offset + order[i - 1]->data.size())
208 return fail(errc::overlap, "zip: " + quote_name(order[i]->name) + " overlaps " + quote_name(order[i - 1]->name));
209 std::sort(order.begin(), order.end(), [](const entry* l, const entry* r) { return l->name < r->name; });
210 for (std::size_t i = 1; i < order.size(); ++i)
211 if (order[i]->name == order[i - 1]->name)
212 return fail(errc::duplicate_name, "zip: two members named " + quote_name(order[i]->name));
213 return ok(std::move(a));
214}

218{
219 if (formats::detail::crc32(e.data) != e.crc)
220 return fail(errc::checksum_mismatch, "zip: the CRC-32 of " + quote_name(e.name) + " does not match");
221 return ok();

240[[nodiscard]] inline result<std::vector<std::byte>> write(std::span<const member> members, const write_options& opt = {})
241{
242 std::vector<std::string_view> names;
243 for (const auto& m : members) {
244 if (m.name.size() > 0xffff || m.name.empty())
245 return fail(errc::bad_argument, "zip: a member name must be 1 to 65535 bytes");
246 names.push_back(m.name);
247 }
248 std::sort(names.begin(), names.end());
249 if (const auto d = std::adjacent_find(names.begin(), names.end()); d != names.end())
250 return fail(errc::duplicate_name, "zip: two members named " + quote_name(*d));
251 std::vector<std::byte> out, cd;
252 const auto put16 = [](std::vector<std::byte>& v, std::uint16_t x) { for (const auto c : to_le_bytes(x)) v.push_back(std::byte{c}); };
253 const auto put32 = [](std::vector<std::byte>& v, std::uint32_t x) { for (const auto c : to_le_bytes(x)) v.push_back(std::byte{c}); };
254 const auto put64 = [](std::vector<std::byte>& v, std::uint64_t x) { for (const auto c : to_le_bytes(x)) v.push_back(std::byte{c}); };
255 const auto put = [](std::vector<std::byte>& v, std::string_view s) { for (const char c : s) v.push_back(static_cast<std::byte>(c)); };
256 constexpr std::uint32_t max32 = 0xffffffff;
257 const std::size_t align = opt.alignment ? opt.alignment : 1;
258 bool any64 = opt.force_zip64 || members.size() >= 0xffff;
259 for (const auto& m : members) {
260 const std::uint64_t local = out.size();
261 const std::uint64_t size = m.data.size();
262 const bool big = opt.force_zip64 || size >= max32 || local >= max32;
263 any64 = any64 || big;
264 const std::uint32_t crc = formats::detail::crc32(m.data);
265 // local header: the ZIP64 field (sizes) when big, then padding so the data is aligned
266 const std::size_t fixed = 30 + m.name.size() + (big ? 20 : 0);
267 std::size_t pad = (align - (local + fixed + 4) % align) % align;
268 put32(out, detail::local_sig);
269 put16(out, big ? 45 : 20);
270 put16(out, 0); // flags
271 put16(out, 0); // stored
272 put16(out, 0); // time 00:00
273 put16(out, 0x21); // date 1980-01-01
274 put32(out, crc);
275 put32(out, big ? max32 : static_cast<std::uint32_t>(size));
276 put32(out, big ? max32 : static_cast<std::uint32_t>(size));
277 put16(out, static_cast<std::uint16_t>(m.name.size()));
278 put16(out, static_cast<std::uint16_t>((big ? 20 : 0) + 4 + pad));
279 put(out, m.name);
280 if (big) {
281 put16(out, 0x0001);
282 put16(out, 16);
283 put64(out, size);
284 put64(out, size);
285 }
286 put16(out, 0x4246); // "FB", PyTorch's padding field
287 put16(out, static_cast<std::uint16_t>(pad));
288 out.insert(out.end(), pad, std::byte{0});
289 out.insert(out.end(), m.data.begin(), m.data.end());
290 // central directory entry
291 const bool big_cd = big || local >= max32;
292 put32(cd, detail::central_sig);
293 put16(cd, big_cd ? 45 : 20);
294 put16(cd, big_cd ? 45 : 20);
295 put16(cd, 0);
296 put16(cd, 0);
297 put16(cd, 0);
298 put16(cd, 0x21);
299 put32(cd, crc);
300 put32(cd, big_cd ? max32 : static_cast<std::uint32_t>(size));
301 put32(cd, big_cd ? max32 : static_cast<std::uint32_t>(size));
302 put16(cd, static_cast<std::uint16_t>(m.name.size()));
303 put16(cd, big_cd ? 28 : 0);
304 put16(cd, 0); // comment
305 put16(cd, 0); // disk
306 put16(cd, 0); // internal attributes
307 put32(cd, 0); // external attributes
308 put32(cd, big_cd ? max32 : static_cast<std::uint32_t>(local));
309 put(cd, m.name);
310 if (big_cd) {
311 put16(cd, 0x0001);
312 put16(cd, 24);
313 put64(cd, size);
314 put64(cd, size);
315 put64(cd, local);
316 }
317 }
318 const std::uint64_t cd_offset = out.size(), cd_size = cd.size();
319 out.insert(out.end(), cd.begin(), cd.end());
320 any64 = any64 || cd_offset >= max32;
321 if (any64) {
322 const std::uint64_t rec = out.size();
323 put32(out, detail::end64_sig);
324 put64(out, 44);
325 put16(out, 45);
326 put16(out, 45);
327 put32(out, 0);
328 put32(out, 0);
329 put64(out, members.size());
330 put64(out, members.size());
331 put64(out, cd_size);
332 put64(out, cd_offset);
333 put32(out, detail::locator64_sig);
334 put32(out, 0);
335 put64(out, rec);
336 put32(out, 1);
337 }
338 put32(out, detail::end_sig);
339 put16(out, 0);
340 put16(out, 0);
341 put16(out, any64 ? 0xffff : static_cast<std::uint16_t>(members.size()));
342 put16(out, any64 ? 0xffff : static_cast<std::uint16_t>(members.size()));
343 put32(out, any64 ? max32 : static_cast<std::uint32_t>(cd_size));
344 put32(out, any64 ? max32 : static_cast<std::uint32_t>(cd_offset));
345 put16(out, 0);
346 return ok(std::move(out));
347}
