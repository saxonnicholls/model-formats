165[[nodiscard]] constexpr std::optional<ggml_type_traits> traits(std::uint32_t code) noexcept
166{
167 for (const auto& t : ggml_types)
168 if (static_cast<std::uint32_t>(t.type) == code)
169 return t;
170 return std::nullopt;
171}

174[[nodiscard]] constexpr ggml_type_traits traits(ggml_type t) noexcept { return *traits(static_cast<std::uint32_t>(t)); }

177[[nodiscard]] constexpr std::optional<ggml_type> ggml_type_from_name(std::string_view name) noexcept
178{
179 for (const auto& t : ggml_types)
180 if (t.name == name)
181 return t.type;
182 return std::nullopt;
183}

187[[nodiscard]] constexpr std::optional<std::uint64_t> row_bytes(ggml_type t, std::uint64_t elements) noexcept
188{
189 const auto tr = traits(t);
190 if (elements % tr.block_size != 0)
191 return std::nullopt;
192 return checked_mul(elements / tr.block_size, std::uint64_t{tr.type_size});
193}

196[[nodiscard]] constexpr std::optional<dtype> to_dtype(ggml_type t) noexcept
197{
198 switch (t) {
199 case ggml_type::f32: return dtype::f32;
200 case ggml_type::f16: return dtype::f16;
201 case ggml_type::bf16: return dtype::bf16;
202 case ggml_type::f64: return dtype::f64;
203 case ggml_type::i8: return dtype::i8;
204 case ggml_type::i16: return dtype::i16;
205 case ggml_type::i32: return dtype::i32;
206 case ggml_type::i64: return dtype::i64;
207 case ggml_type::f8_e4m3: return dtype::f8_e4m3;
208 default: return std::nullopt;
209 }
210}
