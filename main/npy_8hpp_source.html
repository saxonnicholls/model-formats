73[[nodiscard]] constexpr std::optional<std::string_view> descr_of(dtype t) noexcept
74{
75 switch (t) {
76 case dtype::f64: return "<f8";
77 case dtype::f32: return "<f4";
78 case dtype::f16: return "<f2";
79 case dtype::i64: return "<i8";
80 case dtype::i32: return "<i4";
81 case dtype::i16: return "<i2";
82 case dtype::i8: return "|i1";
83 case dtype::u64: return "<u8";
84 case dtype::u32: return "<u4";
85 case dtype::u16: return "<u2";
86 case dtype::u8: return "|u1";
87 case dtype::boolean: return "|b1";
88 default: return std::nullopt;
89 }

94[[nodiscard]] constexpr std::optional<dtype> dtype_of_descr(std::string_view d) noexcept
95{
96 if (d.size() != 3 || (d[0] != '<' && d[0] != '=' && d[0] != '|'))
97 return std::nullopt;
98 const char canonical[3] = {d[2] == '1' ? '|' : '<', d[1], d[2]};
99 for (const dtype t : all_dtypes)
100 if (const auto s = descr_of(t); s && *s == std::string_view(canonical, 3))
101 return t;
102 return std::nullopt;
103}

107 inline constexpr std::string_view magic = "\x93NUMPY";
108 inline void skip_space(std::string_view& s)
109 {
110 while (!s.empty() && (s.front() == ' ' || s.front() == '\n'))
111 s.remove_prefix(1);
112 }
113 inline bool eat(std::string_view& s, char c)
114 {
115 skip_space(s);
116 if (s.empty() || s.front() != c)
117 return false;
118 s.remove_prefix(1);
119 return true;
120 }
121 inline std::optional<std::string_view> quoted_string(std::string_view& s)
122 {
123 skip_space(s);
124 if (s.empty() || (s.front() != '\'' && s.front() != '"'))
125 return std::nullopt;
126 const char q = s.front();
127 const auto end = s.find(q, 1);
128 if (end == std::string_view::npos)
129 return std::nullopt;
130 const auto v = s.substr(1, end - 1);
131 s.remove_prefix(end + 1);
132 return v;
133 }
134 struct header
135 {
136 std::string_view descr;
137 bool fortran = false;
138 tensor_shape shape;
139 };
140 inline result<header> parse_dict(std::string_view s)
141 {
142 header h;
143 bool have_descr = false, have_order = false, have_shape = false;
144 if (!eat(s, '{'))
145 return fail(errc::bad_header, "npy: the header is not a dict");
146 while (true) {
147 if (eat(s, '}'))
148 break;
149 const auto key = quoted_string(s);
150 if (!key || !eat(s, ':'))
151 return fail(errc::bad_header, "npy: a malformed key in the header");
152 skip_space(s);
153 if (*key == "descr") {
154 const auto v = quoted_string(s);
155 if (!v)
156 return fail(errc::unsupported, "npy: a structured dtype is not read");
157 h.descr = *v;
158 have_descr = true;
159 } else if (*key == "fortran_order") {
160 if (s.starts_with("True"))
161 h.fortran = true;
162 else if (!s.starts_with("False"))
163 return fail(errc::bad_header, "npy: fortran_order is neither True nor False");
164 s.remove_prefix(h.fortran ? 4 : 5);
165 have_order = true;
166 } else if (*key == "shape") {
167 if (!eat(s, '('))
168 return fail(errc::bad_header, "npy: shape is not a tuple");
169 while (!eat(s, ')')) {
170 skip_space(s);
171 std::uint64_t v = 0;
172 const auto [p, ec] = std::from_chars(s.data(), s.data() + s.size(), v);
173 if (ec != std::errc{})
174 return fail(errc::bad_shape, "npy: an extent is not a non-negative integer");
175 if (!h.shape.push_back(v))
176 return fail(errc::bad_shape, "npy: more than " + std::to_string(max_rank) + " dimensions");
177 s.remove_prefix(static_cast<std::size_t>(p - s.data()));
178 if (!eat(s, ',')) {
179 if (!eat(s, ')'))
180 return fail(errc::bad_header, "npy: a malformed shape");
181 break;
182 }
183 }
184 have_shape = true;
185 } else {
186 return fail(errc::bad_header, "npy: unknown header key " + quote_name(*key));
187 }
188 if (!eat(s, ',')) {
189 if (!eat(s, '}'))
190 return fail(errc::bad_header, "npy: a malformed header dict");
191 break;
192 }
193 }
194 if (!have_descr || !have_order || !have_shape)
195 return fail(errc::bad_header, "npy: the header needs descr, fortran_order and shape");
196 return ok(h);
197 }

202[[nodiscard]] inline result<tensor_view> parse_npy(std::span<const std::byte> bytes, std::string_view name = {})
203{
204 if (bytes.size() < 10)
205 return fail(errc::truncated, "npy: shorter than the header");
206 if (std::string_view(reinterpret_cast<const char*>(bytes.data()), 6) != detail::magic)
207 return fail(errc::bad_magic, "npy: no \\x93NUMPY magic");
208 const auto major = std::to_integer<int>(bytes[6]);
209 std::size_t len = 0, start = 0;
210 if (major == 1) {
211 len = load_le<std::uint16_t>(bytes.data() + 8);
212 start = 10;
213 } else if (major == 2 || major == 3) {
214 if (bytes.size() < 12)
215 return fail(errc::truncated, "npy: shorter than the header");
216 len = load_le<std::uint32_t>(bytes.data() + 8);
217 start = 12;
218 } else {
219 return fail(errc::unsupported_version, "npy: version " + std::to_string(major));
220 }
221 if (len > bytes.size() - start)
222 return fail(errc::truncated, "npy: the header runs past the end");
223 SNICHOLLS_TRY_ASSIGN(const detail::header h, detail::parse_dict(std::string_view(reinterpret_cast<const char*>(bytes.data() + start), len)));
224 if (h.fortran)
225 return fail(errc::unsupported, "npy: a Fortran-ordered array is not read");
226 if (!h.descr.empty() && h.descr[0] == '>')
227 return fail(errc::unsupported, "npy: a big-endian array is not read");
228 const auto t = dtype_of_descr(h.descr);
229 if (!t)
230 return fail(errc::bad_dtype, "npy: dtype " + quote_name(h.descr) + " is not read");
231 const auto size = checked_byte_size(h.shape, *t);
232 const std::size_t data = start + len;
233 if (!size || *size != bytes.size() - data)
234 return fail(errc::bad_shape, "npy: " + std::to_string(bytes.size() - data) + " bytes of data for shape " + to_string(h.shape));
235 return ok(tensor_view{name, *t, h.shape, bytes.subspan(data)});
236}

240[[nodiscard]] inline result<std::vector<std::byte>> npy_header(const tensor_view& t)
241{
242 const auto d = descr_of(t.type);
243 if (!d)
244 return fail(errc::bad_dtype, "npy: NumPy has no " + std::string(safetensors_name(t.type)));
245 std::string dict = "{'descr': '" + std::string(*d) + "', 'fortran_order': False, 'shape': (";
246 for (std::size_t i = 0; i < t.shape.rank; ++i)
247 dict += std::to_string(t.shape[i]) + (t.shape.rank == 1 || i + 1 < t.shape.rank ? ", " : "");
248 if (t.shape.rank == 1)
249 dict.pop_back(); // "(3,)"
250 dict += "), }";
251 // numpy's spare space, so the first extent can grow in place (GROWTH_AXIS_MAX_DIGITS = 21): byte for byte np.save
252 if (t.shape.rank > 0)
253 dict.append(21 - std::to_string(t.shape[0]).size(), ' ');
254 const bool v2 = dict.size() + 1 + 10 > 0xffff;
255 const std::size_t prefix = v2 ? 12 : 10;
256 while ((prefix + dict.size() + 1) % 64 != 0)
257 dict.push_back(' ');
258 dict.push_back('\n');
259 std::vector<std::byte> h(prefix);
260 for (std::size_t i = 0; i < 6; ++i)
261 h[i] = static_cast<std::byte>(detail::magic[i]);
262 h[6] = std::byte{static_cast<unsigned char>(v2 ? 2 : 1)};
263 h[7] = std::byte{0};
264 if (v2)
265 store_le(h.data() + 8, static_cast<std::uint32_t>(dict.size()));
266 else
267 store_le(h.data() + 8, static_cast<std::uint16_t>(dict.size()));
268 for (const char c : dict)
269 h.push_back(static_cast<std::byte>(c));
270 return ok(std::move(h));
271}

274[[nodiscard]] inline result<std::vector<std::byte>> write_npy(const tensor_view& t)
275{
276 const auto size = checked_byte_size(t.shape, t.type);
277 if (!size || *size != t.data.size())
278 return fail(errc::bad_shape, "npy: " + std::to_string(t.data.size()) + " bytes for shape " + to_string(t.shape));
279 SNICHOLLS_TRY_ASSIGN(std::vector<std::byte> out, npy_header(t));
280 out.insert(out.end(), t.data.begin(), t.data.end());
281 return ok(std::move(out));
282}

286[[nodiscard]] inline result<std::vector<tensor_view>> parse_npz(std::span<const std::byte> bytes)
287{
288 SNICHOLLS_TRY_ASSIGN(const zip::archive a, zip::parse(bytes));
289 std::vector<tensor_view> out;
290 for (const auto& e : a.entries) {
291 if (!e.name.ends_with(".npy"))
292 return fail(errc::bad_header, "npz: member " + quote_name(e.name) + " is not a .npy");
293 SNICHOLLS_TRY_ASSIGN(tensor_view v, parse_npy(e.data, e.name.substr(0, e.name.size() - 4)));
294 out.push_back(v);
295 }
296 return ok(std::move(out));
297}

301[[nodiscard]] inline result<std::vector<std::byte>> write_npz(std::span<const tensor_view> tensors)
302{
303 std::vector<std::vector<std::byte>> files;
304 std::vector<std::string> names;
305 for (const auto& t : tensors) {
306 SNICHOLLS_TRY_ASSIGN(std::vector<std::byte> f, write_npy(t));
307 files.push_back(std::move(f));
308 names.push_back(std::string(t.name) + ".npy");
309 }
310 std::vector<zip::member> members;
311 for (std::size_t i = 0; i < files.size(); ++i)
312 members.push_back({names[i], files[i]});
313 return zip::write(members);
314}
