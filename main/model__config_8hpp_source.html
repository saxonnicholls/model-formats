115[[nodiscard]] constexpr std::optional<architecture> architecture_from_hf(std::string_view model_type) noexcept
116{
117 for (const auto a : {architecture::gpt2, architecture::llama, architecture::mistral, architecture::qwen2, architecture::qwen3,
118 architecture::olmoe, architecture::deepseek_v2, architecture::deepseek_v3})
119 if (hf_model_type(a) == model_type)
120 return a;
121 return std::nullopt;
