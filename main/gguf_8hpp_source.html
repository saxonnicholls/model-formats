110[[nodiscard]] constexpr std::string_view to_string(value_type t) noexcept

119[[nodiscard]] constexpr std::size_t fixed_size(value_type t) noexcept
120{
121 switch (t) {
122 case value_type::u8: case value_type::i8: case value_type::boolean: return 1;
123 case value_type::u16: case value_type::i16: return 2;
124 case value_type::u32: case value_type::i32: case value_type::f32: return 4;
125 case value_type::u64: case value_type::i64: case value_type::f64: return 8;
126 default: return 0;
127 }

131struct value
132{
133 value_type type = value_type::u8;
134 std::span<const std::byte> bytes;
135 value_type element_type = value_type::u8;
136 std::uint64_t count = 0;

139 [[nodiscard]] std::optional<std::uint64_t> as_u64() const noexcept
140 {
141 switch (type) {
142 case value_type::u8: return std::to_integer<std::uint8_t>(bytes[0]);
143 case value_type::u16: return load_le<std::uint16_t>(bytes.data());
144 case value_type::u32: return load_le<std::uint32_t>(bytes.data());
145 case value_type::u64: return load_le<std::uint64_t>(bytes.data());
146 case value_type::i8: case value_type::i16: case value_type::i32: case value_type::i64: {
147 const auto v = as_i64();
148 return v && *v >= 0 ? std::optional<std::uint64_t>(static_cast<std::uint64_t>(*v)) : std::nullopt;
149 }
150 default: return std::nullopt;
151 }
152 }
153
154 [[nodiscard]] std::optional<std::int64_t> as_i64() const noexcept
155 {
156 switch (type) {
157 case value_type::i8: return static_cast<std::int8_t>(std::to_integer<std::uint8_t>(bytes[0]));
158 case value_type::i16: return static_cast<std::int16_t>(load_le<std::uint16_t>(bytes.data()));
159 case value_type::i32: return static_cast<std::int32_t>(load_le<std::uint32_t>(bytes.data()));
160 case value_type::i64: return static_cast<std::int64_t>(load_le<std::uint64_t>(bytes.data()));
161 case value_type::u8: case value_type::u16: case value_type::u32: case value_type::u64: {
162 const auto v = as_u64();
163 return v && *v <= static_cast<std::uint64_t>(INT64_MAX) ? std::optional<std::int64_t>(static_cast<std::int64_t>(*v)) : std::nullopt;
164 }
165 default: return std::nullopt;
166 }
167 }
168
169 [[nodiscard]] std::optional<double> as_f64() const noexcept
170 {
171 if (type == value_type::f32)
172 return static_cast<double>(std::bit_cast<float>(load_le<std::uint32_t>(bytes.data())));
173 if (type == value_type::f64)
174 return std::bit_cast<double>(load_le<std::uint64_t>(bytes.data()));
175 if (const auto i = as_i64())
176 return static_cast<double>(*i);
177 if (const auto u = as_u64())
178 return static_cast<double>(*u);
179 return std::nullopt;
180 }
181
182 [[nodiscard]] std::optional<bool> as_bool() const noexcept
183 {
184 if (type != value_type::boolean)
185 return std::nullopt;
186 return bytes[0] != std::byte{0};
187 }
188
189 [[nodiscard]] std::optional<std::string_view> as_string() const noexcept
190 {
191 if (type != value_type::string)
192 return std::nullopt;
193 return std::string_view(reinterpret_cast<const char*>(bytes.data()), bytes.size());
194 }
195
197 [[nodiscard]] std::optional<value> element(std::uint64_t i) const noexcept
198 {
199 const std::size_t s = fixed_size(element_type);
200 if (type != value_type::array || s == 0 || i >= count)
201 return std::nullopt;
202 return value{element_type, bytes.subspan(static_cast<std::size_t>(i) * s, s), value_type::u8, 0};
203 }
204
205 [[nodiscard]] std::optional<std::vector<std::string_view>> strings() const
206 {
207 if (type != value_type::array || element_type != value_type::string)
208 return std::nullopt;
209 std::vector<std::string_view> out;
210 out.reserve(static_cast<std::size_t>(count));
211 std::size_t at = 0;
212 for (std::uint64_t i = 0; i < count; ++i) { // validated by parse: every length fits
213 const auto n = static_cast<std::size_t>(load_le<std::uint64_t>(bytes.data() + at));
214 out.emplace_back(reinterpret_cast<const char*>(bytes.data() + at + 8), n);
215 at += 8 + n;
216 }
217 return out;
218 }
219
220 [[nodiscard]] std::optional<std::vector<double>> numbers() const
221 {
222 if (type != value_type::array || fixed_size(element_type) == 0 || element_type == value_type::boolean)
223 return std::nullopt;
224 std::vector<double> out;
225 out.reserve(static_cast<std::size_t>(count));
226 for (std::uint64_t i = 0; i < count; ++i)
227 out.push_back(*element(i)->as_f64());
228 return out;
229 }
230
232 [[nodiscard]] std::optional<std::vector<std::int64_t>> integers() const
233 {
234 if (type != value_type::array || fixed_size(element_type) == 0 || element_type == value_type::f32
235 || element_type == value_type::f64 || element_type == value_type::boolean)
236 return std::nullopt;
237 std::vector<std::int64_t> out;
238 out.reserve(static_cast<std::size_t>(count));
239 for (std::uint64_t i = 0; i < count; ++i) {
240 const auto v = element(i)->as_i64();
241 if (!v)
242 return std::nullopt;
243 out.push_back(*v);
244 }
245 return out;
246 }

320 class reader
321 {
322 public:
323 explicit reader(std::span<const std::byte> b) noexcept : b_(b) {}
324 std::size_t at() const noexcept { return at_; }
325 std::size_t left() const noexcept { return b_.size() - at_; }
326 bool take(std::size_t n, std::span<const std::byte>& out) noexcept
327 {
328 if (n > left())
329 return false;
330 out = b_.subspan(at_, n);
331 at_ += n;
332 return true;
333 }
334 template <typename T>
335 bool get(T& v) noexcept
336 {
337 std::span<const std::byte> s;
338 if (!take(sizeof(T), s))
339 return false;
340 v = load_le<T>(s.data());
341 return true;
342 }
343 bool string(std::string_view& s) noexcept
344 {
345 std::uint64_t n = 0;
346 std::span<const std::byte> bytes;
347 if (!get(n) || n > left() || !take(static_cast<std::size_t>(n), bytes))
348 return false;
349 s = std::string_view(reinterpret_cast<const char*>(bytes.data()), bytes.size());
350 return true;
351 }
352
353 private:
354 std::span<const std::byte> b_;
355 std::size_t at_ = 0;
356 };

358 // Reads a value of type `t` (after its type code); arrays recursively, at most `depth` deep
359 inline result<value> read_value(reader& r, value_type t, int depth)
360 {
361 value v;
362 v.type = t;
363 if (const std::size_t s = fixed_size(t)) {
364 if (!r.take(s, v.bytes))
365 return fail(errc::truncated, "gguf: a value runs past the end");
366 return ok(v);
367 }
368 if (t == value_type::string) {
369 std::string_view s;
370 if (!r.string(s))
371 return fail(errc::truncated, "gguf: a string runs past the end");
372 v.bytes = std::as_bytes(std::span(s.data(), s.size()));
373 return ok(v);
374 }
375 if (t != value_type::array)
376 return fail(errc::bad_header, "gguf: unknown value type " + std::to_string(static_cast<std::uint32_t>(t)));
377 if (depth >= 8)
378 return fail(errc::bad_header, "gguf: arrays nested more than 8 deep");
379 std::uint32_t et = 0;
380 if (!r.get(et) || !r.get(v.count))
381 return fail(errc::truncated, "gguf: an array header runs past the end");
382 if (et > 12)
383 return fail(errc::bad_header, "gguf: unknown array element type " + std::to_string(et));
384 v.element_type = static_cast<value_type>(et);
385 const std::size_t start = r.at();
386 if (const std::size_t s = fixed_size(v.element_type)) {
387 if (v.count > r.left() / s)
388 return fail(errc::truncated, "gguf: an array of " + std::to_string(v.count) + " elements runs past the end");
389 if (!r.take(static_cast<std::size_t>(v.count) * s, v.bytes))
390 return fail(errc::truncated, "gguf: an array runs past the end");
391 return ok(v);
392 }
393 // Strings and arrays: each element at least 8 bytes, so the count is bounded by what is left
394 if (v.count > r.left() / 8)
395 return fail(errc::truncated, "gguf: an array of " + std::to_string(v.count) + " elements runs past the end");
396 std::span<const std::byte> all;
397 reader copy = r;
398 for (std::uint64_t i = 0; i < v.count; ++i) {
399 SNICHOLLS_TRY_VOID(read_value(r, v.element_type, depth + 1));
400 }
401 if (!copy.take(r.at() - start, all))
402 return fail(errc::truncated, "gguf: an array runs past the end");
403 v.bytes = all;
404 return ok(v);
405 }

412[[nodiscard]] inline result<file> parse(std::span<const std::byte> bytes, const parse_options& opt = {})
413{
414 detail::reader r(bytes);
415 std::uint32_t m = 0, version = 0;
416 std::uint64_t n_tensors = 0, n_kv = 0;
417 if (!r.get(m))
418 return fail(errc::truncated, "gguf: shorter than the magic");
419 if (m != magic)
420 return fail(errc::bad_magic, "gguf: the file does not start with GGUF");
421 if (!r.get(version))
422 return fail(errc::truncated, "gguf: no version");
423 if (version != 2 && version != 3)
424 return fail(errc::unsupported_version, "gguf: version " + std::to_string(version)
425 + (version > 0xffff ? " (a big-endian file?)" : ""));
426 if (!r.get(n_tensors) || !r.get(n_kv))
427 return fail(errc::truncated, "gguf: no counts");
428 if (n_tensors > opt.max_tensors || n_kv > opt.max_metadata)
429 return fail(errc::too_large, "gguf: " + std::to_string(n_tensors) + " tensors and " + std::to_string(n_kv) + " keys");
430 if (n_kv > r.left() / 12 || n_tensors > r.left() / 24)
431 return fail(errc::truncated, "gguf: the counts are larger than the file");
432 file f;
433 f.version = version;
434 f.metadata.reserve(static_cast<std::size_t>(n_kv));
435 for (std::uint64_t i = 0; i < n_kv; ++i) {
436 metadata_entry e;
437 std::uint32_t t = 0;
438 if (!r.string(e.key) || !r.get(t))
439 return fail(errc::truncated, "gguf: metadata entry " + std::to_string(i) + " runs past the end");
440 if (t > 12)
441 return fail(errc::bad_header, "gguf: " + quote_name(e.key) + " has unknown type " + std::to_string(t));
442 SNICHOLLS_TRY_ASSIGN(e.val, detail::read_value(r, static_cast<value_type>(t), 0));
443 if (f.find(e.key))
444 return fail(errc::duplicate_name, "gguf: key " + quote_name(e.key) + " appears twice");
445 f.metadata.push_back(e);
446 }
447 if (const value* a = f.find("general.alignment")) {
448 const auto v = a->type == value_type::u32 ? a->as_u64() : std::nullopt;
449 if (!v || *v == 0 || !std::has_single_bit(*v) || *v > (1u << 20))
450 return fail(errc::bad_header, "gguf: general.alignment must be a u32 power of two");
451 f.alignment = static_cast<std::uint32_t>(*v);
452 }
453 f.tensors.reserve(static_cast<std::size_t>(n_tensors));
454 for (std::uint64_t i = 0; i < n_tensors; ++i) {
455 tensor_info ti;
456 std::uint32_t n_dims = 0, type = 0;
457 if (!r.string(ti.name) || !r.get(n_dims))
458 return fail(errc::truncated, "gguf: tensor info " + std::to_string(i) + " runs past the end");
459 if (n_dims > max_dims)
460 return fail(errc::bad_shape, "gguf: " + quote_name(ti.name) + " has " + std::to_string(n_dims) + " dimensions");
461 tensor_shape ne;
462 for (std::uint32_t d = 0; d < n_dims; ++d) {
463 std::uint64_t e = 0;
464 if (!r.get(e))
465 return fail(errc::truncated, "gguf: tensor info " + quote_name(ti.name) + " runs past the end");
466 ne.push_back(e);
467 }
468 if (!r.get(type) || !r.get(ti.offset))
469 return fail(errc::truncated, "gguf: tensor info " + quote_name(ti.name) + " runs past the end");
470 if (!traits(type))
471 return fail(errc::bad_dtype, "gguf: " + quote_name(ti.name) + " has unknown ggml type " + std::to_string(type));
472 ti.type = static_cast<ggml_type>(type);
473 ti.shape = ne.reversed();
474 f.tensors.push_back(ti);
475 }
476 {
477 std::vector<std::string_view> names;
478 for (const auto& t : f.tensors)
479 names.push_back(t.name);
480 std::sort(names.begin(), names.end());
481 if (const auto d = std::adjacent_find(names.begin(), names.end()); d != names.end())
482 return fail(errc::duplicate_name, "gguf: tensor " + quote_name(*d) + " appears twice");
483 }
484 const std::uint64_t data_start = (r.at() + f.alignment - 1) / f.alignment * f.alignment;
485 if (data_start > bytes.size() && !f.tensors.empty())
486 return fail(errc::truncated, "gguf: no data section");
487 f.data_offset = std::min<std::uint64_t>(data_start, bytes.size());
488 f.data = bytes.subspan(static_cast<std::size_t>(f.data_offset));
489 for (auto& ti : f.tensors) {
490 const tensor_shape ne = ti.ne();
491 std::optional<std::uint64_t> size = ne.rank == 0 ? row_bytes(ti.type, 1) : row_bytes(ti.type, ne[0]);
492 for (std::size_t d = 1; d < ne.rank && size; ++d)
493 size = checked_mul(*size, ne[d]);
494 if (!size)
495 return fail(errc::bad_shape, "gguf: " + quote_name(ti.name) + " " + to_string(ne) + " is not whole blocks of "
496 + std::string(traits(ti.type).name));
497 if (ti.offset % f.alignment != 0)
498 return fail(errc::misaligned, "gguf: " + quote_name(ti.name) + " is not at a multiple of " + std::to_string(f.alignment));
499 if (ti.offset > f.data.size() || *size > f.data.size() - ti.offset)
500 return fail(errc::out_of_bounds, "gguf: " + quote_name(ti.name) + " runs past the end of the data");
501 ti.data = f.data.subspan(static_cast<std::size_t>(ti.offset), static_cast<std::size_t>(*size));
502 }
503 std::vector<const tensor_info*> order;
504 for (const auto& t : f.tensors)
505 order.push_back(&t);
506 std::sort(order.begin(), order.end(), [](const tensor_info* a, const tensor_info* b) { return a->offset < b->offset; });
507 for (std::size_t i = 1; i < order.size(); ++i)
508 if (order[i]->offset < order[i - 1]->offset + order[i - 1]->data.size())
509 return fail(errc::overlap, "gguf: " + quote_name(order[i]->name) + " overlaps " + quote_name(order[i - 1]->name));
510 return ok(std::move(f));
511}

568[[nodiscard]] inline metadata_value to_owned(const value& v)
569{
570 metadata_value o;
571 o.type = v.type;
572 switch (v.type) {
573 case value_type::u8: case value_type::u16: case value_type::u32: case value_type::u64: o.u = *v.as_u64(); break;
574 case value_type::boolean: o.u = *v.as_bool() ? 1 : 0; break;
575 case value_type::i8: case value_type::i16: case value_type::i32: case value_type::i64: o.i = *v.as_i64(); break;
576 case value_type::f32: case value_type::f64: o.f = *v.as_f64(); break;
577 case value_type::string: o.s = std::string(*v.as_string()); break;
578 case value_type::array: {
579 o.element_type = v.element_type;
580 detail::reader r(v.bytes);
581 for (std::uint64_t k = 0; k < v.count; ++k)
582 o.elements.push_back(to_owned(*detail::read_value(r, v.element_type, 0)));
583 break;
584 }
585 }
586 return o;
587}

595 for (const auto c : b)

598 inline void put_string(std::vector<std::byte>& out, std::string_view s)
599 {
600 put(out, static_cast<std::uint64_t>(s.size()));
601 for (const char c : s)
602 out.push_back(static_cast<std::byte>(c));
603 }
604 inline result<void> put_value(std::vector<std::byte>& out, const metadata_value& v, int depth)
605 {
606 switch (v.type) {
607 case value_type::u8: case value_type::boolean: put(out, static_cast<std::uint8_t>(v.u)); break;
608 case value_type::i8: put(out, static_cast<std::int8_t>(v.i)); break;
609 case value_type::u16: put(out, static_cast<std::uint16_t>(v.u)); break;
610 case value_type::i16: put(out, static_cast<std::int16_t>(v.i)); break;
611 case value_type::u32: put(out, static_cast<std::uint32_t>(v.u)); break;
612 case value_type::i32: put(out, static_cast<std::int32_t>(v.i)); break;
613 case value_type::u64: put(out, v.u); break;
614 case value_type::i64: put(out, v.i); break;
615 case value_type::f32: put(out, static_cast<float>(v.f)); break;
616 case value_type::f64: put(out, v.f); break;
617 case value_type::string: put_string(out, v.s); break;
618 case value_type::array:
619 if (depth >= 8)
620 return fail(errc::bad_argument, "gguf: arrays nested more than 8 deep");
621 put(out, static_cast<std::uint32_t>(v.element_type));
622 put(out, static_cast<std::uint64_t>(v.elements.size()));
623 for (const auto& e : v.elements) {
624 if (e.type != v.element_type)
625 return fail(errc::bad_argument, "gguf: an array element of type " + std::string(to_string(e.type))
626 + " in an array of " + std::string(to_string(v.element_type)));
627 SNICHOLLS_TRY_VOID(put_value(out, e, depth + 1));
628 }
629 break;
630 default: return fail(errc::bad_argument, "gguf: unknown value type");
631 }
632 return ok();
633 }
634} // namespace detail

642 {

644 if (k == key) {
645 old = std::move(v);

657 }
658
660 void add_tensor(std::string_view name, ggml_type type, tensor_shape shape, std::span<const std::byte> data)
661 {
662 tensors_.push_back({std::string(name), type, shape, data});
663 }
664
666 [[nodiscard]] result<void> add_tensor(const tensor_view& t)
667 {
668 const auto g = from_dtype(t.type);
669 if (!g)
670 return fail(errc::bad_dtype, "gguf: no ggml type for " + std::string(safetensors_name(t.type)) + " (" + quote_name(t.name) + ")");
671 add_tensor(t.name, *g, t.shape, t.data);
672 return ok();
673 }
674
675 [[nodiscard]] std::size_t tensor_count() const noexcept { return tensors_.size(); }
676
679 template <typename Sink>
680 [[nodiscard]] result<void> write_to(Sink&& sink) const
681 {
682 std::uint32_t alignment = default_alignment;
683 if (const metadata_value* a = get("general.alignment")) {
684 if (a->type != value_type::u32 || a->u == 0 || !std::has_single_bit(a->u))
685 return fail(errc::bad_argument, "gguf: general.alignment must be a u32 power of two");
686 alignment = static_cast<std::uint32_t>(a->u);
687 }
688 std::vector<std::byte> head;
689 detail::put(head, magic);
690 detail::put(head, current_version);
691 detail::put(head, static_cast<std::uint64_t>(tensors_.size()));
692 detail::put(head, static_cast<std::uint64_t>(metadata_.size()));
693 for (const auto& [k, v] : metadata_) {
694 detail::put_string(head, k);
695 detail::put(head, static_cast<std::uint32_t>(v.type));
696 SNICHOLLS_TRY_VOID(detail::put_value(head, v, 0));
697 }
698 {
699 std::vector<std::string_view> names;
700 for (const auto& t : tensors_)
701 names.push_back(t.name);
702 std::sort(names.begin(), names.end());
703 if (const auto d = std::adjacent_find(names.begin(), names.end()); d != names.end())
704 return fail(errc::duplicate_name, "gguf: two tensors named " + quote_name(*d));
705 }
706 std::uint64_t offset = 0;
707 for (const auto& t : tensors_) {
708 if (t.shape.rank > max_dims)
709 return fail(errc::bad_shape, "gguf: " + quote_name(t.name) + " has more than 4 dimensions");
710 const tensor_shape ne = t.shape.reversed();
711 std::optional<std::uint64_t> size = row_bytes(t.type, ne.rank ? ne[0] : 1);
712 for (std::size_t d = 1; d < ne.rank && size; ++d)
713 size = checked_mul(*size, ne[d]);
714 if (!size || *size != t.data.size())
715 return fail(errc::bad_shape, "gguf: " + quote_name(t.name) + " has " + std::to_string(t.data.size()) + " bytes for "
716 + std::string(traits(t.type).name) + " " + to_string(t.shape));
717 detail::put_string(head, t.name);
718 detail::put(head, static_cast<std::uint32_t>(ne.rank));
719 for (std::size_t d = 0; d < ne.rank; ++d)
720 detail::put(head, ne[d]);
721 detail::put(head, static_cast<std::uint32_t>(t.type));
722 detail::put(head, offset);
723 offset += (*size + alignment - 1) / alignment * alignment;
724 }
725 while (head.size() % alignment != 0)
726 head.push_back(std::byte{0});
727 sink(std::span<const std::byte>(head));
728 static constexpr std::byte zeros[64]{};
729 for (const auto& t : tensors_) {
730 sink(t.data);
731 for (std::size_t pad = (alignment - t.data.size() % alignment) % alignment; pad > 0;) {
732 const std::size_t n = std::min<std::size_t>(pad, sizeof zeros);
733 sink(std::span<const std::byte>(zeros, n));
734 pad -= n;
735 }
736 }
737 return ok();
738 }
739
740 [[nodiscard]] result<std::vector<std::byte>> write() const
741 {
742 std::vector<std::byte> out;
743 SNICHOLLS_TRY_VOID(write_to([&out](std::span<const std::byte> s) { out.insert(out.end(), s.begin(), s.end()); }));
744 return ok(std::move(out));
745 }
