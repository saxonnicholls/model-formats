101[[nodiscard]] constexpr std::size_t size_of(dtype t) noexcept
102{
103 switch (t) {
104 case dtype::f64: case dtype::i64: case dtype::u64: return 8;
105 case dtype::f32: case dtype::i32: case dtype::u32: return 4;
106 case dtype::f16: case dtype::bf16: case dtype::i16: case dtype::u16: return 2;
107 case dtype::f8_e4m3: case dtype::f8_e5m2: case dtype::i8: case dtype::u8: case dtype::boolean: return 1;

152 // One element as a double (floats) or as an integer (the rest)
153 inline double load_float(dtype t, const std::byte* p) noexcept
154 {
155 switch (t) {
156 case dtype::f64: return std::bit_cast<double>(load_le<std::uint64_t>(p));
157 case dtype::f32: return static_cast<double>(std::bit_cast<float>(load_le<std::uint32_t>(p)));
158 case dtype::f16: return ml::to_double(ml::float16::from_bits(load_le<std::uint16_t>(p)));
159 case dtype::bf16: return ml::to_double(ml::bfloat16::from_bits(load_le<std::uint16_t>(p)));
160 case dtype::f8_e4m3: return ml::to_double(ml::float8_e4m3::from_bits(std::to_integer<std::uint32_t>(*p)));
161 case dtype::f8_e5m2: return ml::to_double(ml::float8_e5m2::from_bits(std::to_integer<std::uint32_t>(*p)));
162 default: return 0;
163 }
164 }
165 struct integer
166 {
167 bool negative; // value is -magnitude
168 std::uint64_t magnitude;
169 };
170 inline integer load_integer(dtype t, const std::byte* p) noexcept
171 {
172 const auto s = [](std::int64_t v) { return v < 0 ? integer{true, std::uint64_t(0) - std::uint64_t(v)} : integer{false, std::uint64_t(v)}; };
173 switch (t) {
174 case dtype::i64: return s(static_cast<std::int64_t>(load_le<std::uint64_t>(p)));
175 case dtype::i32: return s(static_cast<std::int32_t>(load_le<std::uint32_t>(p)));
176 case dtype::i16: return s(static_cast<std::int16_t>(load_le<std::uint16_t>(p)));
177 case dtype::i8: return s(static_cast<std::int8_t>(std::to_integer<std::uint8_t>(*p)));
178 case dtype::u64: return {false, load_le<std::uint64_t>(p)};
179 case dtype::u32: return {false, load_le<std::uint32_t>(p)};
180 case dtype::u16: return {false, load_le<std::uint16_t>(p)};
181 default: return {false, std::to_integer<std::uint8_t>(*p)};
182 }
183 }
184 inline double integer_as_double(integer v) noexcept
185 {
186 return v.negative ? -static_cast<double>(v.magnitude) : static_cast<double>(v.magnitude);
187 }
188 inline void store_float(dtype t, std::byte* p, double v) noexcept
189 {
190 switch (t) {
191 case dtype::f64: store_le(p, std::bit_cast<std::uint64_t>(v)); break;
192 case dtype::f32: store_le(p, std::bit_cast<std::uint32_t>(static_cast<float>(v))); break;
193 case dtype::f16: store_le(p, ml::to_minifloat<ml::float16>(v).bits); break;
194 case dtype::bf16: store_le(p, ml::to_minifloat<ml::bfloat16>(v).bits); break;
195 case dtype::f8_e4m3: *p = std::byte{ml::to_minifloat<ml::float8_e4m3>(v).bits}; break;
196 case dtype::f8_e5m2: *p = std::byte{ml::to_minifloat<ml::float8_e5m2>(v).bits}; break;
197 default: break;
198 }
199 }
200 // The range of an integer dtype, as (max magnitude when negative, max when positive)
201 inline bool fits(dtype t, integer v) noexcept
202 {
203 const std::size_t bits = 8 * size_of(t);
204 if (t == dtype::boolean)
205 return !v.negative && v.magnitude <= 1;
206 if (is_signed_integer(t)) {
207 const std::uint64_t limit = std::uint64_t{1} << (bits - 1);
208 return v.negative ? v.magnitude <= limit : v.magnitude < limit;
209 }
210 return !v.negative && (bits == 64 || v.magnitude < (std::uint64_t{1} << bits));
211 }
212 inline void store_integer(dtype t, std::byte* p, integer v) noexcept
213 {
214 const std::uint64_t bits = v.negative ? std::uint64_t(0) - v.magnitude : v.magnitude;
215 switch (size_of(t)) {
216 case 8: store_le(p, bits); break;
217 case 4: store_le(p, static_cast<std::uint32_t>(bits)); break;
218 case 2: store_le(p, static_cast<std::uint16_t>(bits)); break;
219 default: *p = static_cast<std::byte>(bits & 0xff); break;
220 }
221 }

230[[nodiscard]] inline result<void> convert_elements(dtype from, std::span<const std::byte> in, dtype to, std::span<std::byte> out)
231{
232 const std::size_t a = size_of(from), b = size_of(to);
233 if (in.size() % a != 0 || out.size() % b != 0 || in.size() / a != out.size() / b)
234 return fail(errc::bad_shape, "convert_elements: the spans hold different numbers of elements");
235 const std::size_t n = in.size() / a;
236 if (from == to) { // the fast paths: a copy, and the widenings and narrowings checkpoints use
237 std::memcpy(out.data(), in.data(), in.size());
238 return ok();
239 }
240 if (from == dtype::bf16 && to == dtype::f32) {
241 for (std::size_t i = 0; i < n; ++i) { // memcpy per element: the compiler vectorises it (a little-endian host)
242 std::uint16_t h;
243 std::memcpy(&h, in.data() + 2 * i, 2);
244 const std::uint32_t w = std::uint32_t{h} << 16;
245 std::memcpy(out.data() + 4 * i, &w, 4);
246 }
247 return ok();
248 }
249 if (from == dtype::f32 && to == dtype::bf16) {
250 for (std::size_t i = 0; i < n; ++i)
251 store_le(out.data() + 2 * i, ml::to_bfloat16(std::bit_cast<float>(load_le<std::uint32_t>(in.data() + 4 * i))).bits);
252 return ok();
253 }
254 if (to == dtype::f32 && (from == dtype::f8_e4m3 || from == dtype::f8_e5m2)) {
255 const auto& table = from == dtype::f8_e4m3 ? ml::decode_table<ml::float8_e4m3> : ml::decode_table<ml::float8_e5m2>;
256 for (std::size_t i = 0; i < n; ++i)
257 store_le(out.data() + 4 * i, std::bit_cast<std::uint32_t>(table[std::to_integer<std::uint8_t>(in[i])]));
258 return ok();
259 }
260 if (is_float(to)) {
261 for (std::size_t i = 0; i < n; ++i) {
262 const double v = is_float(from) ? detail::dtype_::load_float(from, in.data() + i * a)
263 : detail::dtype_::integer_as_double(detail::dtype_::load_integer(from, in.data() + i * a));
264 detail::dtype_::store_float(to, out.data() + i * b, v);
265 }
266 return ok();
267 }
268 if (is_float(from))
269 return fail(errc::unsupported, "convert_elements: a float to an integer is refused");
270 for (std::size_t i = 0; i < n; ++i)
271 if (!detail::dtype_::fits(to, detail::dtype_::load_integer(from, in.data() + i * a)))
272 return fail(errc::out_of_bounds, "convert_elements: element " + std::to_string(i) + " does not fit " + std::string(safetensors_name(to)));
273 for (std::size_t i = 0; i < n; ++i)
274 detail::dtype_::store_integer(to, out.data() + i * b, detail::dtype_::load_integer(from, in.data() + i * a));
275 return ok();
276}
