80 {
81 const auto v = formats::detail::u64_at(j, key);

84 inline result<std::size_t> required(const json& j, std::string_view key)
85 {
86 if (const auto v = size_at(j, key))
87 return ok(*v);
88 return fail(errc::bad_header, "config.json: no " + std::string(key));

90 inline std::optional<std::uint32_t> token_at(const json& j, std::string_view key)
91 {
92 const formats::detail::json* m = formats::detail::member(j, key);
93 if (!m)
94 return std::nullopt;
95 if (m->is_array() && !m->empty()) // generation configs list several end tokens; the first is the one
96 m = &(*m)[0];
97 const auto v = formats::detail::as_u64(*m);
98 return v && *v <= 0xffffffffu ? std::optional<std::uint32_t>(static_cast<std::uint32_t>(*v)) : std::nullopt;
99 }

101 // RoPE: base and scaling, shared by every RoPE architecture
102 inline void rope(const json& j, model_config& c)
103 {
104 c.block.rope_base = formats::detail::double_at(j, "rope_theta").value_or(10000.0);
105 const json* s = formats::detail::member(j, "rope_scaling");
106 if (!s || s->is_null())
107 return;
108 const std::string* kind = formats::detail::string_at(*s, "rope_type");
109 if (!kind)
110 kind = formats::detail::string_at(*s, "type");
111 const std::string k = kind ? *kind : "";
112 const double factor = formats::detail::double_at(*s, "factor").value_or(1.0);
113 if (k == "linear") {
114 c.block.rope_scaling = ml::rope_scaling_kind::linear;
115 c.block.rope_factor = factor;
116 } else if (k == "yarn") {
117 c.block.rope_scaling = ml::rope_scaling_kind::yarn_index_ramp;
118 c.block.rope_factor = factor;
119 c.block.rope_original_context = size_at(*s, "original_max_position_embeddings").value_or(c.context);
120 c.block.yarn_beta_fast = formats::detail::double_at(*s, "beta_fast").value_or(32.0);
121 c.block.yarn_beta_slow = formats::detail::double_at(*s, "beta_slow").value_or(1.0);
122 const double mscale = formats::detail::double_at(*s, "mscale").value_or(1.0);
123 const double mscale_all = formats::detail::double_at(*s, "mscale_all_dim").value_or(0.0);
124 c.block.yarn_mscale = mscale_all > 0 ? mscale_all : mscale;
125 if (mscale_all > 0 && mscale != mscale_all)
126 c.notes.push_back("rope_scaling: YaRN's mscale " + std::to_string(mscale) + " differs from mscale_all_dim "
127 + std::to_string(mscale_all) + "; core takes one coefficient (mscale_all_dim)");
128 } else if (!k.empty() && k != "default") {
129 c.notes.push_back("rope_scaling: \"" + k + "\" (factor " + std::to_string(factor) + ") has no counterpart in core; the frequencies are as trained");
130 }
131 }

133 inline result<ml::mlp_kind> gated_act(const json& j, std::string_view fallback)
134 {
135 const std::string* act = formats::detail::string_at(j, "hidden_act");
136 const std::string a = act ? *act : std::string(fallback);
137 if (a == "silu" || a == "swish")
138 return ok(ml::mlp_kind::swiglu);
139 return fail(errc::unsupported, "config.json: gated activation \"" + a + "\" is not core's SwiGLU");

142 // The Llama family: attention, MLP, norms, RoPE
143 inline result<void> llama_like(const json& j, model_config& c)
144 {
145 SNICHOLLS_TRY_ASSIGN(c.block.d_model, required(j, "hidden_size"));
146 SNICHOLLS_TRY_ASSIGN(c.block.n_heads, required(j, "num_attention_heads"));
147 SNICHOLLS_TRY_ASSIGN(c.layers, required(j, "num_hidden_layers"));
148 SNICHOLLS_TRY_ASSIGN(c.vocab, required(j, "vocab_size"));
149 SNICHOLLS_TRY_ASSIGN(c.block.d_ff, required(j, "intermediate_size"));
150 c.block.n_kv_heads = size_at(j, "num_key_value_heads").value_or(c.block.n_heads);
151 c.block.head_dim = size_at(j, "head_dim").value_or(c.block.n_heads ? c.block.d_model / c.block.n_heads : 0);
152 c.context = size_at(j, "max_position_embeddings").value_or(2048);
153 c.block.norm = ml::norm_kind::rms;
154 c.block.norm_eps = formats::detail::double_at(j, "rms_norm_eps").value_or(1e-6);
155 SNICHOLLS_TRY_ASSIGN(c.block.mlp, gated_act(j, "silu"));
156 c.block.rope = ml::rope_kind::half;
157 c.block.bias = formats::detail::bool_at(j, "attention_bias").value_or(false);
158 if (c.block.bias)
159 c.notes.push_back("attention_bias: biases on Q, K, V and O; core's bias puts them on the MLP too, where they are zero");
160 c.tied_embeddings = formats::detail::bool_at(j, "tie_word_embeddings").value_or(false);
161 rope(j, c);
162 return ok();
163 }

167[[nodiscard]] inline result<model_config> parse_config(std::string_view text)
168{
169 using formats::detail::bool_at;
170 using formats::detail::double_at;
171 using formats::detail::string_at;
172 SNICHOLLS_TRY_ASSIGN(const detail::json j, formats::detail::parse_json(text, "config.json"));
173 const std::string* type = string_at(j, "model_type");
174 if (!type)
175 return fail(errc::bad_header, "config.json: no model_type");
176 const auto arch = architecture_from_hf(*type);
177 if (!arch)
178 return fail(errc::unsupported, "config.json: model_type \"" + *type + "\" is not mapped onto core");
179 model_config c;
180 c.arch = *arch;
181 c.bos = detail::token_at(j, "bos_token_id");
182 c.eos = detail::token_at(j, "eos_token_id");
183 switch (c.arch) {
184 case architecture::gpt2: {
185 SNICHOLLS_TRY_ASSIGN(c.block.d_model, detail::required(j, "n_embd"));
186 SNICHOLLS_TRY_ASSIGN(c.block.n_heads, detail::required(j, "n_head"));
187 SNICHOLLS_TRY_ASSIGN(c.layers, detail::required(j, "n_layer"));
188 SNICHOLLS_TRY_ASSIGN(c.vocab, detail::required(j, "vocab_size"));
189 c.block.n_kv_heads = c.block.n_heads;
190 c.block.head_dim = c.block.n_heads ? c.block.d_model / c.block.n_heads : 0;
191 c.block.d_ff = detail::size_at(j, "n_inner").value_or(4 * c.block.d_model);
192 c.context = detail::size_at(j, "n_positions").value_or(detail::size_at(j, "n_ctx").value_or(1024));
193 c.block.norm = ml::norm_kind::layer;
194 c.block.norm_eps = double_at(j, "layer_norm_epsilon").value_or(1e-5);
195 const std::string* act = string_at(j, "activation_function");
196 const std::string a = act ? *act : "gelu_new";
197 if (a == "gelu_new" || a == "gelu_pytorch_tanh" || a == "gelu_fast")
198 c.block.mlp = ml::mlp_kind::gelu_tanh;
199 else if (a == "gelu")
200 c.block.mlp = ml::mlp_kind::gelu;
201 else
202 return fail(errc::unsupported, "config.json: GPT-2 activation \"" + a + "\" is not one of core's");
203 c.block.rope = ml::rope_kind::none;
204 c.block.bias = true;
205 c.tied_embeddings = bool_at(j, "tie_word_embeddings").value_or(true);
206 if (bool_at(j, "scale_attn_by_inverse_layer_idx").value_or(false))
207 c.notes.push_back("scale_attn_by_inverse_layer_idx: per-layer score scaling has no counterpart in core");
208 break;
209 }
210 case architecture::llama:
211 case architecture::mistral:
212 case architecture::qwen2:
213 case architecture::qwen3: {
214 SNICHOLLS_TRY_VOID(detail::llama_like(j, c));
215 if (c.arch == architecture::qwen2) {
216 c.block.bias = true; // Q, K and V carry biases whatever attention_bias says
217 c.notes.assign(1, "qwen2: biases on Q, K and V only; core's bias puts them on O and the MLP too, where they are zero");
218 }
219 if (c.arch == architecture::qwen3)
220 c.block.qk_norm = ml::qk_norm_kind::before_rope;
221 const bool sliding = c.arch == architecture::mistral || bool_at(j, "use_sliding_window").value_or(false);
222 if (sliding)
223 if (const auto w = detail::size_at(j, "sliding_window"))
224 c.block.window = *w;
225 if (c.arch == architecture::qwen2 || c.arch == architecture::qwen3)
226 if (const auto m = detail::size_at(j, "max_window_layers"); m && c.block.window && *m < c.layers)
227 c.notes.push_back("max_window_layers: the window applies to some layers only; core's is per block");
228 break;
229 }
230 case architecture::olmoe: {
231 SNICHOLLS_TRY_VOID(detail::llama_like(j, c));
232 SNICHOLLS_TRY_ASSIGN(c.block.experts, detail::required(j, "num_experts"));
233 SNICHOLLS_TRY_ASSIGN(c.block.top_k, detail::required(j, "num_experts_per_tok"));
234 c.block.expert_d_ff = c.block.d_ff;
235 c.block.scoring = bool_at(j, "norm_topk_prob").value_or(false) ? ml::moe_scoring::softmax_top_k : ml::moe_scoring::softmax;
236 c.block.balance_loss_weight = double_at(j, "router_aux_loss_coef").value_or(0.01);
237 c.block.z_loss_weight = double_at(j, "router_z_loss_coef").value_or(0.001);
238 c.notes.push_back("olmoe: q_norm and k_norm normalise the whole projection; core's qk_norm is per head, so it is left none");
239 if (const auto clip = double_at(j, "clip_qkv"); clip && *clip > 0)
240 c.notes.push_back("clip_qkv: clipping Q, K and V has no counterpart in core");
241 break;
242 }
243 case architecture::deepseek_v2:
244 case architecture::deepseek_v3: {
245 SNICHOLLS_TRY_ASSIGN(c.block.d_model, detail::required(j, "hidden_size"));
246 SNICHOLLS_TRY_ASSIGN(c.block.n_heads, detail::required(j, "num_attention_heads"));
247 SNICHOLLS_TRY_ASSIGN(c.layers, detail::required(j, "num_hidden_layers"));
248 SNICHOLLS_TRY_ASSIGN(c.vocab, detail::required(j, "vocab_size"));
249 SNICHOLLS_TRY_ASSIGN(c.block.d_ff, detail::required(j, "intermediate_size"));
250 SNICHOLLS_TRY_ASSIGN(c.block.kv_lora_rank, detail::required(j, "kv_lora_rank"));
251 SNICHOLLS_TRY_ASSIGN(c.block.qk_rope_dim, detail::required(j, "qk_rope_head_dim"));
252 SNICHOLLS_TRY_ASSIGN(const std::size_t nope, detail::required(j, "qk_nope_head_dim"));
253 SNICHOLLS_TRY_ASSIGN(c.block.v_head_dim, detail::required(j, "v_head_dim"));
254 c.block.n_kv_heads = c.block.n_heads; // latent attention: one key and value per query head
255 c.block.head_dim = nope + c.block.qk_rope_dim;
256 c.block.q_lora_rank = detail::size_at(j, "q_lora_rank").value_or(0); // null in V2-Lite: the query is projected directly
257 c.context = detail::size_at(j, "max_position_embeddings").value_or(4096);
258 c.block.norm = ml::norm_kind::rms;
259 c.block.norm_eps = double_at(j, "rms_norm_eps").value_or(1e-6);
260 SNICHOLLS_TRY_ASSIGN(c.block.mlp, detail::gated_act(j, "silu"));
261 c.block.rope = ml::rope_kind::interleaved;
262 c.tied_embeddings = bool_at(j, "tie_word_embeddings").value_or(false);
263 detail::rope(j, c);
264 if (const auto experts = detail::size_at(j, "n_routed_experts"); experts && *experts > 0) {
265 c.block.experts = *experts;
266 SNICHOLLS_TRY_ASSIGN(c.block.top_k, detail::required(j, "num_experts_per_tok"));
267 SNICHOLLS_TRY_ASSIGN(c.block.expert_d_ff, detail::required(j, "moe_intermediate_size"));
268 c.block.shared_experts = detail::size_at(j, "n_shared_experts").value_or(0);
269 c.block.route_scale = double_at(j, "routed_scaling_factor").value_or(1.0);
270 c.block.balance_loss_weight = double_at(j, "aux_loss_alpha").value_or(0.0);
271 c.leading_dense_layers = detail::size_at(j, "first_k_dense_replace").value_or(0);
272 const std::string* scoring = string_at(j, "scoring_func");
273 const std::string s = scoring ? *scoring : "softmax";
274 const bool renormalise = bool_at(j, "norm_topk_prob").value_or(false);
275 if (s == "sigmoid") {
276 c.block.scoring = ml::moe_scoring::sigmoid;
277 if (!renormalise)
278 c.notes.push_back("norm_topk_prob false with sigmoid scoring: core's sigmoid weights are always renormalised");
279 } else if (s == "softmax") {
280 c.block.scoring = renormalise ? ml::moe_scoring::softmax_top_k : ml::moe_scoring::softmax;
281 } else {
282 return fail(errc::unsupported, "config.json: scoring_func \"" + s + "\" is not one of core's");
283 }
284 const std::string* method = string_at(j, "topk_method");
285 if (method && (*method == "group_limited_greedy" || *method == "noaux_tc")) {
286 c.block.expert_groups = detail::size_at(j, "n_group").value_or(0);
287 c.block.group_top_k = detail::size_at(j, "topk_group").value_or(0);
288 }
289 if (method && *method == "noaux_tc")
290 c.notes.push_back("noaux_tc: the selection bias's update rate is not in config.json (DeepSeek-V3 used 0.001); left 0");
291 }
292 c.block.mtp_depth = detail::size_at(j, "num_nextn_predict_layers").value_or(0);
293 break;
294 }
295 }
296 if (!ml::valid(c.block))
297 return fail(errc::bad_header, "config.json: the " + *type + " configuration is not one core accepts (ml::valid)");
298 return ok(std::move(c));
299}

317[[nodiscard]] inline result<generation_config> parse_generation_config(std::string_view text)
318{
319 SNICHOLLS_TRY_ASSIGN(const detail::json j, formats::detail::parse_json(text, "generation_config.json"));
320 if (!j.is_object())
321 return fail(errc::bad_json, "generation_config.json: not an object");
322 generation_config g;
323 g.bos = detail::token_at(j, "bos_token_id");
324 g.pad = detail::token_at(j, "pad_token_id");
325 if (const detail::json* e = formats::detail::member(j, "eos_token_id")) {
326 if (e->is_array()) {
327 for (const auto& t : *e)
328 if (const auto v = formats::detail::as_u64(t); v && *v <= 0xffffffffu)
329 g.eos.push_back(static_cast<std::uint32_t>(*v));
330 } else if (const auto v = formats::detail::as_u64(*e); v && *v <= 0xffffffffu) {
331 g.eos.push_back(static_cast<std::uint32_t>(*v));
332 }
333 }
334 g.do_sample = formats::detail::bool_at(j, "do_sample");
335 g.temperature = formats::detail::double_at(j, "temperature");
336 g.top_p = formats::detail::double_at(j, "top_p");
337 g.top_k = detail::size_at(j, "top_k");
338 g.repetition_penalty = formats::detail::double_at(j, "repetition_penalty");
339 g.max_new_tokens = detail::size_at(j, "max_new_tokens");
340 g.max_length = detail::size_at(j, "max_length");
341 return ok(std::move(g));
342}
