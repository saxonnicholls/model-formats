96 // A 2-D [rows, cols] tensor transposed to [cols, rows]
97 inline tensor transposed(const tensor_view& t, std::string name)
98 {
99 const std::size_t rows = static_cast<std::size_t>(t.shape[0]), cols = static_cast<std::size_t>(t.shape[1]), es = size_of(t.type);
100 tensor out{std::move(name), t.type, tensor_shape{cols, rows}, std::vector<std::byte>(t.data.size())};
101 for (std::size_t r = 0; r < rows; ++r)
102 for (std::size_t c = 0; c < cols; ++c)
103 std::memcpy(out.data.data() + (c * rows + r) * es, t.data.data() + (r * cols + c) * es, es);
104 return out;
105 }
106 inline tensor_view slice(const tensor_view& t, std::string_view name, std::size_t index) // t[index] of a stacked tensor
107 {
108 tensor_shape s;
109 for (std::size_t d = 1; d < t.shape.rank; ++d)
110 s.push_back(t.shape[d]);
111 const auto n = static_cast<std::size_t>(*s.elements()) * size_of(t.type);
112 return {name, t.type, s, t.data.subspan(index * n, n)};
113 }

118[[nodiscard]] inline result<converted> llmc_to_hf(const llmc::checkpoint& c)
119{
120 converted out;
121 const auto& s = c.shape;
122 const std::size_t es = size_of(c.type), C = s.channels;
123 const auto& t = c.tensors;
124 const auto add = [&](std::string name, tensor_view v) {
125 out.owned.push_back(tensor{std::move(name), v.type, v.shape, {}});
126 out.tensors.push_back(tensor_view{out.owned.back().name, v.type, v.shape, v.data});
127 };
128 add("wte.weight", tensor_view{"", c.type, {s.vocab, C}, t[0].data.first(s.vocab * C * es)});
129 add("wpe.weight", t[1]);
130 for (std::size_t l = 0; l < s.layers; ++l) {
131 const std::string h = "h." + std::to_string(l) + ".";
132 add(h + "ln_1.weight", detail::slice(t[2], "", l));
133 add(h + "ln_1.bias", detail::slice(t[3], "", l));
134 out.owned.push_back(detail::transposed(detail::slice(t[4], "", l), h + "attn.c_attn.weight"));
135 out.tensors.push_back(out.owned.back().view());
136 add(h + "attn.c_attn.bias", detail::slice(t[5], "", l));
137 out.owned.push_back(detail::transposed(detail::slice(t[6], "", l), h + "attn.c_proj.weight"));
138 out.tensors.push_back(out.owned.back().view());
139 add(h + "attn.c_proj.bias", detail::slice(t[7], "", l));
140 add(h + "ln_2.weight", detail::slice(t[8], "", l));
141 add(h + "ln_2.bias", detail::slice(t[9], "", l));
142 out.owned.push_back(detail::transposed(detail::slice(t[10], "", l), h + "mlp.c_fc.weight"));
143 out.tensors.push_back(out.owned.back().view());
144 add(h + "mlp.c_fc.bias", detail::slice(t[11], "", l));
145 out.owned.push_back(detail::transposed(detail::slice(t[12], "", l), h + "mlp.c_proj.weight"));
146 out.tensors.push_back(out.owned.back().view());
147 add(h + "mlp.c_proj.bias", detail::slice(t[13], "", l));
148 }
149 add("ln_f.weight", t[14]);
150 add("ln_f.bias", t[15]);
151 if (s.padded_vocab > s.vocab)
152 out.notes.push_back("wte: the " + std::to_string(s.padded_vocab - s.vocab) + " padding rows are left out");
153 return ok(std::move(out));
154}

170[[nodiscard]] inline result<gguf_export> llmc_to_gguf(const llmc::checkpoint& c, const vocabulary* vocab, std::string_view name)
171{
172 gguf_export out;
173 const auto& s = c.shape;
174 const auto g = gguf::from_dtype(c.type);
175 model_config m;
176 m.arch = architecture::gpt2;
177 m.block = ml::gpt2_small;
178 m.block.d_model = s.channels;
179 m.block.n_heads = m.block.n_kv_heads = s.heads;
180 m.block.head_dim = s.channels / s.heads;
181 m.block.d_ff = 4 * s.channels;
182 m.layers = s.layers;
183 m.context = s.max_seq;
184 if (vocab) {
185 if (vocab->tokens.size() != s.vocab)
186 return fail(errc::bad_shape, "llmc_to_gguf: the vocabulary has " + std::to_string(vocab->tokens.size()) + " tokens, the model "
187 + std::to_string(s.vocab));
188 m.bos = vocab->bos;
189 m.eos = vocab->eos;
190 }
191 SNICHOLLS_TRY_VOID(gguf::write_model_config(out.writer, m, name));
192 out.writer.set(gguf::general::file_type, gguf::value_u32(static_cast<std::uint32_t>(c.type == dtype::bf16 ? gguf::file_type::mostly_bf16
193 : gguf::file_type::all_f32)));
194 if (vocab) {
195 vocabulary v = *vocab;
196 if (v.pre.empty())
197 v.pre = "gpt-2";
198 gguf::write_vocabulary(out.writer, v);
199 }
200 const std::size_t es = size_of(c.type), C = s.channels;
201 const auto& t = c.tensors;
202 const auto add = [&](const std::string& n, tensor_view v) { out.writer.add_tensor(n, *g, v.shape, v.data); }; // views of the checkpoint
203 add("token_embd.weight", tensor_view{"", c.type, {s.vocab, C}, t[0].data.first(s.vocab * C * es)});
204 add("position_embd.weight", t[1]);
205 for (std::size_t l = 0; l < s.layers; ++l) {
206 const std::string b = "blk." + std::to_string(l) + ".";
207 add(b + "attn_norm.weight", detail::slice(t[2], "", l));
208 add(b + "attn_norm.bias", detail::slice(t[3], "", l));
209 add(b + "attn_qkv.weight", detail::slice(t[4], "", l));
210 add(b + "attn_qkv.bias", detail::slice(t[5], "", l));
211 add(b + "attn_output.weight", detail::slice(t[6], "", l));
212 add(b + "attn_output.bias", detail::slice(t[7], "", l));
213 add(b + "ffn_norm.weight", detail::slice(t[8], "", l));
214 add(b + "ffn_norm.bias", detail::slice(t[9], "", l));
215 add(b + "ffn_up.weight", detail::slice(t[10], "", l));
216 add(b + "ffn_up.bias", detail::slice(t[11], "", l));
217 add(b + "ffn_down.weight", detail::slice(t[12], "", l));
218 add(b + "ffn_down.bias", detail::slice(t[13], "", l));
219 }
220 add("output_norm.weight", t[14]);
221 add("output_norm.bias", t[15]);
222 out.notes.push_back("output.weight is not written: llama.cpp ties it to token_embd");
223 return ok(std::move(out));
224}

229[[nodiscard]] inline result<gguf_export> hf_to_gguf(const model_config& c, std::span<const tensor_view> tensors, const vocabulary* vocab,
230 std::string_view chat_template, std::string_view name)
231{
232 if (c.arch == architecture::deepseek_v2 || c.arch == architecture::deepseek_v3)
233 return fail(errc::unsupported, "hf_to_gguf: DeepSeek's latent attention is split by llama.cpp's converter in a way not done here");
234 gguf_export out;
235 SNICHOLLS_TRY_VOID(gguf::write_model_config(out.writer, c, name));
236 if (vocab) {
237 vocabulary v = *vocab;
238 if (v.pre.empty())
239 v.pre = c.arch == architecture::gpt2 ? "gpt-2" : "default";
240 gguf::write_vocabulary(out.writer, v, chat_template);
241 }
242 // routed experts, gathered per layer and role, stacked after the loop
243 std::map<std::pair<std::size_t, int>, std::vector<const tensor_view*>> experts;
244 std::optional<dtype> common;
245 for (const auto& t : tensors) {
246 const auto n = hf::parse_hf_name(c.arch, t.name);
247 if (!n)
248 return fail(errc::unsupported, "hf_to_gguf: no rule names " + quote_name(t.name));
249 if (n->what == hf::role::mask_buffer) {
250 out.notes.push_back(std::string(t.name) + ": a mask buffer, dropped");
251 continue;
252 }
253 if (n->what == hf::role::lm_head && c.tied_embeddings) {
254 out.notes.push_back(std::string(t.name) + ": tied to the token embedding, dropped");
255 continue;
256 }
257 const auto g = gguf::from_dtype(t.type);
258 if (!g)
259 return fail(errc::bad_dtype, "hf_to_gguf: " + quote_name(t.name) + " is " + std::string(safetensors_name(t.type)));
260 common = common ? common : t.type;
261 if (n->expert) {
262 auto& list = experts[{*n->layer, static_cast<int>(n->what)}];
263 if (list.size() <= *n->expert)
264 list.resize(*n->expert + 1, nullptr);
265 list[*n->expert] = &t;
266 continue;
267 }
268 const std::string gname = hf::gguf_name(*n);
269 if (hf::hf_transposed(c.arch, *n)) {
270 if (t.shape.rank != 2)
271 return fail(errc::bad_shape, "hf_to_gguf: " + quote_name(t.name) + " is not a matrix");
272 out.owned.push_back(detail::transposed(t, gname));
273 out.writer.add_tensor(out.owned.back().name, *g, out.owned.back().shape, out.owned.back().data);
274 } else if (hf::gguf_permutes(c.arch, *n)) {
275 const std::size_t heads = n->what == hf::role::q ? c.block.n_heads : c.block.n_kv_heads;
276 const std::size_t rows = static_cast<std::size_t>(t.shape[0]);
277 if (rows != heads * c.block.head_dim)
278 return fail(errc::bad_shape, "hf_to_gguf: " + quote_name(t.name) + " has " + std::to_string(rows) + " rows for "
279 + std::to_string(heads) + " heads of " + std::to_string(c.block.head_dim));
280 out.owned.push_back(tensor{gname, t.type, t.shape, std::vector<std::byte>(t.data.size())});
281 hf::permute_rope(t.data, out.owned.back().data, heads, t.data.size() / rows);
282 out.writer.add_tensor(out.owned.back().name, *g, t.shape, out.owned.back().data);
283 } else {
284 out.owned.push_back(tensor{gname, t.type, t.shape, {}});
285 out.writer.add_tensor(out.owned.back().name, *g, t.shape, t.data);
286 }
287 }
288 for (const auto& [key, list] : experts) {
289 if (list.size() != c.block.experts)
290 return fail(errc::not_found, "hf_to_gguf: layer " + std::to_string(key.first) + " has " + std::to_string(list.size()) + " of "
291 + std::to_string(c.block.experts) + " experts");
292 tensor stacked;
293 for (std::size_t e = 0; e < list.size(); ++e) {
294 if (!list[e])
295 return fail(errc::not_found, "hf_to_gguf: layer " + std::to_string(key.first) + " has no expert " + std::to_string(e));
296 if (list[e]->shape != list[0]->shape || list[e]->type != list[0]->type)
297 return fail(errc::bad_shape, "hf_to_gguf: the experts of layer " + std::to_string(key.first) + " differ");
298 stacked.data.insert(stacked.data.end(), list[e]->data.begin(), list[e]->data.end());
299 }
300 hf::tensor_name n{static_cast<hf::role>(key.second), key.first, {}, false};
301 stacked.name = hf::gguf_name(n);
302 stacked.type = list[0]->type;
303 stacked.shape = tensor_shape{list.size()};
304 for (const auto d : list[0]->shape.view())
305 stacked.shape.push_back(d);
306 out.owned.push_back(std::move(stacked));
307 out.writer.add_tensor(out.owned.back().name, *gguf::from_dtype(out.owned.back().type), out.owned.back().shape, out.owned.back().data);
308 }
309 if (common)
310 out.writer.set(gguf::general::file_type,
311 gguf::value_u32(static_cast<std::uint32_t>(*common == dtype::bf16 ? gguf::file_type::mostly_bf16
312 : *common == dtype::f16 ? gguf::file_type::mostly_f16
313 : gguf::file_type::all_f32)));
314 return ok(std::move(out));
315}

320[[nodiscard]] inline result<converted> gguf_to_hf(const gguf::file& f, const model_config& c)
321{
322 converted out;
323 for (const auto& t : f.tensors) {
324 const auto n = hf::parse_gguf_name(t.name);
325 if (!n)
326 return fail(errc::unsupported, "gguf_to_hf: no rule names " + quote_name(t.name));
327 const auto v = t.dense();
328 if (!v)
329 return fail(errc::unsupported, "gguf_to_hf: " + quote_name(t.name) + " is quantised; dequantize it first");
330 if (n->what == hf::role::expert_gate || n->what == hf::role::expert_up || n->what == hf::role::expert_down) {
331 for (std::size_t e = 0; e < t.shape[0]; ++e) {
332 hf::tensor_name one = *n;
333 one.expert = e;
334 out.owned.push_back(tensor::from(detail::slice(*v, "", e)));
335 out.owned.back().name = hf::hf_name(c.arch, one);
336 out.tensors.push_back(out.owned.back().view());
337 }
338 continue;
339 }
340 const std::string hname = hf::hf_name(c.arch, *n);
341 if (hname.empty())
342 return fail(errc::unsupported, "gguf_to_hf: " + quote_name(t.name) + " has no Hugging Face name in " + std::string(hf_model_type(c.arch)));
343 if (hf::hf_transposed(c.arch, *n)) {
344 out.owned.push_back(detail::transposed(*v, hname));
345 } else if (hf::gguf_permutes(c.arch, *n)) {
346 const std::size_t heads = n->what == hf::role::q ? c.block.n_heads : c.block.n_kv_heads;
347 out.owned.push_back(tensor{hname, v->type, v->shape, std::vector<std::byte>(v->data.size())});
348 hf::unpermute_rope(v->data, out.owned.back().data, heads, v->data.size() / static_cast<std::size_t>(v->shape[0]));
349 } else {
350 out.owned.push_back(tensor{hname, v->type, v->shape, {}});
351 out.tensors.push_back(tensor_view{out.owned.back().name, v->type, v->shape, v->data});
352 continue;
353 }
354 out.tensors.push_back(out.owned.back().view());
355 }
356 return ok(std::move(out));
357}
