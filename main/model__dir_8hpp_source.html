105[[nodiscard]] inline result<model_dir> open_model_dir(const std::filesystem::path& root)
106{
107 model_dir m;
108 m.root = root;
109 const auto cfg = detail::read_text(root / "config.json");
110 if (!cfg)
111 return fail(errc::not_found, root.string() + ": no config.json");
112 auto c = hf::parse_config(*cfg);
113 if (c.is_ok())
114 m.config = std::move(*c);
115 else
116 m.config_error = c.unwrap_err_unchecked();
117 if (const auto g = detail::read_text(root / "generation_config.json")) {
118 SNICHOLLS_TRY_ASSIGN(m.generation, hf::parse_generation_config(*g));
119 }
120 if (const auto t = detail::read_text(root / "tokenizer.json")) {
121 SNICHOLLS_TRY_ASSIGN(hf::tokenizer tok, hf::parse_tokenizer(*t));
122 if (const auto tc = detail::read_text(root / "tokenizer_config.json"))
123 SNICHOLLS_TRY_VOID(hf::parse_tokenizer_config(tok, *tc));
124 if (const auto j = detail::read_text(root / "chat_template.jinja"))
125 tok.chat_templates.insert(tok.chat_templates.begin(), hf::chat_template{"default", *j});
126 m.tokenizer = std::move(tok);
127 }
128 // The weights: safetensors first, PyTorch only when there is none
129 std::optional<safetensors::index> ix;
130 std::vector<std::string> shards;
131 for (const char* index_name : {"model.safetensors.index.json", "pytorch_model.bin.index.json"}) {
132 if (const auto text = detail::read_text(root / index_name)) {
133 SNICHOLLS_TRY_ASSIGN(ix, safetensors::parse_index(*text));
134 shards = ix->files();
135 break;
136 }
137 const std::string single = std::string(index_name).substr(0, std::string(index_name).size() - 11); // without ".index.json"
138 if (std::filesystem::exists(root / single)) {
139 shards = {single};
140 break;
141 }
142 }
143 if (shards.empty())
144 return fail(errc::not_found, root.string() + ": no model.safetensors, pytorch_model.bin or index");
145 std::set<std::string_view> seen;
146 for (const auto& s : shards) {
147 SNICHOLLS_TRY_ASSIGN(mapped_file f, mapped_file::open(root / s));
148 m.files.push_back(std::move(f));
149 auto l = convert::read_any(m.files.back().bytes());
150 if (l.is_err())
151 return err(error{l.unwrap_err_unchecked().code, (root / s).string() + ": " + l.unwrap_err_unchecked().message});
152 m.parts.push_back(std::move(*l));
153 for (const auto& t : m.parts.back().tensors) {
154 if (!seen.insert(t.name).second)
155 return fail(errc::duplicate_name, root.string() + ": " + quote_name(t.name) + " is in two shards");
156 if (ix) {
157 const std::string* where = ix->shard_of(t.name);
158 if (!where || *where != s)
159 return fail(errc::bad_header, root.string() + ": " + quote_name(t.name) + " is in " + s + ", the index says "
160 + (where ? *where : std::string("nowhere")));
161 }
162 m.tensors.push_back(t);
163 }
164 }
165 if (ix)
166 for (const auto& [t, s] : ix->weight_map)
167 if (!seen.contains(t))
168 return fail(errc::bad_header, root.string() + ": the index names " + quote_name(t) + " in " + s + ", which lacks it");
169 return ok(std::move(m));
170}
