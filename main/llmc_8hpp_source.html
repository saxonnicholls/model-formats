123[[nodiscard]] inline result<checkpoint> parse_checkpoint(std::span<const std::byte> bytes)
124{
125 if (bytes.size() < header_ints * 4)
126 return fail(errc::truncated, "llm.c checkpoint: shorter than the header");
127 const auto h = [&](std::size_t i) { return static_cast<std::int32_t>(load_le<std::uint32_t>(bytes.data() + 4 * i)); };
128 if (h(0) != checkpoint_magic)
129 return fail(errc::bad_magic, "llm.c checkpoint: magic " + std::to_string(h(0)) + ", not 20240326");
130 checkpoint c;
131 if (h(1) == 3)
132 c.type = dtype::f32;
133 else if (h(1) == 5)
134 c.type = dtype::bf16;
135 else
136 return fail(errc::unsupported_version, "llm.c checkpoint: version " + std::to_string(h(1)));
137 for (int i = 2; i <= 7; ++i)
138 if (h(static_cast<std::size_t>(i)) <= 0)
139 return fail(errc::bad_header, "llm.c checkpoint: header field " + std::to_string(i) + " is not positive");
140 c.shape = {std::size_t(h(2)), std::size_t(h(3)), std::size_t(h(4)), std::size_t(h(5)), std::size_t(h(6)), std::size_t(h(7))};
141 if (c.shape.channels % c.shape.heads != 0 || c.shape.padded_vocab < c.shape.vocab)
142 return fail(errc::bad_header, "llm.c checkpoint: heads must divide channels and the padded vocabulary cover the vocabulary");
143 const std::uint64_t size = parameter_count(c.shape) * size_of(c.type);
144 if (size != bytes.size() - header_ints * 4)
145 return fail(errc::bad_shape, "llm.c checkpoint: " + std::to_string(bytes.size() - header_ints * 4) + " bytes of parameters, "
146 + std::to_string(size) + " expected");
147 c.parameters = bytes.subspan(header_ints * 4);
148 const auto shapes = tensor_shapes(c.shape);
149 std::size_t at = 0;
150 for (std::size_t i = 0; i < 16; ++i) {
151 const auto n = static_cast<std::size_t>(*shapes[i].elements()) * size_of(c.type);
152 c.tensors[i] = tensor_view{tensor_names[i], c.type, shapes[i], c.parameters.subspan(at, n)};
153 at += n;
154 }
155 return ok(c);
156}

160[[nodiscard]] inline result<std::vector<std::byte>> write_checkpoint(const gpt2_shape& s, dtype type, std::span<const std::byte> parameters)
161{
162 if (type != dtype::f32 && type != dtype::bf16)
163 return fail(errc::bad_dtype, "llm.c checkpoint: only f32 and bf16 parameters");
164 if (s.heads == 0 || s.channels % s.heads != 0 || s.padded_vocab < s.vocab || s.layers == 0 || s.max_seq == 0)
165 return fail(errc::bad_header, "llm.c checkpoint: the sizes are not a GPT-2");
166 for (const std::size_t v : {s.max_seq, s.vocab, s.layers, s.heads, s.channels, s.padded_vocab})
167 if (v > 0x7fffffff)
168 return fail(errc::bad_header, "llm.c checkpoint: a size does not fit the int32 header");
169 if (parameters.size() != parameter_count(s) * size_of(type))
170 return fail(errc::bad_shape, "llm.c checkpoint: " + std::to_string(parameters.size()) + " bytes of parameters for this shape");
171 std::vector<std::byte> out(header_ints * 4);
172 const std::array<std::int32_t, 8> head{checkpoint_magic, type == dtype::f32 ? 3 : 5, std::int32_t(s.max_seq), std::int32_t(s.vocab),
173 std::int32_t(s.layers), std::int32_t(s.heads), std::int32_t(s.channels), std::int32_t(s.padded_vocab)};
174 for (std::size_t i = 0; i < head.size(); ++i)
175 store_le(out.data() + 4 * i, static_cast<std::uint32_t>(head[i]));
176 out.insert(out.end(), parameters.begin(), parameters.end());
177 return ok(std::move(out));
178}

182[[nodiscard]] inline result<std::span<const std::byte>> parse_tokens(std::span<const std::byte> bytes)
183{
184 if (bytes.size() < header_ints * 4)
185 return fail(errc::truncated, "llm.c tokens: shorter than the header");
186 const auto h = [&](std::size_t i) { return load_le<std::uint32_t>(bytes.data() + 4 * i); };
187 if (static_cast<std::int32_t>(h(0)) != tokens_magic)
188 return fail(errc::bad_magic, "llm.c tokens: magic " + std::to_string(h(0)) + ", not 20240520");
189 if (h(1) != 1)
190 return fail(errc::unsupported_version, "llm.c tokens: version " + std::to_string(h(1)));
191 const std::uint64_t n = h(2);
192 if (n * 2 != bytes.size() - header_ints * 4)
193 return fail(errc::bad_shape, "llm.c tokens: the header says " + std::to_string(n) + " tokens");
194 return ok(bytes.subspan(header_ints * 4));
195}

198[[nodiscard]] inline result<std::vector<std::byte>> write_tokens(std::span<const std::uint16_t> tokens)
199{
200 if (tokens.size() > 0xffffffffu)
201 return fail(errc::too_large, "llm.c tokens: more than 2^32 - 1 tokens in a shard");
202 std::vector<std::byte> out(header_ints * 4 + tokens.size() * 2);
203 store_le(out.data(), static_cast<std::uint32_t>(tokens_magic));
204 store_le(out.data() + 4, std::uint32_t{1});
205 store_le(out.data() + 8, static_cast<std::uint32_t>(tokens.size()));
206 for (std::size_t i = 0; i < tokens.size(); ++i)
207 store_le(out.data() + header_ints * 4 + 2 * i, tokens[i]);
208 return ok(std::move(out));
209}
